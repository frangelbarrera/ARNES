    27→    async def complete(
    28→        self,
    29→        messages: list[LLMMessage],
    30→        *,
    31→        model: str = "mock",
    32→        tools: list | None = None,
    33→        temperature: float = 0.0,
    34→        max_tokens: int | None = None,
    35→        response_format: dict | None = None,
    36→        response_schema: dict | None = None,
    37→        **kwargs,
    38→    ) -> LLMResponse:
    39→        self.call_count += 1
    40→
    41→        # Detect which specialist based on system prompt content
    42→        sys_msg = next((m for m in messages if m.role == "system"), None)
    43→        sys_content = sys_msg.content if sys_msg else ""
    44→
    45→        if "@planner" in sys_content:
    46→            content = '{"steps": [{"id": "s1", "specialist": "@coder", "input": {}}]}'
    47→        elif "@coder" in sys_content:
    48→            content = '{"files": [{"path": "out.py", "language": "python", "content": "pass"}], "summary": "ok", "assumptions": [], "warnings": []}'
    49→        elif "@reviewer" in sys_content:
    50→            content = '{"verdict": "approve", "issues": [], "summary": "LGTM"}'
    51→        elif "@tester" in sys_content:
    52→            content = '{"test_files": [{"path": "test.py", "content": "pass"}], "test_results": {"passed": 1, "failed": 0, "skipped": 0, "failures": []}, "summary": "ok"}'
    53→        elif "@debugger" in sys_content:

    55→        else:
    56→            content = '{"result": "ok"}'
    57→
    58→        tokens_in = sum(len(m.content) // 4 for m in messages)
    59→        tokens_out = len(content) // 4
    60→
    61→        return LLMResponse(
    62→            content=content,
    63→            tool_calls=[],
    64→            usage=LLMUsage(
    65→                tokens_in=tokens_in,
    66→                tokens_out=tokens_out,

   180→    async def test_parallel_branch_emits_started_and_completed_events(self, executor):
   181→        """FIX-R4-DATA: PARALLEL_BRANCH_STARTED and PARALLEL_BRANCH_COMPLETED
   182→        were previously defined in EventType but never instantiated. The
   183→        executor must now emit them around the asyncio.gather call so the
   184→        audit log marks the parallel-block boundaries."""
   185→        from arnes.thread.events import EventType
   186→
   187→        yaml_str = """
   188→name: parallel_events
   189→objective: Test parallel branch events
   190→steps:
   191→  - id: parallel
   192→    parallel:
   193→      - id: sub1
   194→        specialist: "@planner"
   195→        input: {task: "Subtask 1"}
   196→      - id: sub2
   197→        specialist: "@coder"
   198→        input: {spec: "Subtask 2"}
   199→"""
   200→        playbook = PlaybookCompiler.from_string(yaml_str)
   201→        result = await executor.run(playbook)
   202→
   203→        assert result.success is True, f"Failed: {result.error}"
   204→
   205→        started = [e for e in result.thread if e.type == EventType.PARALLEL_BRANCH_STARTED]
   206→        completed = [e for e in result.thread if e.type == EventType.PARALLEL_BRANCH_COMPLETED]
   207→        assert len(started) == 1, (
   208→            f"Expected 1 PARALLEL_BRANCH_STARTED event, got {len(started)}. "
   209→            f"Event types: {[(e.type.value) for e in result.thread]}"
   210→        )
   211→        assert len(completed) == 1, (
   212→            f"Expected 1 PARALLEL_BRANCH_COMPLETED event, got {len(completed)}"
   213→        )
   214→
   215→        # STARTED carries the sub-step ids so the audit log shows what
   216→        # branches were launched.
   217→        s = started[0]
   218→        assert s.step_id == "parallel"
   219→        assert s.data["sub_step_ids"] == ["sub1", "sub2"]
   220→        assert s.data["sub_step_count"] == 2
   221→
   222→        # COMPLETED carries the per-sub-step outcome.
   223→        c = completed[0]
   224→        assert c.step_id == "parallel"
   225→        assert c.data["all_success"] is True
   226→        outcomes = {o["sub_step_id"]: o for o in c.data["sub_step_outcomes"]}
   227→        assert "sub1" in outcomes
   228→        assert "sub2" in outcomes
   229→        assert outcomes["sub1"]["success"] is True
   230→        assert outcomes["sub2"]["success"] is True
   231→
   232→        # STARTED must come before COMPLETED in the thread (audit order).
   233→        started_idx = result.thread.events.index(started[0])
   234→        completed_idx = result.thread.events.index(completed[0])
   235→        assert started_idx < completed_idx, (
   236→            f"PARALLEL_BRANCH_STARTED (idx={started_idx}) must come before "
   237→            f"PARALLEL_BRANCH_COMPLETED (idx={completed_idx}) in the thread"
   238→        )
