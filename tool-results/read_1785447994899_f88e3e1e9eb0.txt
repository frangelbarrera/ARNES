     1→"""
     2→ARNES Playbook Executor — runs a compiled Playbook as a DAG.
     3→
     4→The executor:
     5→1. Walks the playbook steps in order.
     6→2. For each step, invokes the specialist or tool.
     7→3. Applies conditional branches (if/elif/else).
     8→4. Runs parallel branches concurrently.
     9→5. Handles retries with backoff.
    10→6. Pauses at HITL gates.
    11→7. Tracks budget via CostGuard.
    12→8. Appends events to the Thread.
    13→9. Returns a PlaybookRunResult with full trace.
    14→
    15→The executor is async and supports both fire-and-forget and streaming modes.
    16→"""

    58→def _is_docker_available() -> bool:
    59→    """Return True if the ``docker`` CLI is on PATH.
    60→
    61→    Used by the executor to decide whether to wire the Docker sandbox into
    62→    the default ``ToolContext``. This is a presence check only — it does NOT
    63→    verify the daemon is running or that ``arnes-sandbox:latest`` exists.
    64→    The ``ShellTool`` surfaces a clear error if either is missing at
    65→    execution time (``FileNotFoundError`` on ``docker run``).
    66→
    67→    We deliberately avoid probing the daemon (``docker info`` / ``docker
    68→    version``) here because:
    69→
    70→    1. It spawns a subprocess on every ``PlaybookExecutor`` construction,
    71→       which is wasteful for tests and high-throughput runs.
    72→    2. The daemon may be temporarily down even if the CLI is installed —
    73→       failing fast at construction time would prevent the user from
    74→       running non-shell playbooks that don't need Docker at all.
    75→    3. The ``ShellTool._execute_in_sandbox`` already handles the
    76→       ``FileNotFoundError`` case (docker binary missing) and returns a
    77→       actionable error message.
    78→    """
    79→    return shutil.which("docker") is not None

   103→class PlaybookExecutor:
   104→    """Executes a compiled Playbook.
   105→
   106→    Usage:
   107→        playbook = PlaybookCompiler.from_file("manuals/audit-pr.yaml")
   108→        executor = PlaybookExecutor()
   109→        result = await executor.run(playbook)
   110→        print(result.to_markdown())
   111→    """
   112→
   113→    def __init__(
   114→        self,
   115→        *,
   116→        provider: LLMProvider | None = None,
   117→        specialist_registry: SpecialistRegistry | None = None,
   118→        tool_registry: ToolRegistry | None = None,
   119→        cost_budget: CostBudget | None = None,
   120→        interactive: bool = False,
   121→        sandbox_enabled: bool | None = None,
   122→        sandbox_container: str | None = None,
   123→    ) -> None:
   124→        self.provider = provider or get_provider()
   125→        self.specialist_registry = specialist_registry or get_default_specialist_registry()
   126→        self.tool_registry = tool_registry or get_default_registry()
   127→        self.cost_budget = cost_budget or CostBudget()
   128→        self.interactive = interactive
   129→
   130→        # Sandbox wiring (Issue 1 / FIX-R3-SEC).
   131→        #
   132→        # Default behaviour:
   133→        #   - If ``sandbox_enabled`` is explicitly passed, honour it (caller
   134→        #     knows best — e.g. tests, Harness, MCP server).
   135→        #   - Otherwise, auto-detect: enable the Docker sandbox when the
   136→        #     ``docker`` CLI is on PATH, fall back to non-sandboxed mode
   137→        #     when it isn't.
   138→        #
   139→        # In non-sandboxed mode the ShellTool requires ``ARNES_DEV_MODE=1``
   140→        # as a double-gate before it will execute commands locally. We log
   141→        # a warning so operators know shell calls will be gated rather than
   142→        # sandboxed.
   143→        if sandbox_enabled is not None:
   144→            self._sandbox_enabled = sandbox_enabled
   145→            self._sandbox_container = sandbox_container or (
   146→                DEFAULT_SANDBOX_CONTAINER if sandbox_enabled else None
   147→            )
   148→        elif _is_docker_available():
   149→            self._sandbox_enabled = True
   150→            self._sandbox_container = sandbox_container or DEFAULT_SANDBOX_CONTAINER
   151→            logger.info(
   152→                "sandbox_docker_detected",
   153→                container=self._sandbox_container,
   154→                mode="docker-tier1",
   155→            )
   156→        else:
   157→            self._sandbox_enabled = False
   158→            self._sandbox_container = None
   159→            logger.warning(
   160→                "sandbox_docker_unavailable",
   161→                fallback="ARNES_DEV_MODE=1 required for local shell execution",
   162→                hint="Install Docker or build arnes-sandbox:latest to enable the sandbox",
   163→            )

   165→    async def run(
   166→        self,
   167→        playbook: Playbook,
   168→        *,
   169→        initial_input: dict[str, Any] | None = None,
   170→    ) -> PlaybookRunResult:
   171→        """Execute a playbook. Returns a PlaybookRunResult."""
   172→        # Use a mutable list to hold the thread (so step helpers can append)
   173→        thread_holder: list[Thread] = [Thread.create()]
   174→        start_time = time.monotonic()
   175→        outputs: dict[str, Any] = dict(playbook.variables)
   176→        if initial_input:
   177→            outputs.update(initial_input)
   178→
   179→        # Track cost guard at the run level (shared across steps)
   180→        cost_guard = CostGuard(self.provider, budget=self.cost_budget)
   181→
   182→        steps_executed = 0
   183→        steps_failed = 0
   184→        total_tokens_in = 0
   185→        total_tokens_out = 0
   186→        total_cost_usd = 0.0
   187→        aborted = False
   188→        abort_error: str | None = None
   189→
   190→        try:
   191→            for step in playbook.steps:
   192→                # Check if step should be skipped due to prior saltar_a
   193→                skip_until = outputs.get("__skip_steps_until", {})
   194→                if skip_until:
   195→                    # If we've reached the target step, clear the skip marker
   196→                    if step.id in skip_until:
   197→                        del skip_until[step.id]
   198→                        if not skip_until:
   199→                            del outputs["__skip_steps_until"]
   200→                        logger.info("saltar_a_reached", step_id=step.id)
   201→                        # Don't skip this step — execute it
   202→                    else:
   203→                        # Skip this step
   204→                        logger.info("step_skipped", step_id=step.id, reason="saltar_a")
   205→                        continue
   206→
   207→                # Execute the step
   208→                step_result = await self._execute_step(
   209→                    step,
   210→                    thread_holder,
   211→                    outputs,
   212→                    cost_guard,
   213→                    playbook,
   214→                )
   215→
   216→                if step_result["success"]:
   217→                    steps_executed += 1
   218→                    outputs[step.id] = step_result.get("output")
   219→                    # Track usage
   220→                    usage = step_result.get("usage", {})
   221→                    total_tokens_in += usage.get("tokens_in", 0)
   222→                    total_tokens_out += usage.get("tokens_out", 0)
   223→                    total_cost_usd += usage.get("cost_usd", 0.0)
   224→                else:
   225→                    steps_failed += 1
   226→                    # Check if step has si_no_se_cumple fallback
   227→                    if step.if_not_met:
   228→                        branch_result = await self._handle_conditional_branch(
   229→                            step,
   230→                            step.if_not_met,
   231→                            thread_holder,
   232→                            outputs,
   233→                            cost_guard,
   234→                            playbook,
   235→                        )
   236→                        if branch_result.get("terminate"):
   237→                            logger.info(
   238→                                "run_terminated_by_conditional",
   239→                                step_id=step.id,
   240→                                termination=branch_result["terminate"],
   241→                            )
   242→                            break
   243→                    else:
   244→                        # No fallback — abort
   245→                        thread_holder[0] = thread_holder[0].append(
   246→                            RunFailedEvent(
   247→                                thread_id=thread_holder[0].id,
   248→                                step_id=step.id,
   249→                                data={
   250→                                    "error": step_result.get("error", "Unknown error"),
   251→                                    "recoverable": False,
   252→                                },
   253→                            )
   254→                        )
   255→                        aborted = True
   256→                        abort_error = step_result.get("error")
   257→                        break
   258→
   259→            # Run completed (or aborted)
   260→            if not aborted:
   261→                thread_holder[0] = thread_holder[0].append(
   262→                    RunCompletedEvent(
   263→                        thread_id=thread_holder[0].id,
   264→                        data={
   265→                            "steps_executed": steps_executed,
   266→                            "duration_s": time.monotonic() - start_time,
   267→                            "total_tokens": total_tokens_in + total_tokens_out,
   268→                            "total_cost_usd": total_cost_usd,
   269→                        },
   270→                    )
   271→                )
   272→
   273→            return PlaybookRunResult(
   274→                thread=thread_holder[0],
   275→                success=not aborted,
   276→                steps_executed=steps_executed,
   277→                steps_failed=steps_failed,
   278→                duration_s=time.monotonic() - start_time,
   279→                total_tokens_in=total_tokens_in,
   280→                total_tokens_out=total_tokens_out,
   281→                total_cost_usd=total_cost_usd,
   282→                outputs=_filter_internal_keys(outputs),
   283→                error=abort_error,
   284→            )
   285→
   286→        except BudgetExceeded as e:
   287→            logger.error("budget_exceeded", error=str(e), spent=e.spent, budget=e.budget)
   288→            thread_holder[0] = thread_holder[0].append(
   289→                RunFailedEvent(
   290→                    thread_id=thread_holder[0].id,
   291→                    data={
   292→                        "error": f"Budget exceeded: {e}",
   293→                        "spent_usd": e.spent,
   294→                        "budget_usd": e.budget,
   295→                        "level": e.level,
   296→                    },
   297→                )
   298→            )
   299→            return PlaybookRunResult(
   300→                thread=thread_holder[0],
   301→                success=False,
   302→                steps_executed=steps_executed,
   303→                steps_failed=steps_failed,
   304→                duration_s=time.monotonic() - start_time,
   305→                total_tokens_in=total_tokens_in,
   306→                total_tokens_out=total_tokens_out,
   307→                total_cost_usd=total_cost_usd,
   308→                outputs=_filter_internal_keys(outputs),
   309→                error=str(e),
   310→            )

   312→    async def stream(  # noqa: PLR0915 - mirrors run() body; splitting hurts readability
   313→        self,
   314→        playbook: Playbook,
   315→        *,
   316→        initial_input: dict[str, Any] | None = None,
   317→    ) -> AsyncIterator[Event | PlaybookRunResult]:
   318→        """Stream playbook execution, yielding events as each step completes.
   319→
   320→        Yields (in order):
   321→
   322→        - For each step that runs: the ``StepCompletedEvent`` (or
   323→          ``StepFailedEvent``) emitted when that step finishes. The
   324→          event is the *last* event appended to the thread by
   325→          :meth:`_execute_step` — intermediate events
   326→          (``StepStartedEvent``, ``AssistantMessageEvent``,
   327→          ``CostThresholdEvent``, …) stay in the thread and are visible
   328→          in the final ``PlaybookRunResult.thread``.
   329→        - ``RunCompletedEvent`` or ``RunFailedEvent`` at the end of the run
   330→          (also appended to the thread).
   331→        - Final yield: a :class:`PlaybookRunResult` with the full thread
   332→          and aggregate accounting (steps_executed, tokens, cost).
   333→
   334→        Best-effort streaming contract:
   335→
   336→        - Steps are still executed sequentially in definition order
   337→          (the ``run()`` semantics are preserved). What streaming gives
   338→          you is the ability to surface each step's completion event
   339→          to the caller *immediately* — without waiting for the whole
   340→          playbook to finish.
   341→        - Parallel branches stream in **completion order**, not
   342→          definition order: when a parallel step finishes, the
   343→          ``PARALLEL_BRANCH_COMPLETED`` event is yielded, but the
   344→          per-sub-step ``StepCompletedEvent`` events were appended
   345→          inside ``asyncio.gather`` and arrive in the merged-event
   346→          timestamp order (see :meth:`_execute_parallel`).
   347→        - The final ``PlaybookRunResult`` is always the last yield,
   348→          even on failure / budget-exceeded.
   349→
   350→        Per-token streaming of LLM responses within a step is available
   351→        via :meth:`arnes.specialists.base.Specialist.stream`; this
   352→        executor-level stream yields step-level events, not token-level
   353→        events. The two can be composed: a UI that wants both
   354→        token-by-token rendering AND a final bitácora can consume
   355→        ``Specialist.stream()`` for the rendering and rely on the
   356→        ``AssistantMessageEvent`` emitted into the thread (by both
   357→        ``Specialist.run()`` and ``Specialist.stream()``) for the
   358→        audit trail.
   359→
   360→        Usage::
   361→
   362→            executor = PlaybookExecutor(provider=p)
   363→            async for event in executor.stream(playbook):
   364→                if isinstance(event, PlaybookRunResult):
   365→                    print(event.to_markdown())
   366→                else:
   367→                    print(f"event: {event.type.value}")
   368→        """
   369→        thread_holder: list[Thread] = [Thread.create()]
   370→        start_time = time.monotonic()
   371→        outputs: dict[str, Any] = dict(playbook.variables)
   372→        if initial_input:
   373→            outputs.update(initial_input)
   374→
   375→        cost_guard = CostGuard(self.provider, budget=self.cost_budget)
   376→
   377→        steps_executed = 0
   378→        steps_failed = 0
   379→        total_tokens_in = 0
   380→        total_tokens_out = 0
   381→        total_cost_usd = 0.0
   382→        aborted = False
   383→        abort_error: str | None = None
   384→
   385→        try:
   386→            for step in playbook.steps:
   387→                # Mirror run()'s skip-until logic verbatim.
   388→                skip_until = outputs.get("__skip_steps_until", {})
   389→                if skip_until:
   390→                    if step.id in skip_until:
   391→                        del skip_until[step.id]
   392→                        if not skip_until:
   393→                            del outputs["__skip_steps_until"]
   394→                        logger.info("saltar_a_reached", step_id=step.id)
   395→                    else:
   396→                        logger.info("step_skipped", step_id=step.id, reason="saltar_a")
   397→                        continue
   398→
   399→                step_result = await self._execute_step(
   400→                    step,
   401→                    thread_holder,
   402→                    outputs,
   403→                    cost_guard,
   404→                    playbook,
   405→                )
   406→
   407→                # Yield the last event appended by _execute_step. This is
   408→                # the StepCompletedEvent or StepFailedEvent — exactly the
   409→                # transition a streaming consumer cares about. (Earlier
   410→                # events like StepStartedEvent remain in the thread for
   411→                # the final PlaybookRunResult.)
   412→                last_event = thread_holder[0].last()
   413→                if last_event is not None:
   414→                    yield last_event
   415→
   416→                if step_result["success"]:
   417→                    steps_executed += 1
   418→                    outputs[step.id] = step_result.get("output")
   419→                    usage = step_result.get("usage", {})
   420→                    total_tokens_in += usage.get("tokens_in", 0)
   421→                    total_tokens_out += usage.get("tokens_out", 0)
   422→                    total_cost_usd += usage.get("cost_usd", 0.0)
   423→                else:
   424→                    steps_failed += 1
   425→                    if step.if_not_met:
   426→                        branch_result = await self._handle_conditional_branch(
   427→                            step,
   428→                            step.if_not_met,
   429→                            thread_holder,
   430→                            outputs,
   431→                            cost_guard,
   432→                            playbook,
   433→                        )
   434→                        if branch_result.get("terminate"):
   435→                            logger.info(
   436→                                "run_terminated_by_conditional",
   437→                                step_id=step.id,
   438→                                termination=branch_result["terminate"],
   439→                            )
   440→                            break
   441→                    else:
   442→                        run_failed = RunFailedEvent(
   443→                            thread_id=thread_holder[0].id,
   444→                            step_id=step.id,
   445→                            data={
   446→                                "error": step_result.get("error", "Unknown error"),
   447→                                "recoverable": False,
   448→                            },
   449→                        )
   450→                        thread_holder[0] = thread_holder[0].append(run_failed)
   451→                        aborted = True
   452→                        abort_error = step_result.get("error")
   453→                        # Yield the RunFailedEvent so streaming consumers
   454→                        # see the abort transition immediately (mirrors
   455→                        # the success path which yields RunCompletedEvent).
   456→                        yield run_failed
   457→                        break
   458→
   459→            if not aborted:
   460→                run_completed = RunCompletedEvent(
   461→                    thread_id=thread_holder[0].id,
   462→                    data={
   463→                        "steps_executed": steps_executed,
   464→                        "duration_s": time.monotonic() - start_time,
   465→                        "total_tokens": total_tokens_in + total_tokens_out,
   466→                        "total_cost_usd": total_cost_usd,
   467→                    },
   468→                )
   469→                thread_holder[0] = thread_holder[0].append(run_completed)
   470→                yield run_completed
   471→
   472→            yield self._build_run_result(
   473→                thread_holder[0],
   474→                success=not aborted,
   475→                steps_executed=steps_executed,
   476→                steps_failed=steps_failed,
   477→                start_time=start_time,
   478→                total_tokens_in=total_tokens_in,
   479→                total_tokens_out=total_tokens_out,
   480→                total_cost_usd=total_cost_usd,
   481→                outputs=outputs,
   482→                error=abort_error,
   483→            )
   484→        except BudgetExceeded as e:
   485→            logger.error("budget_exceeded", error=str(e), spent=e.spent, budget=e.budget)
   486→            run_failed = RunFailedEvent(
   487→                thread_id=thread_holder[0].id,
   488→                data={
   489→                    "error": f"Budget exceeded: {e}",
   490→                    "spent_usd": e.spent,
   491→                    "budget_usd": e.budget,
   492→                    "level": e.level,
   493→                },
   494→            )
   495→            thread_holder[0] = thread_holder[0].append(run_failed)
   496→            yield run_failed
   497→            yield self._build_run_result(
   498→                thread_holder[0],
   499→                success=False,
   500→                steps_executed=steps_executed,
   501→                steps_failed=steps_failed,
   502→                start_time=start_time,
   503→                total_tokens_in=total_tokens_in,
   504→                total_tokens_out=total_tokens_out,
   505→                total_cost_usd=total_cost_usd,
   506→                outputs=outputs,
   507→                error=str(e),
   508→            )

   510→    @staticmethod
   511→    def _build_run_result(
   512→        thread: Thread,
   513→        *,
   514→        success: bool,
   515→        steps_executed: int,
   516→        steps_failed: int,
   517→        start_time: float,
   518→        total_tokens_in: int,
   519→        total_tokens_out: int,
   520→        total_cost_usd: float,
   521→        outputs: dict[str, Any],
   522→        error: str | None,
   523→    ) -> PlaybookRunResult:
   524→        """Construct a :class:`PlaybookRunResult` from accumulated run state.
   525→
   526→        Shared by :meth:`run` and :meth:`stream` to avoid duplicating the
   527→        10-field construction (and to keep both methods under the ruff
   528→        PLR0915 statement-count limit).
   529→        """
   530→        return PlaybookRunResult(
   531→            thread=thread,
   532→            success=success,
   533→            steps_executed=steps_executed,
   534→            steps_failed=steps_failed,
   535→            duration_s=time.monotonic() - start_time,
   536→            total_tokens_in=total_tokens_in,
   537→            total_tokens_out=total_tokens_out,
   538→            total_cost_usd=total_cost_usd,
   539→            outputs=_filter_internal_keys(outputs),
   540→            error=error,
   541→        )

   544→    # Step execution
   545→    # ============================================================
   546→
   547→    async def _execute_step(
   548→        self,
   549→        step: PlaybookStep,
   550→        thread_holder: list[Thread],
   551→        outputs: dict[str, Any],
   552→        cost_guard: CostGuard,
   553→        playbook: Playbook,
   554→    ) -> dict[str, Any]:
   555→        """Execute a single step (specialist, tool, or parallel branch)."""
   556→        thread_holder[0] = thread_holder[0].append(
   557→            StepStartedEvent(
   558→                thread_id=thread_holder[0].id,
   559→                step_id=step.id,
   560→                specialist=step.specialist or step.tool,
   561→                data={"step_id": step.id, "specialist": step.specialist or step.tool},
   562→            )
   563→        )
   564→
   565→        step_start = time.monotonic()
   566→
   567→        try:
   568→            # Parallel branch
   569→            if step.parallel:
   570→                result = await self._execute_parallel(
   571→                    step, thread_holder, outputs, cost_guard, playbook
   572→                )
   573→            # Specialist invocation
   574→            elif step.specialist:
   575→                result = await self._execute_specialist(
   576→                    step, thread_holder, outputs, cost_guard, playbook
   577→                )
   578→            # Tool invocation
   579→            elif step.tool:
   580→                result = await self._execute_tool(step, thread_holder, outputs, playbook)
   581→            else:
   582→                raise ValueError(f"Step '{step.id}' has no action defined")
   583→
   584→            # Drain middleware event sink BEFORE recording completion so
   585→            # that AssistantMessageEvent / CostThresholdEvent / CACHE_HIT /
   586→            # REFUSAL_TRIGGERED events emitted during the step appear in
   587→            # the thread before the StepCompletedEvent. The middleware
   588→            # creates events with a nil thread_id placeholder (it does not
   589→            # have access to the Thread); we patch the real thread_id and
   590→            # step_id here.
   591→            self._drain_middleware_events(thread_holder, cost_guard, step.id)
   592→
   593→            # Pull token / cost usage out of the step result so the
   594→            # StepCompletedEvent carries the per-step aggregate. The
   595→            # specialist returns ``usage`` as a model_dump() of LLMUsage
   596→            # (tokens_in, tokens_out, cost_usd, model, cached).
   597→            usage = result.get("usage") or {}
   598→
   599→            # Record completion (with token + cost accounting)
   600→            thread_holder[0] = thread_holder[0].append(
   601→                StepCompletedEvent(
   602→                    thread_id=thread_holder[0].id,
   603→                    step_id=step.id,
   604→                    specialist=step.specialist or step.tool,
   605→                    data={
   606→                        "step_id": step.id,
   607→                        "output": result.get("output"),
   608→                        "duration_s": time.monotonic() - step_start,
   609→                        "tokens_in": usage.get("tokens_in", 0),
   610→                        "tokens_out": usage.get("tokens_out", 0),
   611→                        "cost_usd": usage.get("cost_usd", 0.0),
   612→                    },
   613→                )
   614→            )
   615→
   616→            return result
   617→
   618→        except Exception as e:
   619→            logger.exception("step_failed", step_id=step.id, error=str(e))
   620→            # Drain any events emitted before the failure too — they are
   621→            # still useful for debugging (e.g. a CostThresholdEvent that
   622→            # fired right before the crash, or an AssistantMessageEvent
   623→            # from the failed LLM call).
   624→            self._drain_middleware_events(thread_holder, cost_guard, step.id)
   625→            thread_holder[0] = thread_holder[0].append(
   626→                StepFailedEvent(
   627→                    thread_id=thread_holder[0].id,
   628→                    step_id=step.id,
   629→                    specialist=step.specialist or step.tool,
   630→                    data={"step_id": step.id, "error": str(e), "retry": False},
   631→                )
   632→            )
   633→            return {"success": False, "error": str(e)}
   634→
   635→    def _drain_middleware_events(
   636→        self,
   637→        thread_holder: list[Thread],
   638→        cost_guard: CostGuard,
   639→        step_id: str,
   640→    ) -> None:
   641→        """Drain the middleware event sink and append events to the Thread.
   642→
   643→        Middleware (CostGuard, TokenOptimizer, VerificationLayer) emit
   644→        events to a shared ``_events`` list because they do not have direct
   645→        access to the Thread. The events are created with a nil thread_id
   646→        placeholder; here we patch the real thread_id and step_id and
   647→        append them to the Thread.
   648→
   649→        Idempotent: clears the sink after draining so the same events are
   650→        not appended twice.
   651→        """
   652→        events = getattr(cost_guard, "_events", None)
   653→        if not events:
   654→            return
   655→
   656→        thread_id = thread_holder[0].id
   657→        for event in events:
   658→            # Events are frozen pydantic models; use model_copy(update=...)
   659→            # to set the real thread_id and step_id without mutating the
   660→            # original (which may be referenced by middleware state).
   661→            patched = event.model_copy(
   662→                update={"thread_id": thread_id, "step_id": event.step_id or step_id}
   663→            )
   664→            thread_holder[0] = thread_holder[0].append(patched)
   665→
   666→        events.clear()

   668→    async def _execute_specialist(
   669→        self,
   670→        step: PlaybookStep,
   671→        thread_holder: list[Thread],
   672→        outputs: dict[str, Any],
   673→        cost_guard: CostGuard,
   674→        playbook: Playbook,
   675→    ) -> dict[str, Any]:
   676→        """Invoke a specialist."""
   677→        specialist = self.specialist_registry.get(step.specialist or "")
   678→        if not specialist:
   679→            return {
   680→                "success": False,
   681→                "error": f"Specialist '{step.specialist}' not registered. Available: {self.specialist_registry.list_names()}",
   682→            }
   683→
   684→        # Resolve input (may contain Jinja2-style template refs)
   685→        input_data = self._resolve_input(step.input, outputs)
   686→
   687→        # Build tool context — sandbox state is detected once at executor
   688→        # construction time (see __init__) and propagated to every
   689→        # specialist invocation so the ShellTool can pick the right
   690→        # execution path (Docker sandbox vs. gated local execution).
   691→        ctx = ToolContext(
   692→            thread_id=thread_holder[0].id,
   693→            step_id=step.id,
   694→            specialist=step.specialist,
   695→            working_dir=".",
   696→            sandbox_enabled=self._sandbox_enabled,
   697→            sandbox_container=self._sandbox_container,
   698→            budget_remaining_usd=(
   699→                (cost_guard.budget.effective_budget() or 0) - cost_guard.spent_usd
   700→            ),
   701→            metadata={"interactive": self.interactive},
   702→        )
   703→
   704→        # Use the cost_guard-wrapped provider
   705→        result = await specialist.run(
   706→            input_data,
   707→            ctx,
   708→            provider=cost_guard,
   709→            tool_registry=self.tool_registry,
   710→        )
   711→
   712→        success = result.get("success", False)
   713→        if not success and result.get("budget_exceeded"):
   714→            raise BudgetExceeded(
   715→                f"Budget exceeded during specialist '{step.specialist}' invocation",
   716→                spent=cost_guard.spent_usd,
   717→                budget=cost_guard.budget.effective_budget() or 0.0,
   718→                level="specialist",
   719→            )
   720→
   721→        # Propagate error message for non-budget failures
   722→        if not success and result.get("error"):
   723→            return {
   724→                "success": False,
   725→                "error": result["error"],
   726→                "output": None,
   727→                "usage": result.get("usage", {}),
   728→            }
   729→
   730→        return {
   731→            "success": success,
   732→            "output": result.get("output"),
   733→            "usage": result.get("usage", {}),
   734→        }

   745→        if not tool:
   746→            return {

   748→                "error": f"Tool '{step.tool}' not registered. Available: {self.tool_registry.list_names()}",

   766→    async def _execute_parallel(
   767→        self,
   768→        step: PlaybookStep,
   769→        thread_holder: list[Thread],
   770→        outputs: dict[str, Any],
   771→        cost_guard: CostGuard,
   772→        playbook: Playbook,
   773→    ) -> dict[str, Any]:
   774→        """Execute parallel sub-steps concurrently with ``asyncio.gather``.
   775→
   776→        Each sub-step gets its OWN thread_holder (a copy of the parent thread
   777→        at this point) so appends are isolated — no race on the shared
   778→        ``thread_holder[0]`` reference. After all sub-steps complete, their
   779→        event deltas are merged back into the parent thread_holder in
   780→        timestamp order. This preserves the audit-log pattern while
   781→        enabling true parallelism (the previous implementation ran sub-steps
   782→        sequentially in a for-loop, which was correct but not concurrent).
   783→
   784→        ``Thread.append`` mutates in place (O(1) per append, replacing the
   785→        old O(N²) ``[*self.events, event]`` rebuild). Because the sub-step
   786→        coroutines mutate their own Thread objects, we MUST give each one
   787→        an isolated copy (with its own ``events`` list reference) — sharing
   788→        the parent reference would let every sub-step clobber the others'
   789→        appends. The copy is shallow on the list (a new list of the same
   790→        Event references); Events themselves are frozen pydantic models so
   791→        sharing the references is safe.
   792→
   793→        The shared ``cost_guard._events`` sink is drained by each sub-step's
   794→        own ``_execute_step`` call; because ``_drain_middleware_events`` is
   795→        synchronous, drains run atomically in the single-threaded asyncio
   796→        loop. An event emitted by sub-step B's specialist may end up drained
   797→        into sub-step A's thread_holder, but each event carries its own
   798→        ``step_id`` (set by ``_emit_assistant_message`` from the
   799→        ``ToolContext``) so the merged audit log is still correctly
   800→        attributed — the delta is just a container for the merge, not an
   801→        authoritative attribution.
   802→
   803→        Emits ``PARALLEL_BRANCH_STARTED`` before ``asyncio.gather`` and
   804→        ``PARALLEL_BRANCH_COMPLETED`` after the merge so the audit log
   805→        marks the parallel block boundaries (previously these event types
   806→        were defined but never instantiated).
   807→        """
   808→        if not step.parallel:
   809→            return {"success": False, "error": "No parallel steps defined"}
   810→
   811→        # Snapshot the parent thread so each sub-step's delta is exactly
   812→        # the events it appends beyond this point. The parent thread already
   813→        # has the outer StepStartedEvent(parallel) appended by _execute_step.
   814→        parent_snapshot = thread_holder[0]
   815→        parent_event_count = len(parent_snapshot.events)
   816→
   817→        # Emit PARALLEL_BRANCH_STARTED so the audit log records the
   818→        # parallel-block boundary. ``step_id`` is the outer parallel step's
   819→        # id; sub-step ids live inside the ``sub_steps`` payload.
   820→        thread_holder[0] = thread_holder[0].append(
   821→            Event(
   822→                type=EventType.PARALLEL_BRANCH_STARTED,
   823→                thread_id=parent_snapshot.id,
   824→                step_id=step.id,
   825→                data={
   826→                    "step_id": step.id,
   827→                    "sub_step_ids": [s.id for s in step.parallel],
   828→                    "sub_step_count": len(step.parallel),
   829→                },
   830→            )
   831→        )
   832→        # Re-snapshot so the STARTED event is part of every sub-step's
   833→        # parent_event_count baseline (it belongs to the parent, not to
   834→        # any individual sub-step's delta).
   835→        parent_snapshot = thread_holder[0]
   836→        parent_event_count = len(parent_snapshot.events)
   837→
   838→        # Each sub-step gets its OWN Thread copy with a fresh events list.
   839→        # ``Thread.append`` mutates in place, so without this copy every
   840→        # sub-step would share the same parent.events list and clobber
   841→        # each other. The new list contains the same Event references
   842→        # (Events are frozen, so sharing is safe).
   843→        sub_holders: list[list[Thread]] = [
   844→            [Thread(id=parent_snapshot.id, events=list(parent_snapshot.events))]
   845→            for _ in step.parallel
   846→        ]
   847→
   848→        # Run all sub-steps concurrently. return_exceptions=True so a single
   849→        # failure doesn't cancel the others — every sub-step runs to
   850→        # completion (or to its own failure) and we merge everything.
   851→        coros = [
   852→            self._execute_step(
   853→                sub_step,
   854→                sub_holders[i],
   855→                outputs,
   856→                cost_guard,
   857→                playbook,
   858→            )
   859→            for i, sub_step in enumerate(step.parallel)
   860→        ]
   861→        raw_results = await asyncio.gather(*coros, return_exceptions=True)
   862→
   863→        # Merge sub-step deltas back into the parent thread_holder in
   864→        # timestamp order. Each delta is the events the sub-step appended
   865→        # beyond the shared snapshot (StepStarted, AssistantMessage(s),
   866→        # StepCompleted/Failed for the sub-step, plus any middleware events
   867→        # it drained from the shared cost_guard sink).
   868→        merged_events: list[Event] = []
   869→        for holder in sub_holders:
   870→            merged_events.extend(holder[0].events[parent_event_count:])
   871→        # Stable sort by timestamp preserves intra-sub-step order for events
   872→        # with identical timestamps (the natural audit order within a single
   873→        # sub-step's lifecycle: Started → AssistantMessage → Completed).
   874→        merged_events.sort(key=lambda e: e.timestamp)
   875→
   876→        thread_holder[0] = thread_holder[0].extend(merged_events)
   877→
   878→        # Emit PARALLEL_BRANCH_COMPLETED with the per-sub-step outcome so
   879→        # the audit log marks the parallel-block end and records which
   880→        # branches succeeded / failed.
   881→        sub_step_outcomes: list[dict[str, Any]] = []
   882→        all_success = True
   883→        first_error: str | None = None
   884→        outputs_map: dict[str, dict[str, Any]] = {}
   885→
   886→        for sub_step, raw in zip(step.parallel, raw_results, strict=True):
   887→            if isinstance(raw, BaseException):
   888→                # A sub-step coroutine raised (shouldn't happen — _execute_step
   889→                # catches exceptions — but defend against executor bugs).
   890→                outputs_map[sub_step.id] = {
   891→                    "output": None,
   892→                    "success": False,
   893→                    "error": str(raw),
   894→                }
   895→                sub_step_outcomes.append(
   896→                    {"sub_step_id": sub_step.id, "success": False, "error": str(raw)}
   897→                )
   898→                all_success = False
   899→                if first_error is None:
   900→                    first_error = str(raw)
   901→            else:
   902→                outputs_map[sub_step.id] = {
   903→                    "output": raw.get("output"),
   904→                    "success": raw.get("success", False),
   905→                    "error": raw.get("error"),
   906→                }
   907→                sub_step_outcomes.append(
   908→                    {
   909→                        "sub_step_id": sub_step.id,
   910→                        "success": raw.get("success", False),
   911→                        "error": raw.get("error"),
   912→                    }
   913→                )
   914→                if not raw.get("success", False):
   915→                    all_success = False
   916→                    if first_error is None:
   917→                        first_error = raw.get("error")
   918→
   919→        thread_holder[0] = thread_holder[0].append(
   920→            Event(
   921→                type=EventType.PARALLEL_BRANCH_COMPLETED,
   922→                thread_id=thread_holder[0].id,
   923→                step_id=step.id,
   924→                data={
   925→                    "step_id": step.id,
   926→                    "all_success": all_success,
   927→                    "sub_step_outcomes": sub_step_outcomes,
   928→                    "merged_event_count": len(merged_events),
   929→                },
   930→            )
   931→        )
   932→
   933→        result: dict[str, Any] = {
   934→            "success": all_success,
   935→            "output": outputs_map,
   936→        }
   937→        if first_error is not None:
   938→            result["error"] = first_error
   939→        return result
   940→
   941→    async def _handle_conditional_branch(
   942→        self,
   943→        step: PlaybookStep,
   944→        branch: ConditionalBranch,
   945→        thread_holder: list[Thread],
   946→        outputs: dict[str, Any],
   947→        cost_guard: CostGuard,
   948→        playbook: Playbook,
   949→    ) -> dict[str, Any]:
   950→        """Handle an if_not_met or conditional branch."""
   951→        thread_holder[0] = thread_holder[0].append(
   952→            ConditionalBranchEvent(
   953→                thread_id=thread_holder[0].id,
   954→                step_id=step.id,
   955→                data={
   956→                    "condition": branch.when or "if_not_met",
   957→                    "action": branch.action,
   958→                },
   959→            )
   960→        )
   961→
   962→        if branch.action == "call" and branch.specialist:
   963→            # Invoke the fallback specialist
   964→            fallback_step = PlaybookStep(
   965→                id=f"{step.id}__fallback",
   966→                specialist=branch.specialist,
   967→                input=branch.input or {},
   968→            )
   969→            result = await self._execute_specialist(
   970→                fallback_step, thread_holder, outputs, cost_guard, playbook
   971→            )
   972→            outputs[fallback_step.id] = result.get("output")
   973→            return {"terminate": None, "result": result}
   974→
   975→        if branch.action == "terminate":
   976→            return {"terminate": branch.terminate}
   977→
   978→        if branch.action == "skip" and branch.skip_to:
   979→            # skip_to means "jump TO this step" — skip all steps between
   980→            # current and target
   981→            skip_set = outputs.setdefault("__skip_steps_until", {})
   982→            skip_set[branch.skip_to] = True  # Will be cleared when we reach it
   983→            return {"terminate": None}
   984→
   985→        return {"terminate": None}

   988→    # Template resolution

   991→    _TEMPLATE_RE = re.compile(r"\{\{\s*([^}]+)\s*\}\}")
   992→
   993→    def _resolve_input(
   994→        self,
   995→        input_value: dict[str, Any] | str | None,
   996→        outputs: dict[str, Any],
   997→    ) -> dict[str, Any]:
   998→        """Resolve Jinja2-style template references in input.
   999→
  1000→        Example: "{{ pasos.leer_diff.salida }}" → outputs["leer_diff"]["output"]
  1001→        Handles MULTIPLE templates in the same string.
  1002→        """
  1003→        if input_value is None:
  1004→            return {}
  1005→
  1006→        if isinstance(input_value, str):
  1007→            return {"__resolved_str__": self._resolve_template(input_value, outputs)}
  1008→
  1009→        if isinstance(input_value, dict):
  1010→            resolved = {}
  1011→            for k, v in input_value.items():
  1012→                if isinstance(v, str):
  1013→                    resolved[k] = self._resolve_template(v, outputs)
  1014→                elif isinstance(v, dict):
  1015→                    resolved[k] = self._resolve_input(v, outputs)
  1016→                elif isinstance(v, list):
  1017→                    resolved[k] = [
  1018→                        self._resolve_template(item, outputs)
  1019→                        if isinstance(item, str)
  1020→                        else self._resolve_input(item, outputs)
  1021→                        if isinstance(item, dict)
  1022→                        else item
  1023→                        for item in v
  1024→                    ]
  1025→                else:
  1026→                    resolved[k] = v
  1027→            return resolved
  1028→
  1029→        # Unreachable: ``input_value`` is typed as ``dict[str, Any] | str | None``
  1030→        # and all three branches above return early. Kept as a defensive
  1031→        # fallback for callers that bypass the type system (defence-in-depth).
  1032→        return {"__input__": input_value}  # type: ignore[unreachable]
  1033→
  1034→    def _resolve_template(self, template: str, outputs: dict[str, Any]) -> Any:
  1035→        """Resolve a template string, handling MULTIPLE {{ }} references.
  1036→
  1037→        Examples:
  1038→            "{{ pasos.X.salida }}" → outputs["X"]["output"]
  1039→            "Plan: {{ variables.nombre }} for PR {{ variables.pr_number }}" → "Plan: foo for PR 1234"
  1040→            "Diff: {{ pasos.leer_diff.salida }}, Sec: {{ pasos.auditoria.salida }}" → "Diff: ..., Sec: ..."
  1041→            "{{ }}" → "{{ }}"  (empty template body — returned as literal)
  1042→        """
  1043→        # Find ALL template references
  1044→        matches = list(self._TEMPLATE_RE.finditer(template))
  1045→
  1046→        if not matches:
  1047→            return template
  1048→
  1049→        # If the entire string is ONE template, return the resolved value (preserve type)
  1050→        if len(matches) == 1 and matches[0].group(0) == template:
  1051→            expr = matches[0].group(1).strip()
  1052→            if not expr:
  1053→                # Empty template body (e.g. "{{ }}") — return the original
  1054→                # literal verbatim rather than re-rendering it with different
  1055→                # whitespace.
  1056→                return template
  1057→            return self._resolve_expr(expr, outputs)
  1058→
  1059→        # Otherwise, interpolate ALL matches into the string
  1060→        result = template
  1061→        # Process in reverse order to keep indexes valid
  1062→        for match in reversed(matches):
  1063→            expr = match.group(1).strip()
  1064→            if not expr:
  1065→                # Empty template body — leave the original match untouched
  1066→                continue
  1067→            resolved = self._resolve_expr(expr, outputs)
  1068→            result = result[: match.start()] + str(resolved) + result[match.end() :]
  1069→
  1070→        return result
  1071→
  1072→    def _resolve_expr(self, expr: str, outputs: dict[str, Any]) -> Any:
  1073→        """Resolve a single template expression like 'steps.X.output'.
  1074→
  1075→        Supported forms:
  1076→            "steps.X.output"        → outputs["X"]["output"] (or outputs["X"] if raw)
  1077→            "steps.X.output.field"  → outputs["X"]["output"]["field"] (or outputs["X"]["field"])
  1078→            "variables.X"           → outputs["X"]
  1079→            "pasos.X.salida"        → legacy ES form of "steps.X.output"
  1080→
  1081→        Two important behaviors:
  1082→
  1083→        1. Only the LEADING prefix ("steps.", "variables.", "pasos.") is
  1084→           stripped. Interior occurrences of these substrings (e.g. when a
  1085→           step's output literally contains a key named "steps") are
  1086→           preserved. This makes deep nesting like
  1087→           ``{{ steps.s1.output.steps.s2.output }}`` work correctly.
  1088→
  1089→        2. For STEP references (``steps.*`` / legacy ``pasos.*``), the
  1090→           "output" (and legacy "salida") segment is treated as a VIRTUAL
  1091→           accessor: if the current dict has an "output" key, it is
  1092→           dereferenced; if not, the segment is skipped (the current dict
  1093→           IS the output). This makes ``{{ steps.X.output }}`` work whether
  1094→           the step's output was stored raw (``outputs["X"] = output_dict``)
  1095→           or wrapped (``outputs["X"] = {"output": output_dict, ...}``).
  1096→
  1097→           The virtual accessor does NOT apply to ``variables.*`` refs —
  1098→           variables are user-defined and a missing key is a real error.
  1099→        """
  1100→        # Strip leading prefix only — NOT all occurrences. Stripping all
  1101→        # occurrences of "steps." would corrupt paths whose intermediate
  1102→        # dicts literally contain a key named "steps" (deep nesting).
  1103→        stripped = expr.strip()
  1104→        is_step_ref = False
  1105→        for prefix in ("steps.", "pasos.", "variables."):
  1106→            if stripped.startswith(prefix):
  1107→                stripped = stripped[len(prefix) :]
  1108→                # Only steps/pasos get the virtual "output" accessor;
  1109→                # variables use strict key lookup.
  1110→                is_step_ref = prefix in ("steps.", "pasos.")
