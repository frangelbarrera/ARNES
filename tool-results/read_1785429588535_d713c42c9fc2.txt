     1→"""
     2→ARNES Playbook Executor — runs a compiled Playbook as a DAG.
     3→
     4→The executor:
     5→1. Walks the playbook steps in order.
     6→2. For each step, invokes the specialist or tool.
     7→3. Applies conditional branches (if/elif/else).
     8→4. Runs parallel branches concurrently.
     9→5. Handles retries with backoff.
    10→6. Pauses at HITL gates.
    11→7. Tracks budget via CostGuard.
    12→8. Appends events to the Thread.
    13→9. Returns a PlaybookRunResult with full trace.
    14→
    15→The executor is async and supports both fire-and-forget and streaming modes.
    16→"""
    17→
    18→from __future__ import annotations
    19→
    20→import re
    21→import time
    22→from typing import Any
    23→
    24→import structlog
    25→from pydantic import BaseModel, Field
    26→
    27→from arnes.llm.base import LLMProvider
    28→from arnes.llm.factory import get_provider
    29→from arnes.middleware.cost_guard import BudgetExceeded, CostBudget, CostGuard
    30→from arnes.playbooks.schema import ConditionalBranch, Playbook, PlaybookStep
    31→from arnes.specialists.base import SpecialistRegistry, get_default_specialist_registry
    32→from arnes.thread import Thread
    33→from arnes.thread.events import (
    34→    ConditionalBranchEvent,
    35→    RunCompletedEvent,
    36→    RunFailedEvent,
    37→    StepCompletedEvent,

    39→    StepStartedEvent,
    40→)
    41→from arnes.tools.base import ToolContext, ToolRegistry
    42→from arnes.tools.registry import get_default_registry

    44→logger = structlog.get_logger(__name__)

    47→class PlaybookRunResult(BaseModel):
    48→    """Result of running a playbook."""
    49→
    50→    model_config = {"arbitrary_types_allowed": True}
    51→
    52→    thread: Thread
    53→    success: bool
    54→    steps_executed: int = 0
    55→    steps_failed: int = 0
    56→    duration_s: float = 0.0
    57→    total_tokens_in: int = 0
    58→    total_tokens_out: int = 0
    59→    total_cost_usd: float = 0.0
    60→    outputs: dict[str, Any] = Field(default_factory=dict)  # step_id → output
    61→    error: str | None = None
    62→
    63→    def to_markdown(self) -> str:
    64→        """Render the run as a markdown bitácora."""
    65→        return self.thread.to_markdown()
    66→
    67→
    68→class PlaybookExecutor:
    69→    """Executes a compiled Playbook.
    70→
    71→    Usage:
    72→        playbook = PlaybookCompiler.from_file("manuals/audit-pr.yaml")
    73→        executor = PlaybookExecutor()
    74→        result = await executor.run(playbook)
    75→        print(result.to_markdown())
    76→    """
    77→
    78→    def __init__(
    79→        self,
    80→        *,
    81→        provider: LLMProvider | None = None,
    82→        specialist_registry: SpecialistRegistry | None = None,
    83→        tool_registry: ToolRegistry | None = None,
    84→        cost_budget: CostBudget | None = None,
    85→        interactive: bool = False,
    86→    ) -> None:
    87→        self.provider = provider or get_provider()
    88→        self.specialist_registry = specialist_registry or get_default_specialist_registry()
    89→        self.tool_registry = tool_registry or get_default_registry()
    90→        self.cost_budget = cost_budget or CostBudget()
    91→        self.interactive = interactive
    92→
    93→    async def run(
    94→        self,
    95→        playbook: Playbook,
    96→        *,
    97→        initial_input: dict[str, Any] | None = None,
    98→    ) -> PlaybookRunResult:
    99→        """Execute a playbook. Returns a PlaybookRunResult."""
   100→        # Use a mutable list to hold the thread (so step helpers can append)
   101→        thread_holder: list[Thread] = [Thread.create()]
   102→        start_time = time.monotonic()
   103→        outputs: dict[str, Any] = dict(playbook.variables)
   104→        if initial_input:
   105→            outputs.update(initial_input)
   106→
   107→        # Track cost guard at the run level (shared across steps)
   108→        cost_guard = CostGuard(self.provider, budget=self.cost_budget)
   109→
   110→        steps_executed = 0
   111→        steps_failed = 0
   112→        total_tokens_in = 0
   113→        total_tokens_out = 0
   114→        total_cost_usd = 0.0
   115→        aborted = False
   116→        abort_error: str | None = None
   117→
   118→        try:
   119→            for step in playbook.steps:
   120→                # Check if step should be skipped due to prior saltar_a
   121→                skip_until = outputs.get("__skip_steps_until", {})
   122→                if skip_until:
   123→                    # If we've reached the target step, clear the skip marker
   124→                    if step.id in skip_until:
   125→                        del skip_until[step.id]
   126→                        if not skip_until:
   127→                            del outputs["__skip_steps_until"]
   128→                        logger.info("saltar_a_reached", step_id=step.id)
   129→                        # Don't skip this step — execute it
   130→                    else:
   131→                        # Skip this step
   132→                        logger.info("step_skipped", step_id=step.id, reason="saltar_a")
   133→                        continue
   134→
   135→                # Execute the step
   136→                step_result = await self._execute_step(
   137→                    step,
   138→                    thread_holder,
   139→                    outputs,
   140→                    cost_guard,
   141→                    playbook,
   142→                )
   143→
   144→                if step_result["success"]:
   145→                    steps_executed += 1
   146→                    outputs[step.id] = step_result.get("output")
   147→                    # Track usage
   148→                    usage = step_result.get("usage", {})
   149→                    total_tokens_in += usage.get("tokens_in", 0)
   150→                    total_tokens_out += usage.get("tokens_out", 0)
   151→                    total_cost_usd += usage.get("cost_usd", 0.0)
   152→                else:
   153→                    steps_failed += 1
   154→                    # Check if step has si_no_se_cumple fallback
   155→                    if step.if_not_met:
   156→                        branch_result = await self._handle_conditional_branch(
   157→                            step,
   158→                            step.if_not_met,
   159→                            thread_holder,
   160→                            outputs,
   161→                            cost_guard,
   162→                            playbook,
   163→                        )
   164→                        if branch_result.get("terminate"):
   165→                            logger.info(
   166→                                "run_terminated_by_conditional",
   167→                                step_id=step.id,
   168→                                termination=branch_result["terminate"],
   169→                            )
   170→                            break
   171→                    else:
   172→                        # No fallback — abort
   173→                        thread_holder[0] = thread_holder[0].append(
   174→                            RunFailedEvent(
   175→                                thread_id=thread_holder[0].id,
   176→                                step_id=step.id,
   177→                                data={
   178→                                    "error": step_result.get("error", "Unknown error"),
   179→                                    "recoverable": False,
   180→                                },
   181→                            )
   182→                        )
   183→                        aborted = True
   184→                        abort_error = step_result.get("error")
   185→                        break
   186→
   187→            # Run completed (or aborted)
   188→            if not aborted:
   189→                thread_holder[0] = thread_holder[0].append(
   190→                    RunCompletedEvent(
   191→                        thread_id=thread_holder[0].id,
   192→                        data={
   193→                            "steps_executed": steps_executed,
   194→                            "duration_s": time.monotonic() - start_time,
   195→                            "total_tokens": total_tokens_in + total_tokens_out,
   196→                            "total_cost_usd": total_cost_usd,
   197→                        },
   198→                    )
   199→                )
   200→
   201→            return PlaybookRunResult(
   202→                thread=thread_holder[0],
   203→                success=not aborted,
   204→                steps_executed=steps_executed,
   205→                steps_failed=steps_failed,
   206→                duration_s=time.monotonic() - start_time,
   207→                total_tokens_in=total_tokens_in,
   208→                total_tokens_out=total_tokens_out,
   209→                total_cost_usd=total_cost_usd,
   210→                outputs=outputs,
   211→                error=abort_error,
   212→            )
   213→
   214→        except BudgetExceeded as e:
   215→            logger.error("budget_exceeded", error=str(e), spent=e.spent, budget=e.budget)
   216→            thread_holder[0] = thread_holder[0].append(
   217→                RunFailedEvent(
   218→                    thread_id=thread_holder[0].id,
   219→                    data={
   220→                        "error": f"Budget exceeded: {e}",
   221→                        "spent_usd": e.spent,
   222→                        "budget_usd": e.budget,
   223→                        "level": e.level,
   224→                    },
   225→                )
   226→            )
   227→            return PlaybookRunResult(
   228→                thread=thread_holder[0],
   229→                success=False,
   230→                steps_executed=steps_executed,
   231→                steps_failed=steps_failed,
   232→                duration_s=time.monotonic() - start_time,
   233→                total_tokens_in=total_tokens_in,
   234→                total_tokens_out=total_tokens_out,
   235→                total_cost_usd=total_cost_usd,
   236→                outputs=outputs,
   237→                error=str(e),
   238→            )
   239→
   240→    # ============================================================
   241→    # Step execution
   242→    # ============================================================
   243→
   244→    async def _execute_step(
   245→        self,
   246→        step: PlaybookStep,
   247→        thread_holder: list[Thread],
   248→        outputs: dict[str, Any],
   249→        cost_guard: CostGuard,
   250→        playbook: Playbook,
   251→    ) -> dict[str, Any]:
   252→        """Execute a single step (specialist, tool, or parallel branch)."""
   253→        thread_holder[0] = thread_holder[0].append(
   254→            StepStartedEvent(
   255→                thread_id=thread_holder[0].id,
   256→                step_id=step.id,
   257→                specialist=step.specialist or step.tool,
   258→                data={"step_id": step.id, "specialist": step.specialist or step.tool},
   259→            )
   260→        )
   261→
   262→        step_start = time.monotonic()
   263→
   264→        try:
   265→            # Parallel branch
   266→            if step.parallel:
   267→                result = await self._execute_parallel(
   268→                    step, thread_holder, outputs, cost_guard, playbook
   269→                )
   270→            # Specialist invocation
   271→            elif step.specialist:
   272→                result = await self._execute_specialist(
   273→                    step, thread_holder, outputs, cost_guard, playbook
   274→                )
   275→            # Tool invocation
   276→            elif step.tool:
   277→                result = await self._execute_tool(step, thread_holder, outputs, playbook)
   278→            else:
   279→                raise ValueError(f"Step '{step.id}' has no action defined")
   280→
   281→            # Drain middleware event sink BEFORE recording completion so
   282→            # that AssistantMessageEvent / CostThresholdEvent / CACHE_HIT /
   283→            # REFUSAL_TRIGGERED events emitted during the step appear in
   284→            # the thread before the StepCompletedEvent. The middleware
   285→            # creates events with a nil thread_id placeholder (it does not
   286→            # have access to the Thread); we patch the real thread_id and
   287→            # step_id here.
   288→            self._drain_middleware_events(thread_holder, cost_guard, step.id)
   289→
   290→            # Pull token / cost usage out of the step result so the
   291→            # StepCompletedEvent carries the per-step aggregate. The
   292→            # specialist returns ``usage`` as a model_dump() of LLMUsage
   293→            # (tokens_in, tokens_out, cost_usd, model, cached).
   294→            usage = result.get("usage") or {}
   295→
   296→            # Record completion (with token + cost accounting)
   297→            thread_holder[0] = thread_holder[0].append(
   298→                StepCompletedEvent(
   299→                    thread_id=thread_holder[0].id,
   300→                    step_id=step.id,
   301→                    specialist=step.specialist or step.tool,
   302→                    data={
   303→                        "step_id": step.id,
   304→                        "output": result.get("output"),
   305→                        "duration_s": time.monotonic() - step_start,
   306→                        "tokens_in": usage.get("tokens_in", 0),
   307→                        "tokens_out": usage.get("tokens_out", 0),
   308→                        "cost_usd": usage.get("cost_usd", 0.0),
   309→                    },
   310→                )
   311→            )
   312→
   313→            return result
   314→
   315→        except Exception as e:
   316→            logger.exception("step_failed", step_id=step.id, error=str(e))
   317→            # Drain any events emitted before the failure too — they are
   318→            # still useful for debugging (e.g. a CostThresholdEvent that
   319→            # fired right before the crash, or an AssistantMessageEvent
   320→            # from the failed LLM call).
   321→            self._drain_middleware_events(thread_holder, cost_guard, step.id)
   322→            thread_holder[0] = thread_holder[0].append(
   323→                StepFailedEvent(
   324→                    thread_id=thread_holder[0].id,
   325→                    step_id=step.id,
   326→                    specialist=step.specialist or step.tool,
   327→                    data={"step_id": step.id, "error": str(e), "retry": False},
   328→                )
   329→            )
   330→            return {"success": False, "error": str(e)}
   331→
   332→    def _drain_middleware_events(
   333→        self,
   334→        thread_holder: list[Thread],
   335→        cost_guard: CostGuard,
   336→        step_id: str,
   337→    ) -> None:
   338→        """Drain the middleware event sink and append events to the Thread.
   339→
   340→        Middleware (CostGuard, TokenOptimizer, VerificationLayer) emit
   341→        events to a shared ``_events`` list because they do not have direct
   342→        access to the Thread. The events are created with a nil thread_id
   343→        placeholder; here we patch the real thread_id and step_id and
   344→        append them to the Thread.
   345→
   346→        Idempotent: clears the sink after draining so the same events are
   347→        not appended twice.
   348→        """
   349→        events = getattr(cost_guard, "_events", None)
   350→        if not events:
   351→            return
   352→
   353→        thread_id = thread_holder[0].id
   354→        for event in events:
   355→            # Events are frozen pydantic models; use model_copy(update=...)
   356→            # to set the real thread_id and step_id without mutating the
   357→            # original (which may be referenced by middleware state).
   358→            patched = event.model_copy(
   359→                update={"thread_id": thread_id, "step_id": event.step_id or step_id}
   360→            )
   361→            thread_holder[0] = thread_holder[0].append(patched)
   362→
   363→        events.clear()

   365→    async def _execute_specialist(
   366→        self,
   367→        step: PlaybookStep,
   368→        thread_holder: list[Thread],
   369→        outputs: dict[str, Any],
   370→        cost_guard: CostGuard,
   371→        playbook: Playbook,
   372→    ) -> dict[str, Any]:
   373→        """Invoke a specialist."""
   374→        specialist = self.specialist_registry.get(step.specialist or "")
   375→        if not specialist:
   376→            return {
   377→                "success": False,
   378→                "error": f"Specialist '{step.specialist}' not registered. Available: {self.specialist_registry.list_names()}",
   379→            }
   380→
   381→        # Resolve input (may contain Jinja2-style template refs)
   382→        input_data = self._resolve_input(step.input, outputs)
   383→
   384→        # Build tool context
   385→        ctx = ToolContext(
   386→            thread_id=thread_holder[0].id,
   387→            step_id=step.id,
   388→            specialist=step.specialist,
   389→            working_dir=".",
   390→            sandbox_enabled=False,  # Disabled for MVP; enable in v0.2
   391→            budget_remaining_usd=(
   392→                (cost_guard.budget.effective_budget() or 0) - cost_guard.spent_usd
   393→            ),
   394→            metadata={"interactive": self.interactive},
   395→        )
   396→
   397→        # Use the cost_guard-wrapped provider
   398→        result = await specialist.run(
   399→            input_data,
   400→            ctx,
   401→            provider=cost_guard,
   402→            tool_registry=self.tool_registry,
   403→        )
   404→
   405→        success = result.get("success", False)
   406→        if not success and result.get("budget_exceeded"):
   407→            raise BudgetExceeded(
   408→                f"Budget exceeded during specialist '{step.specialist}' invocation",
   409→                spent=cost_guard.spent_usd,
   410→                budget=cost_guard.budget.effective_budget() or 0.0,
   411→                level="specialist",
   412→            )
   413→
   414→        # Propagate error message for non-budget failures
   415→        if not success and result.get("error"):
   416→            return {
   417→                "success": False,
   418→                "error": result["error"],
   419→                "output": None,
   420→                "usage": result.get("usage", {}),
   421→            }
   422→
   423→        return {
   424→            "success": success,
   425→            "output": result.get("output"),
   426→            "usage": result.get("usage", {}),
   427→        }
   428→
   429→    async def _execute_tool(
   430→        self,
   431→        step: PlaybookStep,
   432→        thread_holder: list[Thread],
   433→        outputs: dict[str, Any],
   434→        playbook: Playbook,
   435→    ) -> dict[str, Any]:
   436→        """Invoke a tool."""
   437→        tool = self.tool_registry.get(step.tool or "")
   438→        if not tool:
   439→            return {
   440→                "success": False,
   441→                "error": f"Tool '{step.tool}' not registered. Available: {self.tool_registry.list_names()}",
   442→            }
   443→
   444→        input_data = self._resolve_input(step.input, outputs)
   445→
   446→        ctx = ToolContext(
   447→            thread_id=thread_holder[0].id,
   448→            step_id=step.id,
   449→            metadata={"interactive": self.interactive},
   450→        )
   451→
   452→        result = await tool.execute(input_data, ctx)
   453→        return {
   454→            "success": result.success,
   455→            "output": result.output if result.success else None,
   456→            "error": result.error,
   457→        }

   459→    async def _execute_parallel(
   460→        self,
   461→        step: PlaybookStep,
   462→        thread_holder: list[Thread],
   463→        outputs: dict[str, Any],
   464→        cost_guard: CostGuard,
   465→        playbook: Playbook,
   466→    ) -> dict[str, Any]:
   467→        """Execute parallel sub-steps concurrently.
   468→
   469→        Note: parallel sub-steps share the same thread_holder, but since each
   470→        appends to the immutable Thread, we need to merge results back.
   471→        For MVP, parallel steps are executed sequentially (true parallelism
   472→        requires a different state model — coming in v0.2).
   473→        """
   474→        if not step.parallel:
   475→            return {"success": False, "error": "No parallel steps defined"}
   476→
   477→        outputs_map = {}
   478→        all_success = True
   479→
   480→        # For MVP: sequential execution of "parallel" steps (correctness > parallelism)
   481→        # In v0.2 we'll use asyncio.gather with proper thread merging
   482→        for sub_step in step.parallel:
   483→            result = await self._execute_step(
   484→                sub_step, thread_holder, outputs, cost_guard, playbook
   485→            )
   486→            # Wrap output in {"output": ...} structure so templates like
   487→            # {{ steps.parallel.sub_step_id.output }} resolve correctly
   488→            outputs_map[sub_step.id] = {
   489→                "output": result.get("output"),
   490→                "success": result.get("success", False),
   491→            }
   492→            if not result.get("success", False):
   493→                all_success = False
   494→
   495→        return {
   496→            "success": all_success,
   497→            "output": outputs_map,
   498→        }

   500→    async def _handle_conditional_branch(
   501→        self,
   502→        step: PlaybookStep,
   503→        branch: ConditionalBranch,
   504→        thread_holder: list[Thread],
   505→        outputs: dict[str, Any],
   506→        cost_guard: CostGuard,
   507→        playbook: Playbook,
   508→    ) -> dict[str, Any]:
   509→        """Handle an if_not_met or conditional branch."""
   510→        thread_holder[0] = thread_holder[0].append(
   511→            ConditionalBranchEvent(
   512→                thread_id=thread_holder[0].id,
   513→                step_id=step.id,
   514→                data={
   515→                    "condition": branch.when or "if_not_met",
   516→                    "action": branch.action,
   517→                },
   518→            )
   519→        )
   520→
   521→        if branch.action == "call" and branch.specialist:
   522→            # Invoke the fallback specialist
   523→            fallback_step = PlaybookStep(
   524→                id=f"{step.id}__fallback",
   525→                specialist=branch.specialist,
   526→                input=branch.input or {},
   527→            )
   528→            result = await self._execute_specialist(
   529→                fallback_step, thread_holder, outputs, cost_guard, playbook
   530→            )
   531→            outputs[fallback_step.id] = result.get("output")
   532→            return {"terminate": None, "result": result}
   533→
   534→        if branch.action == "terminate":
   535→            return {"terminate": branch.terminate}
   536→
   537→        if branch.action == "skip" and branch.skip_to:
   538→            # skip_to means "jump TO this step" — skip all steps between
   539→            # current and target
   540→            skip_set = outputs.setdefault("__skip_steps_until", {})
   541→            skip_set[branch.skip_to] = True  # Will be cleared when we reach it
   542→            return {"terminate": None}
   543→
   544→        return {"terminate": None}
   545→

   547→    # Template resolution
   548→    # ============================================================
   549→
   550→    _TEMPLATE_RE = re.compile(r"\{\{\s*([^}]+)\s*\}\}")
   551→
   552→    def _resolve_input(
   553→        self,
   554→        input_value: dict[str, Any] | str | None,
   555→        outputs: dict[str, Any],
   556→    ) -> dict[str, Any]:
   557→        """Resolve Jinja2-style template references in input.
   558→
   559→        Example: "{{ pasos.leer_diff.salida }}" → outputs["leer_diff"]["output"]
   560→        Handles MULTIPLE templates in the same string.
   561→        """
   562→        if input_value is None:
   563→            return {}
   564→
   565→        if isinstance(input_value, str):
   566→            return {"__resolved_str__": self._resolve_template(input_value, outputs)}
   567→
   568→        if isinstance(input_value, dict):
   569→            resolved = {}
   570→            for k, v in input_value.items():
   571→                if isinstance(v, str):
   572→                    resolved[k] = self._resolve_template(v, outputs)
   573→                elif isinstance(v, dict):
   574→                    resolved[k] = self._resolve_input(v, outputs)
   575→                elif isinstance(v, list):
   576→                    resolved[k] = [
   577→                        self._resolve_template(item, outputs)
   578→                        if isinstance(item, str)
   579→                        else self._resolve_input(item, outputs)
   580→                        if isinstance(item, dict)
   581→                        else item
   582→                        for item in v
   583→                    ]
   584→                else:
   585→                    resolved[k] = v
   586→            return resolved
   587→
   588→        # Unreachable: ``input_value`` is typed as ``dict[str, Any] | str | None``
   589→        # and all three branches above return early. Kept as a defensive
   590→        # fallback for callers that bypass the type system (defence-in-depth).
   591→        return {"__input__": input_value}  # type: ignore[unreachable]
   592→
   593→    def _resolve_template(self, template: str, outputs: dict[str, Any]) -> Any:
   594→        """Resolve a template string, handling MULTIPLE {{ }} references.
   595→
   596→        Examples:
   597→            "{{ pasos.X.salida }}" → outputs["X"]["output"]
   598→            "Plan: {{ variables.nombre }} for PR {{ variables.pr_number }}" → "Plan: foo for PR 1234"
   599→            "Diff: {{ pasos.leer_diff.salida }}, Sec: {{ pasos.auditoria.salida }}" → "Diff: ..., Sec: ..."
   600→            "{{ }}" → "{{ }}"  (empty template body — returned as literal)
   601→        """
   602→        # Find ALL template references
   603→        matches = list(self._TEMPLATE_RE.finditer(template))
   604→
   605→        if not matches:
   606→            return template
   607→
   608→        # If the entire string is ONE template, return the resolved value (preserve type)
   609→        if len(matches) == 1 and matches[0].group(0) == template:
   610→            expr = matches[0].group(1).strip()
   611→            if not expr:
   612→                # Empty template body (e.g. "{{ }}") — return the original
   613→                # literal verbatim rather than re-rendering it with different
   614→                # whitespace.
   615→                return template
   616→            return self._resolve_expr(expr, outputs)
   617→
   618→        # Otherwise, interpolate ALL matches into the string
   619→        result = template
   620→        # Process in reverse order to keep indexes valid
   621→        for match in reversed(matches):
   622→            expr = match.group(1).strip()
   623→            if not expr:
   624→                # Empty template body — leave the original match untouched
   625→                continue
   626→            resolved = self._resolve_expr(expr, outputs)
   627→            result = result[: match.start()] + str(resolved) + result[match.end() :]
   628→
   629→        return result
   630→
   631→    def _resolve_expr(self, expr: str, outputs: dict[str, Any]) -> Any:
   632→        """Resolve a single template expression like 'steps.X.output'.
   633→
   634→        Supported forms:
   635→            "steps.X.output"        → outputs["X"]["output"] (or outputs["X"] if raw)
   636→            "steps.X.output.field"  → outputs["X"]["output"]["field"] (or outputs["X"]["field"])
   637→            "variables.X"           → outputs["X"]
   638→            "pasos.X.salida"        → legacy ES form of "steps.X.output"
   639→
   640→        Two important behaviors:
   641→
   642→        1. Only the LEADING prefix ("steps.", "variables.", "pasos.") is
   643→           stripped. Interior occurrences of these substrings (e.g. when a
   644→           step's output literally contains a key named "steps") are
   645→           preserved. This makes deep nesting like
   646→           ``{{ steps.s1.output.steps.s2.output }}`` work correctly.
   647→
   648→        2. For STEP references (``steps.*`` / legacy ``pasos.*``), the
   649→           "output" (and legacy "salida") segment is treated as a VIRTUAL
   650→           accessor: if the current dict has an "output" key, it is
   651→           dereferenced; if not, the segment is skipped (the current dict
   652→           IS the output). This makes ``{{ steps.X.output }}`` work whether
   653→           the step's output was stored raw (``outputs["X"] = output_dict``)
   654→           or wrapped (``outputs["X"] = {"output": output_dict, ...}``).
   655→
   656→           The virtual accessor does NOT apply to ``variables.*`` refs —
   657→           variables are user-defined and a missing key is a real error.
   658→        """
   659→        # Strip leading prefix only — NOT all occurrences. Stripping all
   660→        # occurrences of "steps." would corrupt paths whose intermediate
   661→        # dicts literally contain a key named "steps" (deep nesting).
   662→        stripped = expr.strip()
   663→        is_step_ref = False
   664→        for prefix in ("steps.", "pasos.", "variables."):
   665→            if stripped.startswith(prefix):
   666→                stripped = stripped[len(prefix) :]
   667→                # Only steps/pasos get the virtual "output" accessor;
   668→                # variables use strict key lookup.
   669→                is_step_ref = prefix in ("steps.", "pasos.")
   670→                break
   671→
   672→        parts = stripped.split(".")
   673→
   674→        current: Any = outputs
   675→        for raw_part in parts:
   676→            # Legacy ES translation per-segment
   677→            part = "output" if raw_part == "salida" else raw_part
   678→            if isinstance(current, dict):
   679→                if part in current:
   680→                    current = current[part]
   681→                elif is_step_ref and part == "output":
   682→                    # Virtual accessor: the current dict is itself the
   683→                    # step's output (stored raw). Skip this segment so
   684→                    # downstream segments (e.g. ".verdict") resolve against
   685→                    # the output dict directly.
   686→                    continue
   687→                else:
   688→                    return f"{{{{ {expr} }}}}"  # Leave template as-is if not found
   689→            else:
   690→                return f"{{{{ {expr} }}}}"
   691→
   692→        return current
