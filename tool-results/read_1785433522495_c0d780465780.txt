     1→"""
     2→ARNES Playbook Executor — runs a compiled Playbook as a DAG.
     3→
     4→The executor:
     5→1. Walks the playbook steps in order.
     6→2. For each step, invokes the specialist or tool.
     7→3. Applies conditional branches (if/elif/else).
     8→4. Runs parallel branches concurrently.
     9→5. Handles retries with backoff.
    10→6. Pauses at HITL gates.
    11→7. Tracks budget via CostGuard.
    12→8. Appends events to the Thread.
    13→9. Returns a PlaybookRunResult with full trace.
    14→
    15→The executor is async and supports both fire-and-forget and streaming modes.
    16→"""

    57→def _is_docker_available() -> bool:
    58→    """Return True if the ``docker`` CLI is on PATH.
    59→
    60→    Used by the executor to decide whether to wire the Docker sandbox into
    61→    the default ``ToolContext``. This is a presence check only — it does NOT
    62→    verify the daemon is running or that ``arnes-sandbox:latest`` exists.
    63→    The ``ShellTool`` surfaces a clear error if either is missing at
    64→    execution time (``FileNotFoundError`` on ``docker run``).
    65→
    66→    We deliberately avoid probing the daemon (``docker info`` / ``docker
    67→    version``) here because:
    68→
    69→    1. It spawns a subprocess on every ``PlaybookExecutor`` construction,
    70→       which is wasteful for tests and high-throughput runs.
    71→    2. The daemon may be temporarily down even if the CLI is installed —
    72→       failing fast at construction time would prevent the user from
    73→       running non-shell playbooks that don't need Docker at all.
    74→    3. The ``ShellTool._execute_in_sandbox`` already handles the
    75→       ``FileNotFoundError`` case (docker binary missing) and returns a
    76→       actionable error message.
    77→    """
    78→    return shutil.which("docker") is not None

   102→class PlaybookExecutor:
   103→    """Executes a compiled Playbook.
   104→
   105→    Usage:
   106→        playbook = PlaybookCompiler.from_file("manuals/audit-pr.yaml")
   107→        executor = PlaybookExecutor()
   108→        result = await executor.run(playbook)
   109→        print(result.to_markdown())
   110→    """
   111→
   112→    def __init__(
   113→        self,
   114→        *,
   115→        provider: LLMProvider | None = None,
   116→        specialist_registry: SpecialistRegistry | None = None,
   117→        tool_registry: ToolRegistry | None = None,
   118→        cost_budget: CostBudget | None = None,
   119→        interactive: bool = False,
   120→        sandbox_enabled: bool | None = None,
   121→        sandbox_container: str | None = None,
   122→    ) -> None:
   123→        self.provider = provider or get_provider()
   124→        self.specialist_registry = specialist_registry or get_default_specialist_registry()
   125→        self.tool_registry = tool_registry or get_default_registry()
   126→        self.cost_budget = cost_budget or CostBudget()
   127→        self.interactive = interactive
   128→
   129→        # Sandbox wiring (Issue 1 / FIX-R3-SEC).
   130→        #
   131→        # Default behaviour:
   132→        #   - If ``sandbox_enabled`` is explicitly passed, honour it (caller
   133→        #     knows best — e.g. tests, Harness, MCP server).
   134→        #   - Otherwise, auto-detect: enable the Docker sandbox when the
   135→        #     ``docker`` CLI is on PATH, fall back to non-sandboxed mode
   136→        #     when it isn't.
   137→        #
   138→        # In non-sandboxed mode the ShellTool requires ``ARNES_DEV_MODE=1``
   139→        # as a double-gate before it will execute commands locally. We log
   140→        # a warning so operators know shell calls will be gated rather than
   141→        # sandboxed.
   142→        if sandbox_enabled is not None:
   143→            self._sandbox_enabled = sandbox_enabled
   144→            self._sandbox_container = sandbox_container or (
   145→                DEFAULT_SANDBOX_CONTAINER if sandbox_enabled else None
   146→            )
   147→        elif _is_docker_available():
   148→            self._sandbox_enabled = True
   149→            self._sandbox_container = sandbox_container or DEFAULT_SANDBOX_CONTAINER
   150→            logger.info(
   151→                "sandbox_docker_detected",
   152→                container=self._sandbox_container,
   153→                mode="docker-tier1",
   154→            )
   155→        else:
   156→            self._sandbox_enabled = False
   157→            self._sandbox_container = None
   158→            logger.warning(
   159→                "sandbox_docker_unavailable",
   160→                fallback="ARNES_DEV_MODE=1 required for local shell execution",
   161→                hint="Install Docker or build arnes-sandbox:latest to enable the sandbox",
   162→            )

   164→    async def run(
   165→        self,
   166→        playbook: Playbook,
   167→        *,
   168→        initial_input: dict[str, Any] | None = None,
   169→    ) -> PlaybookRunResult:
   170→        """Execute a playbook. Returns a PlaybookRunResult."""
   171→        # Use a mutable list to hold the thread (so step helpers can append)
   172→        thread_holder: list[Thread] = [Thread.create()]
   173→        start_time = time.monotonic()
   174→        outputs: dict[str, Any] = dict(playbook.variables)
   175→        if initial_input:
   176→            outputs.update(initial_input)
   177→
   178→        # Track cost guard at the run level (shared across steps)
   179→        cost_guard = CostGuard(self.provider, budget=self.cost_budget)
   180→
   181→        steps_executed = 0
   182→        steps_failed = 0
   183→        total_tokens_in = 0
   184→        total_tokens_out = 0
   185→        total_cost_usd = 0.0
   186→        aborted = False
   187→        abort_error: str | None = None
   188→
   189→        try:
   190→            for step in playbook.steps:
   191→                # Check if step should be skipped due to prior saltar_a
   192→                skip_until = outputs.get("__skip_steps_until", {})
   193→                if skip_until:
   194→                    # If we've reached the target step, clear the skip marker
   195→                    if step.id in skip_until:
   196→                        del skip_until[step.id]
   197→                        if not skip_until:
   198→                            del outputs["__skip_steps_until"]
   199→                        logger.info("saltar_a_reached", step_id=step.id)
   200→                        # Don't skip this step — execute it
   201→                    else:
   202→                        # Skip this step
   203→                        logger.info("step_skipped", step_id=step.id, reason="saltar_a")
   204→                        continue
   205→
   206→                # Execute the step
   207→                step_result = await self._execute_step(
   208→                    step,
   209→                    thread_holder,
   210→                    outputs,
   211→                    cost_guard,
   212→                    playbook,
   213→                )
   214→
   215→                if step_result["success"]:
   216→                    steps_executed += 1
   217→                    outputs[step.id] = step_result.get("output")
   218→                    # Track usage
   219→                    usage = step_result.get("usage", {})
   220→                    total_tokens_in += usage.get("tokens_in", 0)
   221→                    total_tokens_out += usage.get("tokens_out", 0)
   222→                    total_cost_usd += usage.get("cost_usd", 0.0)
   223→                else:
   224→                    steps_failed += 1
   225→                    # Check if step has si_no_se_cumple fallback
   226→                    if step.if_not_met:
   227→                        branch_result = await self._handle_conditional_branch(
   228→                            step,
   229→                            step.if_not_met,
   230→                            thread_holder,
   231→                            outputs,
   232→                            cost_guard,
   233→                            playbook,
   234→                        )
   235→                        if branch_result.get("terminate"):
   236→                            logger.info(
   237→                                "run_terminated_by_conditional",
   238→                                step_id=step.id,
   239→                                termination=branch_result["terminate"],
   240→                            )
   241→                            break
   242→                    else:
   243→                        # No fallback — abort
   244→                        thread_holder[0] = thread_holder[0].append(
   245→                            RunFailedEvent(
   246→                                thread_id=thread_holder[0].id,
   247→                                step_id=step.id,
   248→                                data={
   249→                                    "error": step_result.get("error", "Unknown error"),
   250→                                    "recoverable": False,
   251→                                },
   252→                            )
   253→                        )
   254→                        aborted = True
   255→                        abort_error = step_result.get("error")
   256→                        break
   257→
   258→            # Run completed (or aborted)
   259→            if not aborted:
   260→                thread_holder[0] = thread_holder[0].append(
   261→                    RunCompletedEvent(
   262→                        thread_id=thread_holder[0].id,
   263→                        data={
   264→                            "steps_executed": steps_executed,
   265→                            "duration_s": time.monotonic() - start_time,
   266→                            "total_tokens": total_tokens_in + total_tokens_out,
   267→                            "total_cost_usd": total_cost_usd,
   268→                        },
   269→                    )
   270→                )
   271→
   272→            return PlaybookRunResult(
   273→                thread=thread_holder[0],
   274→                success=not aborted,
   275→                steps_executed=steps_executed,
   276→                steps_failed=steps_failed,
   277→                duration_s=time.monotonic() - start_time,
   278→                total_tokens_in=total_tokens_in,
   279→                total_tokens_out=total_tokens_out,
   280→                total_cost_usd=total_cost_usd,
   281→                outputs=_filter_internal_keys(outputs),
   282→                error=abort_error,
   283→            )
   284→
   285→        except BudgetExceeded as e:
   286→            logger.error("budget_exceeded", error=str(e), spent=e.spent, budget=e.budget)
   287→            thread_holder[0] = thread_holder[0].append(
   288→                RunFailedEvent(
   289→                    thread_id=thread_holder[0].id,
   290→                    data={
   291→                        "error": f"Budget exceeded: {e}",
   292→                        "spent_usd": e.spent,
   293→                        "budget_usd": e.budget,
   294→                        "level": e.level,
   295→                    },
   296→                )
   297→            )
   298→            return PlaybookRunResult(
   299→                thread=thread_holder[0],
   300→                success=False,
   301→                steps_executed=steps_executed,
   302→                steps_failed=steps_failed,
   303→                duration_s=time.monotonic() - start_time,
   304→                total_tokens_in=total_tokens_in,
   305→                total_tokens_out=total_tokens_out,
   306→                total_cost_usd=total_cost_usd,
   307→                outputs=_filter_internal_keys(outputs),
   308→                error=str(e),
   309→            )

   315→    async def _execute_step(
   316→        self,
   317→        step: PlaybookStep,
   318→        thread_holder: list[Thread],
   319→        outputs: dict[str, Any],
   320→        cost_guard: CostGuard,
   321→        playbook: Playbook,
   322→    ) -> dict[str, Any]:
   323→        """Execute a single step (specialist, tool, or parallel branch)."""
   324→        thread_holder[0] = thread_holder[0].append(
   325→            StepStartedEvent(
   326→                thread_id=thread_holder[0].id,
   327→                step_id=step.id,
   328→                specialist=step.specialist or step.tool,
   329→                data={"step_id": step.id, "specialist": step.specialist or step.tool},
   330→            )
   331→        )
   332→
   333→        step_start = time.monotonic()
   334→
   335→        try:
   336→            # Parallel branch
   337→            if step.parallel:
   338→                result = await self._execute_parallel(
   339→                    step, thread_holder, outputs, cost_guard, playbook
   340→                )
   341→            # Specialist invocation
   342→            elif step.specialist:
   343→                result = await self._execute_specialist(
   344→                    step, thread_holder, outputs, cost_guard, playbook
   345→                )
   346→            # Tool invocation
   347→            elif step.tool:
   348→                result = await self._execute_tool(step, thread_holder, outputs, playbook)
   349→            else:
   350→                raise ValueError(f"Step '{step.id}' has no action defined")
   351→
   352→            # Drain middleware event sink BEFORE recording completion so
   353→            # that AssistantMessageEvent / CostThresholdEvent / CACHE_HIT /
   354→            # REFUSAL_TRIGGERED events emitted during the step appear in
   355→            # the thread before the StepCompletedEvent. The middleware
   356→            # creates events with a nil thread_id placeholder (it does not
   357→            # have access to the Thread); we patch the real thread_id and
   358→            # step_id here.
   359→            self._drain_middleware_events(thread_holder, cost_guard, step.id)
   360→
   361→            # Pull token / cost usage out of the step result so the
   362→            # StepCompletedEvent carries the per-step aggregate. The
   363→            # specialist returns ``usage`` as a model_dump() of LLMUsage
   364→            # (tokens_in, tokens_out, cost_usd, model, cached).
   365→            usage = result.get("usage") or {}
   366→
   367→            # Record completion (with token + cost accounting)
   368→            thread_holder[0] = thread_holder[0].append(
   369→                StepCompletedEvent(
   370→                    thread_id=thread_holder[0].id,
   371→                    step_id=step.id,
   372→                    specialist=step.specialist or step.tool,
   373→                    data={
   374→                        "step_id": step.id,
   375→                        "output": result.get("output"),
   376→                        "duration_s": time.monotonic() - step_start,
   377→                        "tokens_in": usage.get("tokens_in", 0),
   378→                        "tokens_out": usage.get("tokens_out", 0),
   379→                        "cost_usd": usage.get("cost_usd", 0.0),
   380→                    },
   381→                )
   382→            )
   383→
   384→            return result
   385→
   386→        except Exception as e:
   387→            logger.exception("step_failed", step_id=step.id, error=str(e))
   388→            # Drain any events emitted before the failure too — they are
   389→            # still useful for debugging (e.g. a CostThresholdEvent that
   390→            # fired right before the crash, or an AssistantMessageEvent
   391→            # from the failed LLM call).
   392→            self._drain_middleware_events(thread_holder, cost_guard, step.id)
   393→            thread_holder[0] = thread_holder[0].append(
   394→                StepFailedEvent(
   395→                    thread_id=thread_holder[0].id,
   396→                    step_id=step.id,
   397→                    specialist=step.specialist or step.tool,
   398→                    data={"step_id": step.id, "error": str(e), "retry": False},
   399→                )
   400→            )
   401→            return {"success": False, "error": str(e)}
   402→
   403→    def _drain_middleware_events(
   404→        self,
   405→        thread_holder: list[Thread],
   406→        cost_guard: CostGuard,
   407→        step_id: str,
   408→    ) -> None:
   409→        """Drain the middleware event sink and append events to the Thread.
   410→
   411→        Middleware (CostGuard, TokenOptimizer, VerificationLayer) emit
   412→        events to a shared ``_events`` list because they do not have direct
   413→        access to the Thread. The events are created with a nil thread_id
   414→        placeholder; here we patch the real thread_id and step_id and
   415→        append them to the Thread.
   416→
   417→        Idempotent: clears the sink after draining so the same events are
   418→        not appended twice.
   419→        """
   420→        events = getattr(cost_guard, "_events", None)
   421→        if not events:
   422→            return
   423→
   424→        thread_id = thread_holder[0].id
   425→        for event in events:
   426→            # Events are frozen pydantic models; use model_copy(update=...)
   427→            # to set the real thread_id and step_id without mutating the
   428→            # original (which may be referenced by middleware state).
   429→            patched = event.model_copy(
   430→                update={"thread_id": thread_id, "step_id": event.step_id or step_id}
   431→            )
   432→            thread_holder[0] = thread_holder[0].append(patched)
   433→
   434→        events.clear()

   436→    async def _execute_specialist(
   437→        self,
   438→        step: PlaybookStep,
   439→        thread_holder: list[Thread],
   440→        outputs: dict[str, Any],
   441→        cost_guard: CostGuard,
   442→        playbook: Playbook,
   443→    ) -> dict[str, Any]:
   444→        """Invoke a specialist."""
   445→        specialist = self.specialist_registry.get(step.specialist or "")
   446→        if not specialist:
   447→            return {
   448→                "success": False,
   449→                "error": f"Specialist '{step.specialist}' not registered. Available: {self.specialist_registry.list_names()}",
   450→            }
   451→
   452→        # Resolve input (may contain Jinja2-style template refs)
   453→        input_data = self._resolve_input(step.input, outputs)
   454→
   455→        # Build tool context — sandbox state is detected once at executor
   456→        # construction time (see __init__) and propagated to every
   457→        # specialist invocation so the ShellTool can pick the right
   458→        # execution path (Docker sandbox vs. gated local execution).
   459→        ctx = ToolContext(
   460→            thread_id=thread_holder[0].id,
   461→            step_id=step.id,
   462→            specialist=step.specialist,
   463→            working_dir=".",
   464→            sandbox_enabled=self._sandbox_enabled,
   465→            sandbox_container=self._sandbox_container,
   466→            budget_remaining_usd=(
   467→                (cost_guard.budget.effective_budget() or 0) - cost_guard.spent_usd
   468→            ),
   469→            metadata={"interactive": self.interactive},
   470→        )
   471→
   472→        # Use the cost_guard-wrapped provider
   473→        result = await specialist.run(
   474→            input_data,
   475→            ctx,
   476→            provider=cost_guard,
   477→            tool_registry=self.tool_registry,
   478→        )
   479→
   480→        success = result.get("success", False)
   481→        if not success and result.get("budget_exceeded"):
   482→            raise BudgetExceeded(
   483→                f"Budget exceeded during specialist '{step.specialist}' invocation",
   484→                spent=cost_guard.spent_usd,
   485→                budget=cost_guard.budget.effective_budget() or 0.0,
   486→                level="specialist",
   487→            )
   488→
   489→        # Propagate error message for non-budget failures
   490→        if not success and result.get("error"):
   491→            return {
   492→                "success": False,
   493→                "error": result["error"],
   494→                "output": None,
   495→                "usage": result.get("usage", {}),
   496→            }
   497→
   498→        return {
   499→            "success": success,
   500→            "output": result.get("output"),
   501→            "usage": result.get("usage", {}),
   502→        }
   503→
   504→    async def _execute_tool(
   505→        self,
   506→        step: PlaybookStep,
   507→        thread_holder: list[Thread],
   508→        outputs: dict[str, Any],
   509→        playbook: Playbook,
   510→    ) -> dict[str, Any]:
   511→        """Invoke a tool."""
   512→        tool = self.tool_registry.get(step.tool or "")
   513→        if not tool:
   514→            return {
   515→                "success": False,
   516→                "error": f"Tool '{step.tool}' not registered. Available: {self.tool_registry.list_names()}",
   517→            }
   518→
   519→        input_data = self._resolve_input(step.input, outputs)
   520→
   521→        ctx = ToolContext(
   522→            thread_id=thread_holder[0].id,
   523→            step_id=step.id,
   524→            metadata={"interactive": self.interactive},
   525→        )
   526→
   527→        result = await tool.execute(input_data, ctx)
   528→        return {
   529→            "success": result.success,
   530→            "output": result.output if result.success else None,
   531→            "error": result.error,
   532→        }

   534→    async def _execute_parallel(
   535→        self,
   536→        step: PlaybookStep,
   537→        thread_holder: list[Thread],
   538→        outputs: dict[str, Any],
   539→        cost_guard: CostGuard,
   540→        playbook: Playbook,
   541→    ) -> dict[str, Any]:
   542→        """Execute parallel sub-steps concurrently with ``asyncio.gather``.
   543→
   544→        Each sub-step gets its OWN thread_holder (a copy of the parent thread
   545→        at this point) so appends are isolated — no race on the shared
   546→        ``thread_holder[0]`` reference. After all sub-steps complete, their
   547→        event deltas are merged back into the parent thread_holder in
   548→        timestamp order. This preserves the audit-log pattern while
   549→        enabling true parallelism (the previous implementation ran sub-steps
   550→        sequentially in a for-loop, which was correct but not concurrent).
   551→
   552→        ``Thread.append`` mutates in place (O(1) per append, replacing the
   553→        old O(N²) ``[*self.events, event]`` rebuild). Because the sub-step
   554→        coroutines mutate their own Thread objects, we MUST give each one
   555→        an isolated copy (with its own ``events`` list reference) — sharing
   556→        the parent reference would let every sub-step clobber the others'
   557→        appends. The copy is shallow on the list (a new list of the same
   558→        Event references); Events themselves are frozen pydantic models so
   559→        sharing the references is safe.
   560→
   561→        The shared ``cost_guard._events`` sink is drained by each sub-step's
   562→        own ``_execute_step`` call; because ``_drain_middleware_events`` is
   563→        synchronous, drains run atomically in the single-threaded asyncio
   564→        loop. An event emitted by sub-step B's specialist may end up drained
   565→        into sub-step A's thread_holder, but each event carries its own
   566→        ``step_id`` (set by ``_emit_assistant_message`` from the
   567→        ``ToolContext``) so the merged audit log is still correctly
   568→        attributed — the delta is just a container for the merge, not an
   569→        authoritative attribution.
   570→
   571→        Emits ``PARALLEL_BRANCH_STARTED`` before ``asyncio.gather`` and
   572→        ``PARALLEL_BRANCH_COMPLETED`` after the merge so the audit log
   573→        marks the parallel block boundaries (previously these event types
   574→        were defined but never instantiated).
   575→        """
   576→        if not step.parallel:
   577→            return {"success": False, "error": "No parallel steps defined"}
   578→
   579→        # Snapshot the parent thread so each sub-step's delta is exactly
   580→        # the events it appends beyond this point. The parent thread already
   581→        # has the outer StepStartedEvent(parallel) appended by _execute_step.
   582→        parent_snapshot = thread_holder[0]
   583→        parent_event_count = len(parent_snapshot.events)
   584→
   585→        # Emit PARALLEL_BRANCH_STARTED so the audit log records the
   586→        # parallel-block boundary. ``step_id`` is the outer parallel step's
   587→        # id; sub-step ids live inside the ``sub_steps`` payload.
   588→        thread_holder[0] = thread_holder[0].append(
   589→            Event(
   590→                type=EventType.PARALLEL_BRANCH_STARTED,
   591→                thread_id=parent_snapshot.id,
   592→                step_id=step.id,
   593→                data={
   594→                    "step_id": step.id,
   595→                    "sub_step_ids": [s.id for s in step.parallel],
   596→                    "sub_step_count": len(step.parallel),
   597→                },
   598→            )
   599→        )
   600→        # Re-snapshot so the STARTED event is part of every sub-step's
   601→        # parent_event_count baseline (it belongs to the parent, not to
   602→        # any individual sub-step's delta).
   603→        parent_snapshot = thread_holder[0]
   604→        parent_event_count = len(parent_snapshot.events)
   605→
   606→        # Each sub-step gets its OWN Thread copy with a fresh events list.
   607→        # ``Thread.append`` mutates in place, so without this copy every
   608→        # sub-step would share the same parent.events list and clobber
   609→        # each other. The new list contains the same Event references
   610→        # (Events are frozen, so sharing is safe).
   611→        sub_holders: list[list[Thread]] = [
   612→            [Thread(id=parent_snapshot.id, events=list(parent_snapshot.events))]
   613→            for _ in step.parallel
   614→        ]
   615→
   616→        # Run all sub-steps concurrently. return_exceptions=True so a single
   617→        # failure doesn't cancel the others — every sub-step runs to
   618→        # completion (or to its own failure) and we merge everything.
   619→        coros = [
   620→            self._execute_step(
   621→                sub_step,
   622→                sub_holders[i],
   623→                outputs,
   624→                cost_guard,
   625→                playbook,
   626→            )
   627→            for i, sub_step in enumerate(step.parallel)
   628→        ]
   629→        raw_results = await asyncio.gather(*coros, return_exceptions=True)
   630→
   631→        # Merge sub-step deltas back into the parent thread_holder in
   632→        # timestamp order. Each delta is the events the sub-step appended
   633→        # beyond the shared snapshot (StepStarted, AssistantMessage(s),
   634→        # StepCompleted/Failed for the sub-step, plus any middleware events
   635→        # it drained from the shared cost_guard sink).
   636→        merged_events: list[Event] = []
   637→        for holder in sub_holders:
   638→            merged_events.extend(holder[0].events[parent_event_count:])
   639→        # Stable sort by timestamp preserves intra-sub-step order for events
   640→        # with identical timestamps (the natural audit order within a single
   641→        # sub-step's lifecycle: Started → AssistantMessage → Completed).
   642→        merged_events.sort(key=lambda e: e.timestamp)
   643→
   644→        thread_holder[0] = thread_holder[0].extend(merged_events)
   645→
   646→        # Emit PARALLEL_BRANCH_COMPLETED with the per-sub-step outcome so
   647→        # the audit log marks the parallel-block end and records which
   648→        # branches succeeded / failed.
   649→        sub_step_outcomes: list[dict[str, Any]] = []
   650→        all_success = True
   651→        first_error: str | None = None
   652→        outputs_map: dict[str, dict[str, Any]] = {}
   653→
   654→        for sub_step, raw in zip(step.parallel, raw_results, strict=True):
   655→            if isinstance(raw, BaseException):
   656→                # A sub-step coroutine raised (shouldn't happen — _execute_step
   657→                # catches exceptions — but defend against executor bugs).
   658→                outputs_map[sub_step.id] = {
   659→                    "output": None,
   660→                    "success": False,
   661→                    "error": str(raw),
   662→                }
   663→                sub_step_outcomes.append(
   664→                    {"sub_step_id": sub_step.id, "success": False, "error": str(raw)}
   665→                )
   666→                all_success = False
   667→                if first_error is None:
   668→                    first_error = str(raw)
   669→            else:
   670→                outputs_map[sub_step.id] = {
   671→                    "output": raw.get("output"),
   672→                    "success": raw.get("success", False),
   673→                    "error": raw.get("error"),
   674→                }
   675→                sub_step_outcomes.append(
   676→                    {
   677→                        "sub_step_id": sub_step.id,
   678→                        "success": raw.get("success", False),
   679→                        "error": raw.get("error"),
   680→                    }
   681→                )
   682→                if not raw.get("success", False):
   683→                    all_success = False
   684→                    if first_error is None:
   685→                        first_error = raw.get("error")
   686→
   687→        thread_holder[0] = thread_holder[0].append(
   688→            Event(
   689→                type=EventType.PARALLEL_BRANCH_COMPLETED,
   690→                thread_id=thread_holder[0].id,
   691→                step_id=step.id,
   692→                data={
   693→                    "step_id": step.id,
   694→                    "all_success": all_success,
   695→                    "sub_step_outcomes": sub_step_outcomes,
   696→                    "merged_event_count": len(merged_events),
   697→                },
   698→            )
   699→        )
   700→
   701→        result: dict[str, Any] = {
   702→            "success": all_success,
   703→            "output": outputs_map,
   704→        }
   705→        if first_error is not None:
   706→            result["error"] = first_error
   707→        return result
   708→
   709→    async def _handle_conditional_branch(
   710→        self,
   711→        step: PlaybookStep,
   712→        branch: ConditionalBranch,
   713→        thread_holder: list[Thread],
   714→        outputs: dict[str, Any],
   715→        cost_guard: CostGuard,
   716→        playbook: Playbook,
   717→    ) -> dict[str, Any]:
   718→        """Handle an if_not_met or conditional branch."""
   719→        thread_holder[0] = thread_holder[0].append(
   720→            ConditionalBranchEvent(
   721→                thread_id=thread_holder[0].id,
   722→                step_id=step.id,
   723→                data={
   724→                    "condition": branch.when or "if_not_met",
   725→                    "action": branch.action,
   726→                },
   727→            )
   728→        )
   729→
   730→        if branch.action == "call" and branch.specialist:
   731→            # Invoke the fallback specialist
   732→            fallback_step = PlaybookStep(
   733→                id=f"{step.id}__fallback",
   734→                specialist=branch.specialist,
   735→                input=branch.input or {},
   736→            )
   737→            result = await self._execute_specialist(
   738→                fallback_step, thread_holder, outputs, cost_guard, playbook
   739→            )
   740→            outputs[fallback_step.id] = result.get("output")
   741→            return {"terminate": None, "result": result}
   742→
   743→        if branch.action == "terminate":
   744→            return {"terminate": branch.terminate}
   745→
   746→        if branch.action == "skip" and branch.skip_to:
   747→            # skip_to means "jump TO this step" — skip all steps between
   748→            # current and target
   749→            skip_set = outputs.setdefault("__skip_steps_until", {})
   750→            skip_set[branch.skip_to] = True  # Will be cleared when we reach it
   751→            return {"terminate": None}
   752→
   753→        return {"terminate": None}

   759→    _TEMPLATE_RE = re.compile(r"\{\{\s*([^}]+)\s*\}\}")
   760→
   761→    def _resolve_input(
   762→        self,
   763→        input_value: dict[str, Any] | str | None,
   764→        outputs: dict[str, Any],
   765→    ) -> dict[str, Any]:
   766→        """Resolve Jinja2-style template references in input.
   767→
   768→        Example: "{{ pasos.leer_diff.salida }}" → outputs["leer_diff"]["output"]
   769→        Handles MULTIPLE templates in the same string.
   770→        """
   771→        if input_value is None:
   772→            return {}
   773→
   774→        if isinstance(input_value, str):
   775→            return {"__resolved_str__": self._resolve_template(input_value, outputs)}
   776→
   777→        if isinstance(input_value, dict):
   778→            resolved = {}
   779→            for k, v in input_value.items():
   780→                if isinstance(v, str):
   781→                    resolved[k] = self._resolve_template(v, outputs)
   782→                elif isinstance(v, dict):
   783→                    resolved[k] = self._resolve_input(v, outputs)
   784→                elif isinstance(v, list):
   785→                    resolved[k] = [
   786→                        self._resolve_template(item, outputs)
   787→                        if isinstance(item, str)
   788→                        else self._resolve_input(item, outputs)
   789→                        if isinstance(item, dict)
   790→                        else item
   791→                        for item in v
   792→                    ]
   793→                else:
   794→                    resolved[k] = v
   795→            return resolved
   796→
   797→        # Unreachable: ``input_value`` is typed as ``dict[str, Any] | str | None``
   798→        # and all three branches above return early. Kept as a defensive
   799→        # fallback for callers that bypass the type system (defence-in-depth).
   800→        return {"__input__": input_value}  # type: ignore[unreachable]
   801→
   802→    def _resolve_template(self, template: str, outputs: dict[str, Any]) -> Any:
   803→        """Resolve a template string, handling MULTIPLE {{ }} references.
   804→
   805→        Examples:
   806→            "{{ pasos.X.salida }}" → outputs["X"]["output"]
   807→            "Plan: {{ variables.nombre }} for PR {{ variables.pr_number }}" → "Plan: foo for PR 1234"
   808→            "Diff: {{ pasos.leer_diff.salida }}, Sec: {{ pasos.auditoria.salida }}" → "Diff: ..., Sec: ..."
   809→            "{{ }}" → "{{ }}"  (empty template body — returned as literal)
   810→        """
   811→        # Find ALL template references
   812→        matches = list(self._TEMPLATE_RE.finditer(template))
   813→
   814→        if not matches:
   815→            return template
   816→
   817→        # If the entire string is ONE template, return the resolved value (preserve type)
   818→        if len(matches) == 1 and matches[0].group(0) == template:
   819→            expr = matches[0].group(1).strip()
   820→            if not expr:
   821→                # Empty template body (e.g. "{{ }}") — return the original
   822→                # literal verbatim rather than re-rendering it with different
   823→                # whitespace.
   824→                return template
   825→            return self._resolve_expr(expr, outputs)
   826→
   827→        # Otherwise, interpolate ALL matches into the string
   828→        result = template
   829→        # Process in reverse order to keep indexes valid
   830→        for match in reversed(matches):
   831→            expr = match.group(1).strip()
   832→            if not expr:
   833→                # Empty template body — leave the original match untouched
   834→                continue
   835→            resolved = self._resolve_expr(expr, outputs)
   836→            result = result[: match.start()] + str(resolved) + result[match.end() :]
   837→
   838→        return result
   839→
   840→    def _resolve_expr(self, expr: str, outputs: dict[str, Any]) -> Any:
   841→        """Resolve a single template expression like 'steps.X.output'.
   842→
   843→        Supported forms:
   844→            "steps.X.output"        → outputs["X"]["output"] (or outputs["X"] if raw)
   845→            "steps.X.output.field"  → outputs["X"]["output"]["field"] (or outputs["X"]["field"])
   846→            "variables.X"           → outputs["X"]
   847→            "pasos.X.salida"        → legacy ES form of "steps.X.output"
   848→
   849→        Two important behaviors:
   850→
   851→        1. Only the LEADING prefix ("steps.", "variables.", "pasos.") is
   852→           stripped. Interior occurrences of these substrings (e.g. when a
   853→           step's output literally contains a key named "steps") are
   854→           preserved. This makes deep nesting like
   855→           ``{{ steps.s1.output.steps.s2.output }}`` work correctly.
   856→
   857→        2. For STEP references (``steps.*`` / legacy ``pasos.*``), the
   858→           "output" (and legacy "salida") segment is treated as a VIRTUAL
   859→           accessor: if the current dict has an "output" key, it is
   860→           dereferenced; if not, the segment is skipped (the current dict
   861→           IS the output). This makes ``{{ steps.X.output }}`` work whether
   862→           the step's output was stored raw (``outputs["X"] = output_dict``)
   863→           or wrapped (``outputs["X"] = {"output": output_dict, ...}``).
   864→
   865→           The virtual accessor does NOT apply to ``variables.*`` refs —
   866→           variables are user-defined and a missing key is a real error.
   867→        """
   868→        # Strip leading prefix only — NOT all occurrences. Stripping all
   869→        # occurrences of "steps." would corrupt paths whose intermediate
   870→        # dicts literally contain a key named "steps" (deep nesting).
   871→        stripped = expr.strip()
   872→        is_step_ref = False
   873→        for prefix in ("steps.", "pasos.", "variables."):
   874→            if stripped.startswith(prefix):
   875→                stripped = stripped[len(prefix) :]
   876→                # Only steps/pasos get the virtual "output" accessor;
   877→                # variables use strict key lookup.
   878→                is_step_ref = prefix in ("steps.", "pasos.")
   879→                break
   880→
   881→        parts = stripped.split(".")
   882→
   883→        current: Any = outputs
   884→        for raw_part in parts:
   885→            # Legacy ES translation per-segment
   886→            part = "output" if raw_part == "salida" else raw_part
   887→            if isinstance(current, dict):
   888→                if part in current:
   889→                    current = current[part]
   890→                elif is_step_ref and part == "output":
   891→                    # Virtual accessor: the current dict is itself the
   892→                    # step's output (stored raw). Skip this segment so
   893→                    # downstream segments (e.g. ".verdict") resolve against
   894→                    # the output dict directly.
   895→                    continue
   896→                else:
   897→                    return f"{{{{ {expr} }}}}"  # Leave template as-is if not found
   898→            else:
   899→                return f"{{{{ {expr} }}}}"
   900→
   901→        return current

   904→def _filter_internal_keys(outputs: dict[str, Any]) -> dict[str, Any]:
   905→    """Filter internal sentinel keys from outputs before returning to user.
   906→
   907→    These keys are used internally by the executor for control flow
   908→    (skip-until tracking, approved fingerprints, etc.) and should not
   909→    be exposed in PlaybookRunResult.outputs.
   910→    """
   911→    return {
   912→        k: v for k, v in outputs.items() if not k.startswith("__") and k != "_approved_fingerprints"
   913→    }
