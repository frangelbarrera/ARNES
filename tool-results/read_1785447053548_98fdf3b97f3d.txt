     1→"""
     2→ARNES Specialist — a pre-built, role-based agent with tool-use loop.
     3→
     4→A Specialist is a (system_prompt + tools + output_schema) bundle. The default
     5→`run()` implementation executes a ReAct-style tool-use loop:
     6→
     7→1. Format input as user message.
     8→2. Call LLM with tools registered.
     9→3. If LLM returns tool_calls, execute each tool and append results.
    10→4. Repeat until LLM returns final response (no tool_calls) or max_iterations.
    11→5. Validate response against output_schema (pydantic).
    12→6. Return structured result.
    13→
    14→Specialists are stateless. State lives in the Thread, not in the specialist.
    15→"""
    16→
    17→from __future__ import annotations
    18→
    19→import json
    20→from abc import ABC
    21→from collections.abc import AsyncIterator
    22→from typing import Any, ClassVar
    23→
    24→import structlog
    25→from pydantic import BaseModel, ConfigDict, Field, ValidationError
    26→
    27→from arnes.llm.base import LLMMessage, LLMProvider, LLMResponse, LLMUsage
    28→from arnes.middleware.cost_guard import BudgetExceeded
    29→from arnes.thread.events import AssistantMessageEvent
    30→from arnes.tools.base import Tool, ToolContext, ToolRegistry
    31→
    32→logger = structlog.get_logger(__name__)
    33→
    34→
    35→class SpecialistConfig(BaseModel):
    36→    """Configuration for a specialist."""
    37→
    38→    model_config = ConfigDict(arbitrary_types_allowed=True)
    39→
    40→    name: str  # e.g. "@planner"
    41→    description: str
    42→    system_prompt: str
    43→    tools: list[str] = Field(default_factory=list)  # tool names
    44→    output_schema: dict[str, Any] | None = None  # JSON schema for structured output
    45→    pydantic_model: type[BaseModel] | None = None  # Stronger than output_schema
    46→    default_model: str | None = None  # If set, overrides the global default
    47→    temperature: float = 0.0
    48→    max_tokens: int | None = None
    49→    max_iterations: int = 5  # ReAct loop limit
    50→
    51→
    52→class Specialist(ABC):
    53→    """Base class for all ARNES specialists.
    54→
    55→    To add a specialist:
    56→        class MySpecialist(Specialist):
    57→            config = SpecialistConfig(
    58→                name="@my-specialist",
    59→                description="Does X",
    60→                system_prompt="You are an expert in X...",
    61→                tools=["fs_read", "shell"],
    62→                output_schema={"type": "object", "required": ["result"]},
    63→            )
    64→    """
    65→
    66→    config: ClassVar[SpecialistConfig]
    67→
    68→    # Auto-registry
    69→    _registry: ClassVar[dict[str, type[Specialist]]] = {}
    70→
    71→    def __init_subclass__(cls, **kwargs: Any) -> None:
    72→        super().__init_subclass__(**kwargs)
    73→        if hasattr(cls, "config") and cls.config.name:
    74→            Specialist._registry[cls.config.name] = cls
    75→
    76→    async def run(
    77→        self,
    78→        input_data: dict[str, Any],
    79→        ctx: ToolContext,
    80→        *,
    81→        provider: LLMProvider,
    82→        tool_registry: ToolRegistry | None = None,
    83→    ) -> dict[str, Any]:
    84→        """Default specialist run: ReAct tool-use loop + schema validation.
    85→
    86→        Override this for custom logic if needed, but most specialists should
    87→        use the default implementation.
    88→        """
    89→        # Build initial messages
    90→        user_content = self._format_input(input_data)
    91→        messages: list[LLMMessage] = [
    92→            LLMMessage(role="system", content=self.config.system_prompt),
    93→            LLMMessage(role="user", content=user_content),
    94→        ]
    95→
    96→        # Get available tools (intersect config.tools with registry)
    97→        available_tools: list[Tool] = []
    98→        tool_schemas: list[dict[str, Any]] = []
    99→        if tool_registry and self.config.tools:
   100→            for tool_name in self.config.tools:
   101→                tool = tool_registry.get(tool_name)
   102→                if tool:
   103→                    available_tools.append(tool)
   104→                    tool_schemas.append(self._tool_to_schema(tool))
   105→
   106→        # Build middleware-wrapped provider
   107→        # The caller (Harness or PlaybookExecutor) is responsible for wrapping
   108→        # the provider with the full middleware stack (CostGuard → Verification →
   109→        # TokenOptimizer → provider). The specialist should NOT re-wrap.
   110→        #
   111→        # We detect already-wrapped providers by checking for the _arnes_wrapped
   112→        # marker attribute that all our middleware classes set.
   113→        wrapped_provider: LLMProvider = provider
   114→
   115→        # If the provider is not already wrapped, wrap it with the full stack.
   116→        # This is a safety net — in normal usage, the caller always wraps first.
   117→        if not getattr(provider, "_arnes_wrapped", False):
   118→            from arnes.middleware.cost_guard import CostBudget, CostGuard
   119→            from arnes.middleware.token_optimizer import TokenOptimizer
   120→            from arnes.middleware.verification import VerificationConfig, VerificationLayer
   121→
   122→            inner: LLMProvider = TokenOptimizer(provider, enable_cache=True)
   123→            if self.config.output_schema or self.config.pydantic_model:
   124→                inner = VerificationLayer(
   125→                    inner,
   126→                    VerificationConfig(structured_outputs=True, refusal_pattern=True),
   127→                )
   128→            wrapped_provider = CostGuard(inner, budget=CostBudget(task_budget_usd=1.0))
   129→
   130→        # ReAct tool-use loop
   131→        total_usage = LLMUsage()
   132→        all_tool_results: list[dict[str, Any]] = []
   133→        model = self.config.default_model or "ollama/llama3.2"
   134→
   135→        # Derive the response_schema sent to middleware: prefer the explicit
   136→        # output_schema, but fall back to a pydantic_model's JSON schema so that
   137→        # specialists that only declare a `pydantic_model` still get JSON-mode
   138→        # forcing AND schema validation in the VerificationLayer. Without this,
   139→        # `response_schema=None` would silently disable both.
   140→        effective_response_schema = self.config.output_schema
   141→        if effective_response_schema is None and self.config.pydantic_model is not None:
   142→            effective_response_schema = self.config.pydantic_model.model_json_schema()
   143→
   144→        wants_json = bool(self.config.output_schema or self.config.pydantic_model)
   145→
   146→        # Track whether the loop produced a *final* response (no tool_calls).
   147→        # If the LLM keeps calling tools until max_iterations is hit, the last
   148→        # `response` will still have tool_calls and we must NOT validate it as
   149→        # if it were the final answer — that would surface an empty/intermediate
   150→        # tool-call payload as a malformed "final" response.
   151→        final_response: LLMResponse | None = None
   152→        response: LLMResponse | None = None
   153→
   154→        for iteration in range(self.config.max_iterations):
   155→            try:
   156→                response = await wrapped_provider.complete(
   157→                    messages,
   158→                    model=model,
   159→                    tools=tool_schemas if tool_schemas else None,
   160→                    temperature=self.config.temperature,
   161→                    max_tokens=self.config.max_tokens,
   162→                    response_format={"type": "json_object"} if wants_json else None,
   163→                    response_schema=effective_response_schema,
   164→                    interactive=ctx.metadata.get("interactive", False),
   165→                )
   166→            except BudgetExceeded as e:
   167→                logger.error(
   168→                    "specialist_budget_exceeded", specialist=self.config.name, error=str(e)
   169→                )
   170→                return {
   171→                    "specialist": self.config.name,
   172→                    "success": False,
   173→                    "error": f"Budget exceeded: {e}",
   174→                    "budget_exceeded": True,
   175→                }
   176→
   177→            total_usage = total_usage + response.usage
   178→
   179→            # Emit an AssistantMessageEvent for every LLM call so the
   180→            # conversation history and per-call token/cost are observable
   181→            # in the thread bitácora. The specialist has direct access to
   182→            # ctx.thread_id and ctx.step_id, so we can construct the event
   183→            # with the correct ids (no nil-UUID patching needed). The
   184→            # event is appended to the wrapped provider's shared event sink
   185→            # (CostGuard._events), which the PlaybookExecutor drains after
   186→            # the step completes.
   187→            self._emit_assistant_message(wrapped_provider, ctx, response, model)
   188→
   189→            # If no tool calls, we have the final response
   190→            if not response.tool_calls:
   191→                final_response = response
   192→                break
   193→
   194→            # Execute each tool call
   195→            messages.append(
   196→                LLMMessage(
   197→                    role="assistant",
   198→                    content=response.content,
   199→                    tool_calls=response.tool_calls,
   200→                )
   201→            )
   202→
   203→            for tc in response.tool_calls:
   204→                tool_result = await self._execute_tool_call(tc, available_tools, ctx)
   205→                all_tool_results.append(tool_result)
   206→                messages.append(
   207→                    LLMMessage(
   208→                        role="tool",
   209→                        content=json.dumps(tool_result, default=str),
   210→                        tool_call_id=tc.get("id"),
   211→                        name=tc.get("function", {}).get("name"),
   212→                    )
   213→                )
   214→
   215→            # Continue loop for next iteration
   216→
   217→        # If we exited the loop without a final (tool-call-free) response,
   218→        # the LLM kept calling tools past max_iterations. Return a clear error
   219→        # instead of trying to validate an empty / intermediate tool-call payload.
   220→        if final_response is None:
   221→            max_iter = self.config.max_iterations
   222→            logger.error(
   223→                "specialist_max_iterations_exceeded",
   224→                specialist=self.config.name,
   225→                max_iterations=max_iter,
   226→            )
   227→            return {
   228→                "specialist": self.config.name,
   229→                "success": False,
   230→                "error": (
   231→                    f"Specialist exceeded max_iterations ({max_iter}) "
   232→                    "without producing a final response"
   233→                ),
   234→                "raw": response.content if response is not None else None,
   235→                "usage": total_usage.model_dump(),
   236→                "tool_results": all_tool_results,
   237→            }
   238→
   239→        # Validate output against schema
   240→        result = self._parse_and_validate_output(final_response, total_usage, all_tool_results)
   241→        return result
   242→
   243→    async def stream(
   244→        self,
   245→        input_data: dict[str, Any],
   246→        ctx: ToolContext,
   247→        *,
   248→        provider: LLMProvider,
   249→        tool_registry: ToolRegistry | None = None,
   250→    ) -> AsyncIterator[LLMResponse]:
   251→        """Stream a specialist's response token by token.
   252→
   253→        Mirrors :meth:`run` but uses ``provider.stream_complete()`` instead
   254→        of ``provider.complete()``. Yields ``LLMResponse`` chunks as they
   255→        arrive from the provider. The final chunk carries the full
   256→        ``LLMUsage`` (``tokens_in``, ``tokens_out``, ``cost_usd``).
   257→
   258→        Streaming specialists do NOT execute the ReAct tool-use loop — if
   259→        the specialist needs tools, callers should use :meth:`run` instead.
   260→        This is a best-effort streaming path for read-only / generation
   261→        tasks (planning, summarising, drafting code without shell access).
   262→
   263→        Audit trail: after the stream completes, a single
   264→        :class:`AssistantMessageEvent` is appended to the wrapped
   265→        provider's ``_events`` sink (same pattern as :meth:`run`). The
   266→        event carries the full accumulated content and the final usage,
   267→        so the bitácora records streaming runs the same way it records
   268→        non-streaming runs. Streaming produces ONE event per call (not
   269→        per-chunk events) — per-chunk events would balloon the audit log
   270→        without adding forensic value, and the chunks themselves are not
   271→        durable (they're a transport optimisation, not a semantic unit).
   272→
   273→        Usage::
   274→
   275→            async for chunk in specialist.stream(input_data, ctx, provider=p):
   276→                print(chunk.content, end="", flush=True)
   277→        """
   278→        # Build messages (same as run(), minus the tool-use loop)
   279→        user_content = self._format_input(input_data)
   280→        messages: list[LLMMessage] = [
   281→            LLMMessage(role="system", content=self.config.system_prompt),
   282→            LLMMessage(role="user", content=user_content),
   283→        ]
   284→
   285→        # Wrap provider with the full middleware stack if not already wrapped
   286→        # (mirrors run()).
   287→        wrapped_provider: LLMProvider = provider
   288→        if not getattr(provider, "_arnes_wrapped", False):
   289→            from arnes.middleware.cost_guard import CostBudget, CostGuard
   290→            from arnes.middleware.token_optimizer import TokenOptimizer
   291→            from arnes.middleware.verification import VerificationConfig, VerificationLayer
   292→
   293→            inner: LLMProvider = TokenOptimizer(provider, enable_cache=True)
   294→            if self.config.output_schema or self.config.pydantic_model:
   295→                inner = VerificationLayer(
   296→                    inner,
   297→                    VerificationConfig(structured_outputs=True, refusal_pattern=True),
   298→                )
   299→            wrapped_provider = CostGuard(inner, budget=CostBudget(task_budget_usd=1.0))
   300→
   301→        model = self.config.default_model or "ollama/llama3.2"
   302→        effective_response_schema = self.config.output_schema
   303→        if effective_response_schema is None and self.config.pydantic_model is not None:
   304→            effective_response_schema = self.config.pydantic_model.model_json_schema()
   305→        wants_json = bool(self.config.output_schema or self.config.pydantic_model)
   306→
   307→        # Accumulate content + usage so we can emit a single
   308→        # AssistantMessageEvent after the stream ends. Per-chunk usage is
   309→        # zero on intermediate chunks (see LLMProvider.stream_complete
   310→        # contract); the final chunk carries the full usage.
   311→        accumulated_content: list[str] = []
   312→        total_usage = LLMUsage()
   313→
   314→        async for chunk in wrapped_provider.stream_complete(
   315→            messages,
   316→            model=model,
   317→            temperature=self.config.temperature,
   318→            max_tokens=self.config.max_tokens,
   319→            response_format={"type": "json_object"} if wants_json else None,
   320→            response_schema=effective_response_schema,
   321→        ):
   322→            if chunk.content:
   323→                accumulated_content.append(chunk.content)
   324→            # The last non-zero usage wins (providers send the full count
   325→            # on the final chunk, not a running delta).
   326→            if chunk.usage.tokens_in > 0 or chunk.usage.tokens_out > 0 or chunk.usage.cost_usd > 0:
   327→                total_usage = chunk.usage
   328→            yield chunk
   329→
   330→        # Emit a single AssistantMessageEvent for the bitácora. We
   331→        # construct a synthetic LLMResponse carrying the accumulated
   332→        # content + final usage so we can reuse the same _emit_assistant_message
   333→        # helper that run() uses (DRY: same event shape, same sink).
   334→        synthetic_response = LLMResponse(
   335→            content="".join(accumulated_content),
   336→            tool_calls=[],
   337→            usage=total_usage,
   338→            model=model,
   339→        )
   340→        self._emit_assistant_message(wrapped_provider, ctx, synthetic_response, model)
   341→

   346→    async def _execute_tool_call(
   347→        self,
   348→        tool_call: dict[str, Any],
   349→        available_tools: list[Tool],
   350→        ctx: ToolContext,
   351→    ) -> dict[str, Any]:
   352→        """Execute a single tool call from the LLM."""
   353→        function = tool_call.get("function", {})
   354→        tool_name = function.get("name", "")
   355→        args_str = function.get("arguments", "{}")
   356→
   357→        try:
   358→            args = json.loads(args_str) if isinstance(args_str, str) else args_str
   359→        except json.JSONDecodeError:
   360→            return {
   361→                "tool": tool_name,
   362→                "success": False,
   363→                "error": f"Invalid JSON arguments: {args_str}",
   364→            }
   365→
   366→        # Find the tool
   367→        tool = next((t for t in available_tools if t.name == tool_name), None)
   368→        if not tool:
   369→            return {
   370→                "tool": tool_name,
   371→                "success": False,
   372→                "error": f"Tool '{tool_name}' not available",
   373→            }
   374→
   375→        # HITL check: if tool requires approval, fingerprint args and
   376→        # compare against any pre-approved fingerprint (rug-pull defense).
   377→        # NOTE: use setdefault (not .get) so the SAME dict object is stored
   378→        # back into ctx.metadata — otherwise approvals are never persisted
   379→        # and the rug-pull detector is defeated.
   380→        if tool.requires_approval:
   381→            fingerprint = Tool.fingerprint(args)
   382→            approved_fingerprints = ctx.metadata.setdefault("_approved_fingerprints", {})
   383→
   384→            if fingerprint in approved_fingerprints:
   385→                # Pre-approved: execute
   386→                logger.info(
   387→                    "tool_approved_fingerprint_match",
   388→                    tool=tool_name,
   389→                    fingerprint=fingerprint,
   390→                )
   391→            elif not ctx.metadata.get("interactive", False):
   392→                # Non-interactive and not pre-approved: auto-reject
   393→                return {
   394→                    "tool": tool_name,
   395→                    "success": False,
   396→                    "error": f"Tool '{tool_name}' requires human approval. Set interactive=True.",
   397→                    "fingerprint": fingerprint,
   398→                }
   399→            else:
   400→                # Interactive: prompt for approval
   401→                approval = await self._request_human_approval(tool_name, args, fingerprint, ctx)
   402→                if not approval:
   403→                    return {
   404→                        "tool": tool_name,
   405→                        "success": False,
   406→                        "error": f"Human rejected tool call for '{tool_name}'.",
   407→                        "fingerprint": fingerprint,
   408→                    }
   409→                # Record the approved fingerprint so we can detect rug-pull
   410→                # if the LLM tries to call the same tool with different args
   411→                approved_fingerprints[fingerprint] = tool_name
   412→
   413→        try:
   414→            result = await tool.execute(args, ctx)
   415→            return {
   416→                "tool": tool_name,
   417→                "success": result.success,
   418→                "output": result.output,
   419→                "error": result.error,
   420→            }
   421→        except Exception as e:
   422→            logger.exception("tool_execution_failed", tool=tool_name, error=str(e))
   423→            return {
   424→                "tool": tool_name,
   425→                "success": False,
   426→                "error": str(e),
   427→            }
   428→
   429→    async def _request_human_approval(
   430→        self,
   431→        tool_name: str,
   432→        args: dict[str, Any],
   433→        fingerprint: str,
   434→        ctx: ToolContext,
   435→    ) -> bool:
   436→        """Request human approval for a tool call. Returns True if approved."""
   437→        try:
   438→            from rich.console import Console
   439→            from rich.prompt import Confirm
   440→
   441→            console = Console()
   442→            console.print(f"\n[yellow]⚠ Approval required for tool:[/yellow] {tool_name}")
   443→            console.print(f"  [dim]Args fingerprint:[/dim] {fingerprint}")
   444→            console.print(f"  [dim]Args:[/dim] {json.dumps(args, indent=2, default=str)}")
   445→            return Confirm.ask("  Approve this tool call?", default=False)
   446→        except ImportError:
   447→            logger.warning("rich not available — auto-rejecting tool call")
   448→            return False

   451→    # Event emission

   453→
   454→    def _emit_assistant_message(
   455→        self,
   456→        wrapped_provider: Any,
   457→        ctx: ToolContext,
   458→        response: LLMResponse,
   459→        model: str,
   460→    ) -> None:
   461→        """Emit an ``AssistantMessageEvent`` for a single LLM call.
   462→
   463→        The event carries the response content, the model used, and the
   464→        per-call token usage and cost so that the conversation history is
   465→        fully observable in the thread bitácora. The event is appended to
   466→        the wrapped provider's shared ``_events`` sink (set up by
   467→        ``CostGuard``); the ``PlaybookExecutor`` drains that sink after each
   468→        step and appends the events to the ``Thread``.
   469→
   470→        If ``wrapped_provider`` has no ``_events`` attribute (e.g. a raw
   471→        third-party provider), the emission is a no-op — the executor will
   472→        still record step-level tokens/cost on the ``StepCompletedEvent``.
   473→        """
   474→        events_list = getattr(wrapped_provider, "_events", None)
   475→        if not isinstance(events_list, list):
   476→            return
   477→
   478→        event = AssistantMessageEvent(
   479→            thread_id=ctx.thread_id,
   480→            step_id=ctx.step_id,
   481→            specialist=self.config.name,
   482→            data={
   483→                "content": response.content,
   484→                "model": response.usage.model or model,
   485→                "tokens_in": response.usage.tokens_in,
   486→                "tokens_out": response.usage.tokens_out,
   487→                "cost_usd": response.usage.cost_usd,
   488→                "cached": response.usage.cached,
   489→            },
   490→        )
   491→        events_list.append(event)
   492→
   493→    # ============================================================
   494→    # Schema validation
   495→    # ============================================================
   496→
   497→    def _parse_and_validate_output(
   498→        self,
   499→        response: LLMResponse,
   500→        total_usage: LLMUsage,
   501→        tool_results: list[dict[str, Any]],
   502→    ) -> dict[str, Any]:
   503→        """Parse LLM response and validate against pydantic_model or output_schema."""
   504→        if not response.content:
   505→            return {
   506→                "specialist": self.config.name,
   507→                "success": False,
   508→                "error": "Empty response from LLM",
   509→                "raw": None,
   510→                "usage": total_usage.model_dump(),
   511→                "tool_results": tool_results,
   512→            }
   513→
   514→        # Strip ```json ... ``` / ``` ... ``` fences that some models (notably
   515→        # Llama 3.2) wrap around a JSON payload despite being asked for raw JSON.
   516→        cleaned = self._clean_json_response(response.content)
   517→
   518→        # Try JSON parse
   519→        try:
   520→            parsed = json.loads(cleaned)
   521→        except json.JSONDecodeError:
   522→            # If we expected JSON, this is a failure
   523→            if self.config.output_schema or self.config.pydantic_model:
   524→                return {
   525→                    "specialist": self.config.name,
   526→                    "success": False,
   527→                    "error": f"LLM did not return valid JSON. Got: {response.content[:200]}",
   528→                    "raw": response.content,
   529→                    "usage": total_usage.model_dump(),
   530→                    "tool_results": tool_results,
   531→                }
   532→            # If no schema expected, return raw content
   533→            parsed = {"raw": response.content}
   534→
   535→        # Strong validation with pydantic model if defined (preferred over
   536→        # the weak JSON-schema `required`-fields check below — pydantic gives
   537→        # us type coercion + full validation).
   538→        if self.config.pydantic_model:
   539→            try:
   540→                validated = self.config.pydantic_model.model_validate(parsed)
   541→                return {
   542→                    "specialist": self.config.name,
   543→                    "success": True,
   544→                    "output": validated.model_dump(),
   545→                    "usage": total_usage.model_dump(),
   546→                    "tool_results": tool_results,
   547→                }
   548→            except ValidationError as e:
   549→                return {
   550→                    "specialist": self.config.name,
   551→                    "success": False,
   552→                    "error": f"Output schema validation failed: {e}",
   553→                    "raw": parsed,
   554→                    "usage": total_usage.model_dump(),
   555→                    "tool_results": tool_results,
   556→                }
   557→
   558→        # Weak validation with JSON schema (required fields only)
   559→        if self.config.output_schema:
   560→            required = self.config.output_schema.get("required", [])
   561→            missing = [f for f in required if f not in parsed]
   562→            if missing:
   563→                return {
   564→                    "specialist": self.config.name,
   565→                    "success": False,
   566→                    "error": f"Missing required fields: {missing}",
   567→                    "raw": parsed,
   568→                    "usage": total_usage.model_dump(),
   569→                    "tool_results": tool_results,
   570→                }
   571→
   572→        return {
   573→            "specialist": self.config.name,
   574→            "success": True,
   575→            "output": parsed,
   576→            "usage": total_usage.model_dump(),
   577→            "tool_results": tool_results,
   578→        }

   592→    @staticmethod
   593→    def _clean_json_response(content: str) -> str:
   594→        """Strip markdown code-fences wrappers from an LLM JSON response.
   595→
   596→        Many local models (Llama 3.2 in particular) ignore a ``response_format``
   597→        hint and wrap their JSON in ```` ```json ... ``` ```` or plain ```` ``` ... ``` ````.
   598→        ``json.loads`` can't parse that, so peel the fences off before parsing.
   599→
   600→        Only strips the *outer* fence — JSON string values that legitimately
   601→        contain ``` are left untouched because we only trim a leading fence
   602→        and a trailing fence, never interior backticks.
   603→        """
   604→        # ``content`` is typed as ``str``; the isinstance guard below is a
   605→        # defensive runtime check for callers that bypass the type system
   606→        # (e.g. third-party providers returning bytes). It is unreachable per
   607→        # mypy but kept as defense-in-depth.
   608→        if not isinstance(content, str):
   609→            return content  # type: ignore[unreachable]
   610→        text = content.strip()
   611→        # Match an opening fence ```` ```json ```` / ```` ```jsonYAML ```` / ```` ``` ````
   612→        # followed by content and a closing ```` ``` ````. Non-greedy so we grab
   613→        # the smallest outer fence.
   614→        if text.startswith("```"):
   615→            # Drop the opening fence line (and any language tag after it).
   616→            first_newline = text.find("\n")
   617→            # Single-line fence like ```{} ``` (no newline) → strip 3 backticks
   618→            # either side; otherwise slice off the whole opening fence line.
   619→            text = text[first_newline + 1 :] if first_newline != -1 else text[3:]
   620→            # Strip a trailing ``` if present.
   621→            if text.endswith("```"):
   622→                text = text[:-3]
   623→        return text.strip()

   625→    def _tool_to_schema(self, tool: Tool) -> dict[str, Any]:
   626→        """Convert an ARNES Tool to OpenAI tool schema for LLM."""
   627→        args_schema = getattr(tool, "Args", None)
   628→        return {
   629→            "type": "function",
   630→            "function": {
   631→                "name": tool.name,
   632→                "description": tool.description,
   633→                "parameters": args_schema.model_json_schema()
   634→                if args_schema
   635→                else {"type": "object", "properties": {}},
   636→            },
   637→        }

   640→class SpecialistRegistry:
   641→    """Registry of available specialists."""
   642→
   643→    def __init__(self) -> None:
   644→        self._specialists: dict[str, Specialist] = {}
   645→
   646→    def register(self, specialist: Specialist) -> None:
   647→        if not specialist.config.name:
   648→            raise ValueError("Specialist must have a name in its config")
   649→        self._specialists[specialist.config.name] = specialist
   650→
   651→    def register_class(self, specialist_class: type[Specialist]) -> None:
   652→        instance = specialist_class()
   653→        self.register(instance)
   654→
   655→    def get(self, name: str) -> Specialist | None:
   656→        if not name.startswith("@"):
   657→            name = "@" + name
   658→        return self._specialists.get(name)
   659→
   660→    def list_names(self) -> list[str]:
   661→        """Return a sorted list of registered specialist names."""
   662→        return sorted(self._specialists.keys())
   663→
   664→    def has(self, name: str) -> bool:
   665→        return self.get(name) is not None
   666→
   667→    def configs(self) -> list[SpecialistConfig]:
   668→        return [s.config for s in self._specialists.values()]
   669→
   670→
   671→def get_default_specialist_registry() -> SpecialistRegistry:
   672→    """Return a registry with all built-in specialists registered."""
   673→    registry = SpecialistRegistry()
   674→    from arnes.specialists.coder import Coder
   675→    from arnes.specialists.debugger import Debugger
   676→    from arnes.specialists.planner import Planner
   677→    from arnes.specialists.reviewer import Reviewer
   678→    from arnes.specialists.tester import Tester
   679→
   680→    for cls in [Planner, Coder, Reviewer, Tester, Debugger]:
   681→        registry.register_class(cls)
   682→    return registry
