   189→@cli.command()

   194→def stream(specialist: str, task: str, model: str, mock: bool) -> None:
   195→    """Stream a specialist's response token by token.
   196→
   197→    Example:
   198→        arnes stream @planner --task "Plan a blog post"
   199→        arnes stream @planner --task "Plan a blog post" --mock
   200→    """
   201→    asyncio.run(_stream_specialist(specialist, task, model, mock))
   202→
   203→
   204→async def _stream_specialist(specialist: str, task: str, model: str, mock: bool) -> None:
   205→    """Stream a specialist's response."""
   206→    from arnes import Harness, HarnessConfig
   207→
   208→    if mock or model.startswith("mock/"):
   209→        provider: LLMProvider = _SchemaValidMockLLMProvider()
   210→    else:
   211→        provider = get_provider(model)
   212→
   213→    harness = Harness(
   214→        config=HarnessConfig(model=model, budget_usd=0.50),
   215→        provider=provider,
   216→    )
   217→
   218→    # Normalize specialist name
   219→    if not specialist.startswith("@"):
   220→        specialist = "@" + specialist
   221→
   222→    # Check specialist exists before streaming
   223→    specialist_obj = harness.specialist_registry.get(specialist)
   224→    if not specialist_obj:
   225→        available = harness.specialist_registry.list_names()
   226→        console.print(f"[red]✗[/red] Specialist '{specialist}' not found. Available: {available}")
   227→        sys.exit(1)
   228→
   229→    console.print(f"[cyan]Streaming[/cyan] {specialist}...")
   230→    console.print("[dim]---[/dim]")
   231→
   232→    total_in = 0
   233→    total_out = 0
   234→    total_cost = 0.0
   235→    chunks_received = 0
   236→    chunks_list = []  # Store chunks for bitácora (single pass, no double-call)
   237→
   238→    async for chunk in harness.stream(specialist, {"task": task}):
   239→        chunks_received += 1
   240→        chunks_list.append(chunk)
   241→        if chunk.content:
   242→            console.print(chunk.content, end="", style="white")
   243→        total_in += chunk.usage.tokens_in
   244→        total_out += chunk.usage.tokens_out
   245→        total_cost += chunk.usage.cost_usd
   246→
   247→    if chunks_received == 0:
   248→        console.print("[yellow]No response received.[/yellow]")
   249→        sys.exit(1)
   250→
   251→    console.print()
   252→    console.print("[dim]---[/dim]")
   253→    console.print(f"[dim]Tokens: {total_in} in, {total_out} out. Cost: ${total_cost:.4f}[/dim]")
   254→
   255→    # Save bitácora from the accumulated chunks (no second LLM call)
   256→    ts = datetime.now().strftime("%Y%m%d-%H%M%S")
   257→    bitacora_path = f"bitacora-stream-{specialist.lstrip('@')}-{ts}.md"
   258→
   259→    audit_content = f"# ARNES Stream Bitácora — {specialist}\n\n"
   260→    audit_content += f"**Timestamp:** {ts}\n"
   261→    audit_content += f"**Task:** {task}\n\n"
   262→    audit_content += "## Response\n\n"
   263→    audit_content += "```json\n"
   264→    audit_content += "".join(c.content for c in chunks_list)
   265→    audit_content += "\n```\n\n"
   266→    audit_content += "## Usage\n\n"
   267→    audit_content += f"- Tokens in: {total_in}\n"
   268→    audit_content += f"- Tokens out: {total_out}\n"
   269→    audit_content += f"- Cost: ${total_cost:.4f}\n"
   270→
   271→    Path(bitacora_path).write_text(audit_content, encoding="utf-8")
   272→    console.print(f"\n[cyan]Bitácora saved to:[/cyan] {bitacora_path}")
   273→

   314→

   319→    Examples:
   320→

   322→        arnes benchmark --seeds 3                # 3 seeds per playbook
   323→        arnes benchmark --concurrent 4           # 4 playbooks at once
   324→        arnes benchmark --seeds 5 --concurrent 4 # 5 seeds, 4-way parallel
   325→    """
   326→    asyncio.run(_run_benchmark(seeds, concurrent, manuals_dir, output))

   328→
   329→async def _run_benchmark(seeds: int, concurrent: int, manuals_dir: str | None, output: str) -> None:
   330→    """Run the basic benchmark suite and print/save results.
   331→
   332→    Pulls in :class:`arnes.benchmarks.BenchmarkRunner` lazily so the
   333→    CLI startup cost stays low — ``arnes --version`` doesn't pay for
   334→    importing the benchmark harness.
   335→    """
   336→    from pathlib import Path
   337→
   338→    from arnes.benchmarks import BenchmarkRunner
   339→    from arnes.benchmarks.suites.basic import BasicBenchmarkSuite
   340→
   341→    suite_manuals = Path(manuals_dir) if manuals_dir else None
   342→    suite = BasicBenchmarkSuite(manuals_dir=suite_manuals)
   343→
   344→    playbooks = suite.playbooks()
   345→    if not playbooks:
   346→        manuals_display = manuals_dir or "<repo>/manuals"
   347→        console.print(f"[red]✗[/red] No playbooks found in {manuals_display}")
   348→        sys.exit(1)
   349→
   350→    console.print(
   351→        Panel.fit(
   352→            f"[bold cyan]ARNES[/bold cyan] — Benchmark suite: {suite.name}\n"
   353→            f"  [dim]Playbooks:[/dim] {len(playbooks)}\n"
   354→            f"  [dim]Seeds:[/dim] {seeds}\n"
   355→            f"  [dim]Concurrent:[/dim] {concurrent}\n"
   356→            f"  [dim]Total runs:[/dim] {len(playbooks) * seeds}",
   357→            border_style="cyan",
   358→        )
   359→    )
   360→
   361→    runner = BenchmarkRunner()
   362→    seed_values = list(range(seeds))
   363→    with console.status("[cyan]Running benchmarks...[/cyan]"):
   364→        results = await runner.run_suite(suite, seeds=seed_values, concurrent=concurrent)
   365→
   366→    # Print results as a rich table.
   367→    table = Table(title=f"Benchmark Results — {suite.name} suite")
   368→    table.add_column("Playbook", style="cyan")
   369→    table.add_column("Runs", justify="right")
   370→    table.add_column("Success", justify="right")
   371→    table.add_column("Avg dur (s)", justify="right")
   372→    table.add_column("P95 dur (s)", justify="right")
   373→    table.add_column("Avg tokens", justify="right")
   374→    table.add_column("Avg cost", justify="right")
   375→
   376→    for m in results.per_playbook:
   377→        avg_tokens = m.avg_tokens_in + m.avg_tokens_out
   378→        table.add_row(
   379→            m.playbook_name,
   380→            str(m.runs),
   381→            f"{m.success_rate:.0%}",
   382→            f"{m.avg_duration_s:.4f}",
   383→            f"{m.p95_duration_s:.4f}",
   384→            str(avg_tokens),
   385→            f"${m.avg_cost_usd:.6f}",
   386→        )
   387→
   388→    console.print(table)
   389→    console.print(
   390→        f"\n[bold]Overall:[/bold] success={results.overall_success_rate:.0%}, "
   391→        f"avg_dur={results.overall_avg_duration_s:.4f}s, "
   392→        f"avg_tokens={results.overall_avg_tokens_in + results.overall_avg_tokens_out}, "
   393→        f"avg_cost=${results.overall_avg_cost_usd:.6f}"
   394→    )
   395→
   396→    # Save JSON results.
   397→    Path(output).write_text(results.to_json(), encoding="utf-8")
   398→    console.print(f"\n[cyan]Results saved to:[/cyan] {output}")

   461→async def _run_playbook(
   462→    playbook_path: str,
   463→    model: str,
   464→    budget: float,
   465→    mock: bool,
   466→    interactive: bool,
   467→    output: str | None,
   468→    stream: bool = False,
   469→) -> None:
   470→    """Execute a playbook and print results.
   471→
   472→    When ``stream=True``, uses :meth:`PlaybookExecutor.stream` to yield
   473→    step-level events as each step completes (instead of buffering the
   474→    whole run behind a spinner). The final ``PlaybookRunResult`` is
   475→    captured from the last yield for stats + bitácora persistence.
   476→    """
   477→    # Compile
   478→    try:
   479→        playbook = PlaybookCompiler.from_file(playbook_path)
   480→    except PlaybookCompileError as e:
   481→        console.print(f"[red]✗ Compile error:[/red]\n{e}")
   482→        sys.exit(1)
   483→
   484→    # The schema validator guarantees metadata is non-None after compile.
   485→    assert playbook.metadata is not None
   486→
   487→    console.print(
   488→        Panel.fit(
   489→            f"[bold cyan]ARNES[/bold cyan] — Executing playbook\n"
   490→            f"  [dim]Name:[/dim] {playbook.metadata.name}\n"
   491→            f"  [dim]Objective:[/dim] {playbook.metadata.objective}\n"
   492→            f"  [dim]Model:[/dim] {model}\n"
   493→            f"  [dim]Budget:[/dim] ${budget:.2f}",
   494→            border_style="cyan",
   495→        )
   496→    )
   497→
   498→    # Setup provider
   499→    if mock or model.startswith("mock/"):
   500→        provider: LLMProvider = _SchemaValidMockLLMProvider()
   501→    else:
   502→        provider = get_provider(model)
   503→
   504→    # Execute
   505→    executor = PlaybookExecutor(
   506→        provider=provider,
   507→        cost_budget=CostBudget(task_budget_usd=budget),
   508→        interactive=interactive,
   509→    )
   510→
   511→    if stream:
   512→        result = await _run_playbook_streaming(executor, playbook)
   513→    else:
   514→        with console.status("[cyan]Executing...[/cyan]"):
   515→            result = await executor.run(playbook)
   516→
   517→    # Print results
   518→    if result.success:
   519→        console.print("\n[green]✅ Manual executed[/green]")
   520→    else:
   521→        console.print("\n[red]❌ Execution failed[/red]")
   522→        if result.error:
   523→            console.print(f"  [red]Error:[/red] {result.error}")
   524→
   525→    # Stats
   526→    console.print(f"\n[dim]Steps executed:[/dim] {result.steps_executed}")
   527→    console.print(f"[dim]Steps failed:[/dim] {result.steps_failed}")
   528→    console.print(f"[dim]Duration:[/dim] {result.duration_s:.2f}s")
   529→    console.print(f"[dim]Tokens in/out:[/dim] {result.total_tokens_in}/{result.total_tokens_out}")
   530→    console.print(f"[dim]Total cost:[/dim] ${result.total_cost_usd:.4f}")
   531→
   532→    # Save bitácora
   533→    if output:
   534→        Path(output).write_text(result.to_markdown(), encoding="utf-8")
   535→        console.print(f"\n[cyan]Bitácora saved to:[/cyan] {output}")
   536→    else:
   537→        ts = datetime.now().strftime("%Y%m%d-%H%M%S")
   538→        default_path = f"bitacora-{playbook.metadata.name}-{ts}.md"
   539→        Path(default_path).write_text(result.to_markdown(), encoding="utf-8")
   540→        console.print(f"\n[cyan]Bitácora saved to:[/cyan] {default_path}")
   541→
   542→
   543→async def _run_playbook_streaming(
   544→    executor: PlaybookExecutor,
   545→    playbook: Playbook,
   546→) -> PlaybookRunResult:
   547→    """Run a playbook in streaming mode, printing step events as they arrive.
   548→
   549→    Uses :meth:`PlaybookExecutor.stream` to yield ``Event`` objects as each
   550→    step completes, then captures the final ``PlaybookRunResult`` from the
   551→    last yield. Best-effort: parallel branches stream in completion order,
   552→    not definition order.
   553→    """
   554→    from arnes.thread.events import (
   555→        RunCompletedEvent,
   556→        RunFailedEvent,
   557→        StepCompletedEvent,
   558→        StepFailedEvent,
   559→    )
   560→
   561→    final_result: PlaybookRunResult | None = None
   562→    console.print("[cyan]Streaming...[/cyan]\n[dim]---[/dim]")
   563→
   564→    async for item in executor.stream(playbook):
   565→        if isinstance(item, PlaybookRunResult):
   566→            final_result = item
   567→            break
   568→        # Print step-level transitions as they happen
   569→        if isinstance(item, StepCompletedEvent):
   570→            console.print(f"  [green]✓[/green] step_completed: {item.step_id}")
   571→        elif isinstance(item, StepFailedEvent):
   572→            console.print(
   573→                f"  [red]✗[/red] step_failed: {item.step_id} — {item.data.get('error', '')}"
   574→            )
   575→        elif isinstance(item, RunCompletedEvent):
   576→            console.print("  [green]✓[/green] run_completed")
   577→        elif isinstance(item, RunFailedEvent):
   578→            console.print(f"  [red]✗[/red] run_failed — {item.data.get('error', '')}")
   579→        else:
   580→            # Other event types (StepStarted, AssistantMessage, CostThreshold, etc.)
   581→            # are not printed to keep the streaming output focused on step transitions.
   582→            console.print(f"  [dim]·[/dim] {item.type.value}")
   583→
   584→    console.print("[dim]---[/dim]")
   585→    assert final_result is not None, "stream() must yield a final PlaybookRunResult"
   586→    return final_result

   589→class _SchemaValidMockLLMProvider(LLMProvider):
   590→    """Mock LLM provider that returns schema-valid JSON for each specialist.
   591→
   592→    Used by `arnes run --mock` for testing without network calls.
   593→    Detects which specialist is being invoked based on system prompt content.
   594→    """
   595→
   596→    def __init__(self) -> None:
   597→        self.call_count = 0
   598→
   599→    async def complete(
   600→        self,
   601→        messages: list[LLMMessage],
   602→        *,
   603→        model: str = "mock",
   604→        tools: list[dict[str, Any]] | None = None,
   605→        temperature: float = 0.0,
   606→        max_tokens: int | None = None,
   607→        response_format: dict[str, Any] | None = None,
   608→        response_schema: dict[str, Any] | None = None,
   609→        **kwargs: Any,
   610→    ) -> LLMResponse:
   611→        self.call_count += 1
   612→
   613→        # Detect specialist from system prompt
   614→        sys_msg = next((m for m in messages if m.role == "system"), None)
   615→        sys_content = sys_msg.content if sys_msg else ""
   616→
   617→        if "@planner" in sys_content:
   618→            content = '{"steps": [{"id": "s1", "specialist": "@coder", "input": {}}]}'
   619→        elif "@coder" in sys_content:
   620→            content = '{"files": [{"path": "out.py", "language": "python", "content": "# Mock code\\npass"}], "summary": "Mock implementation", "assumptions": [], "warnings": []}'
   621→        elif "@reviewer" in sys_content:
   622→            content = '{"verdict": "approve", "issues": [], "summary": "Mock review: looks good"}'
   623→        elif "@tester" in sys_content:
   624→            content = '{"test_files": [{"path": "test_mock.py", "content": "def test_mock():\\n    pass"}], "test_results": {"passed": 1, "failed": 0, "skipped": 0, "failures": []}, "summary": "Mock tests pass", "coverage_pct": 100.0}'
   625→        elif "@debugger" in sys_content:
   626→            content = '{"root_cause": "Mock root cause identified", "confidence": 0.95, "fix": {"file": "src/app.py", "line": 42, "original": "broken()", "fixed": "fixed()", "explanation": "Mock fix applied"}, "verification": "Run tests to verify", "alternative_causes": []}'
   627→        else:
   628→            content = '{"result": "mock output"}'
   629→
   630→        tokens_in = sum(len(m.content) // 4 for m in messages)
   631→        tokens_out = len(content) // 4
   632→
   633→        return LLMResponse(
   634→            content=content,
   635→            tool_calls=[],
   636→            usage=LLMUsage(
   637→                tokens_in=tokens_in,
   638→                tokens_out=tokens_out,
   639→                cost_usd=0.0,
   640→                model=model,
   641→                cached=False,
   642→            ),
   643→            model=model,
   644→        )
   645→
   646→    def list_models(self) -> list[str]:
   647→        return ["mock"]
   648→
   649→    async def stream_complete(
   650→        self,
   651→        messages: list[LLMMessage],
   652→        *,
   653→        model: str = "mock",
   654→        tools: list[dict[str, Any]] | None = None,
   655→        temperature: float = 0.0,
   656→        max_tokens: int | None = None,
   657→        response_format: dict[str, Any] | None = None,
   658→        response_schema: dict[str, Any] | None = None,
   659→        **kwargs: Any,
   660→    ) -> AsyncIterator[LLMResponse]:
   661→        """Default streaming: yield the full response in a single chunk.
   662→
   663→        Matches the ``MockLLMProvider.stream_complete`` contract. Real
   664→        token-by-token streaming is available in ``OllamaProvider`` and
   665→        ``LiteLLMProvider``; this mock yields the full response on first
   666→        iteration. AG-UI transport support lands in v0.2.
   667→        """
   668→        response = await self.complete(
   669→            messages,
   670→            model=model,
   671→            tools=tools,
   672→            temperature=temperature,
   673→            max_tokens=max_tokens,
   674→            response_format=response_format,
   675→            response_schema=response_schema,
   676→            **kwargs,
   677→        )
   678→        yield response
