     1→     1→"""
     2→     2→ARNES Playbook Executor — runs a compiled Playbook as a DAG.
     3→     3→
     4→     4→The executor:
     5→     5→1. Walks the playbook steps in order.
     6→     6→2. For each step, invokes the specialist or tool.
     7→     7→3. Applies conditional branches (if/elif/else).
     8→     8→4. Runs parallel branches concurrently.
     9→     9→5. Handles retries with backoff.
    10→    10→6. Pauses at HITL gates.
    11→    11→7. Tracks budget via CostGuard.
    12→    12→8. Appends events to the Thread.
    13→    13→9. Returns a PlaybookRunResult with full trace.
    14→    14→
    15→    15→The executor is async and supports both fire-and-forget and streaming modes.
    16→    16→"""

    58→    58→def _is_docker_available() -> bool:
    59→    59→    """Return True if the ``docker`` CLI is on PATH.
    60→    60→
    61→    61→    Used by the executor to decide whether to wire the Docker sandbox into
    62→    62→    the default ``ToolContext``. This is a presence check only — it does NOT
    63→    63→    verify the daemon is running or that ``arnes-sandbox:latest`` exists.

    65→    65→    execution time (``FileNotFoundError`` on ``docker run``).
    66→    66→
    67→    67→    We deliberately avoid probing the daemon (``docker info`` / ``docker
    68→    68→    version``) here because:
    69→    69→
    70→    70→    1. It spawns a subprocess on every ``PlaybookExecutor`` construction,

    73→    73→       failing fast at construction time would prevent the user from
    74→    74→       running non-shell playbooks that don't need Docker at all.
    75→    75→    3. The ``ShellTool._execute_in_sandbox`` already handles the
    76→    76→       ``FileNotFoundError`` case (docker binary missing) and returns a

   103→   103→class PlaybookExecutor:
   104→   104→    """Executes a compiled Playbook.
   105→   105→
   106→   106→    Usage:
   107→   107→        playbook = PlaybookCompiler.from_file("manuals/audit-pr.yaml")

   118→   118→        tool_registry: ToolRegistry | None = None,

   165→   165→    async def run(
   166→   166→        self,
   167→   167→        playbook: Playbook,
   168→   168→        *,
   169→   169→        initial_input: dict[str, Any] | None = None,
   170→   170→    ) -> PlaybookRunResult:
   171→   171→        """Execute a playbook. Returns a PlaybookRunResult."""
   172→   172→        # Use a mutable list to hold the thread (so step helpers can append)
   173→   173→        thread_holder: list[Thread] = [Thread.create()]
   174→   174→        start_time = time.monotonic()
   175→   175→        outputs: dict[str, Any] = dict(playbook.variables)
   176→   176→        if initial_input:
   177→   177→            outputs.update(initial_input)
   178→   178→
   179→   179→        # Track cost guard at the run level (shared across steps)
   180→   180→        cost_guard = CostGuard(self.provider, budget=self.cost_budget)
   181→   181→
   182→   182→        steps_executed = 0
   183→   183→        steps_failed = 0
   184→   184→        total_tokens_in = 0
   185→   185→        total_tokens_out = 0
   186→   186→        total_cost_usd = 0.0
   187→   187→        aborted = False
   188→   188→        abort_error: str | None = None
   189→   189→
   190→   190→        try:
   191→   191→            for step in playbook.steps:
   192→   192→                # Check if step should be skipped due to prior saltar_a
   193→   193→                skip_until = outputs.get("__skip_steps_until", {})
   194→   194→                if skip_until:
   195→   195→                    # If we've reached the target step, clear the skip marker
   196→   196→                    if step.id in skip_until:
   197→   197→                        del skip_until[step.id]
   198→   198→                        if not skip_until:
   199→   199→                            del outputs["__skip_steps_until"]
   200→   200→                        logger.info("saltar_a_reached", step_id=step.id)
   201→   201→                        # Don't skip this step — execute it
   202→   202→                    else:
   203→   203→                        # Skip this step
   204→   204→                        logger.info("step_skipped", step_id=step.id, reason="saltar_a")
   205→   205→                        continue
   206→   206→
   207→   207→                # Execute the step
   208→   208→                step_result = await self._execute_step(
   209→   209→                    step,
   210→   210→                    thread_holder,
   211→   211→                    outputs,
   212→   212→                    cost_guard,
   213→   213→                    playbook,
   214→   214→                )
   215→   215→
   216→   216→                if step_result["success"]:
   217→   217→                    steps_executed += 1
   218→   218→                    outputs[step.id] = step_result.get("output")
   219→   219→                    # Track usage
   220→   220→                    usage = step_result.get("usage", {})
   221→   221→                    total_tokens_in += usage.get("tokens_in", 0)
   222→   222→                    total_tokens_out += usage.get("tokens_out", 0)
   223→   223→                    total_cost_usd += usage.get("cost_usd", 0.0)
   224→   224→                else:
   225→   225→                    steps_failed += 1
   226→   226→                    # Check if step has si_no_se_cumple fallback
   227→   227→                    if step.if_not_met:
   228→   228→                        branch_result = await self._handle_conditional_branch(
   229→   229→                            step,
   230→   230→                            step.if_not_met,
   231→   231→                            thread_holder,
   232→   232→                            outputs,
   233→   233→                            cost_guard,
   234→   234→                            playbook,
   235→   235→                        )
   236→   236→                        if branch_result.get("terminate"):
   237→   237→                            logger.info(
   238→   238→                                "run_terminated_by_conditional",
   239→   239→                                step_id=step.id,
   240→   240→                                termination=branch_result["terminate"],
   241→   241→                            )
   242→   242→                            break
   243→   243→                    else:
   244→   244→                        # No fallback — abort
   245→   245→                        thread_holder[0] = thread_holder[0].append(
   246→   246→                            RunFailedEvent(
   247→   247→                                thread_id=thread_holder[0].id,
   248→   248→                                step_id=step.id,
   249→   249→                                data={
   250→   250→                                    "error": step_result.get("error", "Unknown error"),
   251→   251→                                    "recoverable": False,
   252→   252→                                },
   253→   253→                            )
   254→   254→                        )
   255→   255→                        aborted = True
   256→   256→                        abort_error = step_result.get("error")
   257→   257→                        break
   258→   258→
   259→   259→            # Run completed (or aborted)
   260→   260→            if not aborted:
   261→   261→                thread_holder[0] = thread_holder[0].append(
   262→   262→                    RunCompletedEvent(
   263→   263→                        thread_id=thread_holder[0].id,
   264→   264→                        data={
   265→   265→                            "steps_executed": steps_executed,
   266→   266→                            "duration_s": time.monotonic() - start_time,
   267→   267→                            "total_tokens": total_tokens_in + total_tokens_out,
   268→   268→                            "total_cost_usd": total_cost_usd,
   269→   269→                        },
   270→   270→                    )
   271→   271→                )
   272→   272→
   273→   273→            return PlaybookRunResult(
   274→   274→                thread=thread_holder[0],
   275→   275→                success=not aborted,
   276→   276→                steps_executed=steps_executed,
   277→   277→                steps_failed=steps_failed,
   278→   278→                duration_s=time.monotonic() - start_time,
   279→   279→                total_tokens_in=total_tokens_in,
   280→   280→                total_tokens_out=total_tokens_out,
   281→   281→                total_cost_usd=total_cost_usd,
   282→   282→                outputs=_filter_internal_keys(outputs),
   283→   283→                error=abort_error,
   284→   284→            )
   285→   285→
   286→   286→        except BudgetExceeded as e:
   287→   287→            logger.error("budget_exceeded", error=str(e), spent=e.spent, budget=e.budget)
   288→   288→            thread_holder[0] = thread_holder[0].append(
   289→   289→                RunFailedEvent(
   290→   290→                    thread_id=thread_holder[0].id,
   291→   291→                    data={
   292→   292→                        "error": f"Budget exceeded: {e}",
   293→   293→                        "spent_usd": e.spent,
   294→   294→                        "budget_usd": e.budget,
   295→   295→                        "level": e.level,
   296→   296→                    },
   297→   297→                )
   298→   298→            )
   299→   299→            return PlaybookRunResult(
   300→   300→                thread=thread_holder[0],
   301→   301→                success=False,
   302→   302→                steps_executed=steps_executed,
   303→   303→                steps_failed=steps_failed,
   304→   304→                duration_s=time.monotonic() - start_time,
   305→   305→                total_tokens_in=total_tokens_in,
   306→   306→                total_tokens_out=total_tokens_out,
   307→   307→                total_cost_usd=total_cost_usd,
   308→   308→                outputs=_filter_internal_keys(outputs),
   309→   309→                error=str(e),
   310→   310→            )

   312→   312→    async def stream(  # noqa: PLR0915 - mirrors run() body; splitting hurts readability
   313→   313→        self,
   314→   314→        playbook: Playbook,
   315→   315→        *,
   316→   316→        initial_input: dict[str, Any] | None = None,
   317→   317→    ) -> AsyncIterator[Event | PlaybookRunResult]:
   318→   318→        """Stream playbook execution, yielding events as each step completes.
   319→   319→
   320→   320→        Yields (in order):
   321→   321→
   322→   322→        - For each step that runs: the ``StepCompletedEvent`` (or
   323→   323→          ``StepFailedEvent``) emitted when that step finishes. The
   324→   324→          event is the *last* event appended to the thread by
   325→   325→          :meth:`_execute_step` — intermediate events
   326→   326→          (``StepStartedEvent``, ``AssistantMessageEvent``,
   327→   327→          ``CostThresholdEvent``, …) stay in the thread and are visible
   328→   328→          in the final ``PlaybookRunResult.thread``.
   329→   329→        - ``RunCompletedEvent`` or ``RunFailedEvent`` at the end of the run
   330→   330→          (also appended to the thread).
   331→   331→        - Final yield: a :class:`PlaybookRunResult` with the full thread
   332→   332→          and aggregate accounting (steps_executed, tokens, cost).
   333→   333→
   334→   334→        Best-effort streaming contract:
   335→   335→
   336→   336→        - Steps are still executed sequentially in definition order
   337→   337→          (the ``run()`` semantics are preserved). What streaming gives
   338→   338→          you is the ability to surface each step's completion event
   339→   339→          to the caller *immediately* — without waiting for the whole
   340→   340→          playbook to finish.
   341→   341→        - Parallel branches stream in **completion order**, not
   342→   342→          definition order: when a parallel step finishes, the
   343→   343→          ``PARALLEL_BRANCH_COMPLETED`` event is yielded, but the
   344→   344→          per-sub-step ``StepCompletedEvent`` events were appended
   345→   345→          inside ``asyncio.gather`` and arrive in the merged-event
   346→   346→          timestamp order (see :meth:`_execute_parallel`).
   347→   347→        - The final ``PlaybookRunResult`` is always the last yield,
   348→   348→          even on failure / budget-exceeded.
   349→   349→
   350→   350→        Per-token streaming of LLM responses within a step is available
   351→   351→        via :meth:`arnes.specialists.base.Specialist.stream`; this
   352→   352→        executor-level stream yields step-level events, not token-level
   353→   353→        events. The two can be composed: a UI that wants both
   354→   354→        token-by-token rendering AND a final bitácora can consume
   355→   355→        ``Specialist.stream()`` for the rendering and rely on the
   356→   356→        ``AssistantMessageEvent`` emitted into the thread (by both
   357→   357→        ``Specialist.run()`` and ``Specialist.stream()``) for the
   358→   358→        audit trail.
   359→   359→
   360→   360→        Usage::
   361→   361→
   362→   362→            executor = PlaybookExecutor(provider=p)
   363→   363→            async for event in executor.stream(playbook):
   364→   364→                if isinstance(event, PlaybookRunResult):
   365→   365→                    print(event.to_markdown())
   366→   366→                else:
   367→   367→                    print(f"event: {event.type.value}")
   368→   368→        """
   369→   369→        thread_holder: list[Thread] = [Thread.create()]
   370→   370→        start_time = time.monotonic()
   371→   371→        outputs: dict[str, Any] = dict(playbook.variables)
   372→   372→        if initial_input:
   373→   373→            outputs.update(initial_input)
   374→   374→
   375→   375→        cost_guard = CostGuard(self.provider, budget=self.cost_budget)
   376→   376→
   377→   377→        steps_executed = 0
   378→   378→        steps_failed = 0
   379→   379→        total_tokens_in = 0
   380→   380→        total_tokens_out = 0
   381→   381→        total_cost_usd = 0.0
   382→   382→        aborted = False
   383→   383→        abort_error: str | None = None
   384→   384→
   385→   385→        try:
   386→   386→            for step in playbook.steps:
   387→   387→                # Mirror run()'s skip-until logic verbatim.
   388→   388→                skip_until = outputs.get("__skip_steps_until", {})
   389→   389→                if skip_until:
   390→   390→                    if step.id in skip_until:
   391→   391→                        del skip_until[step.id]
   392→   392→                        if not skip_until:
   393→   393→                            del outputs["__skip_steps_until"]
   394→   394→                        logger.info("saltar_a_reached", step_id=step.id)
   395→   395→                    else:
   396→   396→                        logger.info("step_skipped", step_id=step.id, reason="saltar_a")
   397→   397→                        continue
   398→   398→
   399→   399→                step_result = await self._execute_step(
   400→   400→                    step,
   401→   401→                    thread_holder,
   402→   402→                    outputs,
   403→   403→                    cost_guard,
   404→   404→                    playbook,
   405→   405→                )
   406→   406→
   407→   407→                # Yield the last event appended by _execute_step. This is
   408→   408→                # the StepCompletedEvent or StepFailedEvent — exactly the
   409→   409→                # transition a streaming consumer cares about. (Earlier
   410→   410→                # events like StepStartedEvent remain in the thread for
   411→   411→                # the final PlaybookRunResult.)
   412→   412→                last_event = thread_holder[0].last()
   413→   413→                if last_event is not None:
   414→   414→                    yield last_event
   415→   415→
   416→   416→                if step_result["success"]:
   417→   417→                    steps_executed += 1
   418→   418→                    outputs[step.id] = step_result.get("output")
   419→   419→                    usage = step_result.get("usage", {})
   420→   420→                    total_tokens_in += usage.get("tokens_in", 0)
   421→   421→                    total_tokens_out += usage.get("tokens_out", 0)
   422→   422→                    total_cost_usd += usage.get("cost_usd", 0.0)
   423→   423→                else:
   424→   424→                    steps_failed += 1
   425→   425→                    if step.if_not_met:
   426→   426→                        branch_result = await self._handle_conditional_branch(
   427→   427→                            step,
   428→   428→                            step.if_not_met,
   429→   429→                            thread_holder,
   430→   430→                            outputs,
   431→   431→                            cost_guard,
   432→   432→                            playbook,
   433→   433→                        )
   434→   434→                        if branch_result.get("terminate"):
   435→   435→                            logger.info(
   436→   436→                                "run_terminated_by_conditional",
   437→   437→                                step_id=step.id,
   438→   438→                                termination=branch_result["terminate"],
   439→   439→                            )
   440→   440→                            break
   441→   441→                    else:
   442→   442→                        run_failed = RunFailedEvent(
   443→   443→                            thread_id=thread_holder[0].id,
   444→   444→                            step_id=step.id,
   445→   445→                            data={
   446→   446→                                "error": step_result.get("error", "Unknown error"),
   447→   447→                                "recoverable": False,
   448→   448→                            },
   449→   449→                        )
   450→   450→                        thread_holder[0] = thread_holder[0].append(run_failed)
   451→   451→                        aborted = True
   452→   452→                        abort_error = step_result.get("error")
   453→   453→                        # Yield the RunFailedEvent so streaming consumers
   454→   454→                        # see the abort transition immediately (mirrors
   455→   455→                        # the success path which yields RunCompletedEvent).
   456→   456→                        yield run_failed
   457→   457→                        break
   458→   458→
   459→   459→            if not aborted:
   460→   460→                run_completed = RunCompletedEvent(
   461→   461→                    thread_id=thread_holder[0].id,
   462→   462→                    data={
   463→   463→                        "steps_executed": steps_executed,
   464→   464→                        "duration_s": time.monotonic() - start_time,
   465→   465→                        "total_tokens": total_tokens_in + total_tokens_out,
   466→   466→                        "total_cost_usd": total_cost_usd,
   467→   467→                    },
   468→   468→                )
   469→   469→                thread_holder[0] = thread_holder[0].append(run_completed)
   470→   470→                yield run_completed
   471→   471→
   472→   472→            yield self._build_run_result(
   473→   473→                thread_holder[0],
   474→   474→                success=not aborted,
   475→   475→                steps_executed=steps_executed,
   476→   476→                steps_failed=steps_failed,
   477→   477→                start_time=start_time,
   478→   478→                total_tokens_in=total_tokens_in,
   479→   479→                total_tokens_out=total_tokens_out,
   480→   480→                total_cost_usd=total_cost_usd,
   481→   481→                outputs=outputs,
   482→   482→                error=abort_error,
   483→   483→            )
   484→   484→        except BudgetExceeded as e:
   485→   485→            logger.error("budget_exceeded", error=str(e), spent=e.spent, budget=e.budget)
   486→   486→            run_failed = RunFailedEvent(
   487→   487→                thread_id=thread_holder[0].id,
   488→   488→                data={
   489→   489→                    "error": f"Budget exceeded: {e}",
   490→   490→                    "spent_usd": e.spent,
   491→   491→                    "budget_usd": e.budget,
   492→   492→                    "level": e.level,
   493→   493→                },
   494→   494→            )
   495→   495→            thread_holder[0] = thread_holder[0].append(run_failed)
   496→   496→            yield run_failed
   497→   497→            yield self._build_run_result(
   498→   498→                thread_holder[0],
   499→   499→                success=False,
   500→   500→                steps_executed=steps_executed,
   501→   501→                steps_failed=steps_failed,
   502→   502→                start_time=start_time,
   503→   503→                total_tokens_in=total_tokens_in,
   504→   504→                total_tokens_out=total_tokens_out,
   505→   505→                total_cost_usd=total_cost_usd,
   506→   506→                outputs=outputs,
   507→   507→                error=str(e),
   508→   508→            )

   546→   546→
   547→   547→    async def _execute_step(
   548→   548→        self,
   549→   549→        step: PlaybookStep,
   550→   550→        thread_holder: list[Thread],
   551→   551→        outputs: dict[str, Any],
   552→   552→        cost_guard: CostGuard,
   553→   553→        playbook: Playbook,
   554→   554→    ) -> dict[str, Any]:
   555→   555→        """Execute a single step (specialist, tool, or parallel branch)."""
   556→   556→        thread_holder[0] = thread_holder[0].append(
   557→   557→            StepStartedEvent(
   558→   558→                thread_id=thread_holder[0].id,
   559→   559→                step_id=step.id,
   560→   560→                specialist=step.specialist or step.tool,
   561→   561→                data={"step_id": step.id, "specialist": step.specialist or step.tool},
   562→   562→            )
   563→   563→        )
   564→   564→
   565→   565→        step_start = time.monotonic()
   566→   566→
   567→   567→        try:
   568→   568→            # Parallel branch
   569→   569→            if step.parallel:
   570→   570→                result = await self._execute_parallel(
   571→   571→                    step, thread_holder, outputs, cost_guard, playbook
   572→   572→                )
   573→   573→            # Specialist invocation
   574→   574→            elif step.specialist:
   575→   575→                result = await self._execute_specialist(
   576→   576→                    step, thread_holder, outputs, cost_guard, playbook
   577→   577→                )
   578→   578→            # Tool invocation
   579→   579→            elif step.tool:
   580→   580→                result = await self._execute_tool(step, thread_holder, outputs, playbook)
   581→   581→            else:
   582→   582→                raise ValueError(f"Step '{step.id}' has no action defined")
   583→   583→
   584→   584→            # Drain middleware event sink BEFORE recording completion so
   585→   585→            # that AssistantMessageEvent / CostThresholdEvent / CACHE_HIT /
   586→   586→            # REFUSAL_TRIGGERED events emitted during the step appear in
   587→   587→            # the thread before the StepCompletedEvent. The middleware
   588→   588→            # creates events with a nil thread_id placeholder (it does not
   589→   589→            # have access to the Thread); we patch the real thread_id and
   590→   590→            # step_id here.
   591→   591→            self._drain_middleware_events(thread_holder, cost_guard, step.id)
   592→   592→
   593→   593→            # Pull token / cost usage out of the step result so the
   594→   594→            # StepCompletedEvent carries the per-step aggregate. The
   595→   595→            # specialist returns ``usage`` as a model_dump() of LLMUsage
   596→   596→            # (tokens_in, tokens_out, cost_usd, model, cached).
   597→   597→            usage = result.get("usage") or {}
   598→   598→
   599→   599→            # Record completion (with token + cost accounting)
   600→   600→            thread_holder[0] = thread_holder[0].append(
   601→   601→                StepCompletedEvent(
   602→   602→                    thread_id=thread_holder[0].id,
   603→   603→                    step_id=step.id,
   604→   604→                    specialist=step.specialist or step.tool,
   605→   605→                    data={
   606→   606→                        "step_id": step.id,
   607→   607→                        "output": result.get("output"),
   608→   608→                        "duration_s": time.monotonic() - step_start,
   609→   609→                        "tokens_in": usage.get("tokens_in", 0),
   610→   610→                        "tokens_out": usage.get("tokens_out", 0),
   611→   611→                        "cost_usd": usage.get("cost_usd", 0.0),
   612→   612→                    },
   613→   613→                )
   614→   614→            )
   615→   615→
   616→   616→            return result
   617→   617→
   618→   618→        except Exception as e:
   619→   619→            logger.exception("step_failed", step_id=step.id, error=str(e))
   620→   620→            # Drain any events emitted before the failure too — they are
   621→   621→            # still useful for debugging (e.g. a CostThresholdEvent that
   622→   622→            # fired right before the crash, or an AssistantMessageEvent
   623→   623→            # from the failed LLM call).
   624→   624→            self._drain_middleware_events(thread_holder, cost_guard, step.id)
   625→   625→            thread_holder[0] = thread_holder[0].append(
   626→   626→                StepFailedEvent(
   627→   627→                    thread_id=thread_holder[0].id,
   628→   628→                    step_id=step.id,
   629→   629→                    specialist=step.specialist or step.tool,
   630→   630→                    data={"step_id": step.id, "error": str(e), "retry": False},
   631→   631→                )
   632→   632→            )
   633→   633→            return {"success": False, "error": str(e)}
   634→   634→
   635→   635→    def _drain_middleware_events(
   636→   636→        self,
   637→   637→        thread_holder: list[Thread],
   638→   638→        cost_guard: CostGuard,
   639→   639→        step_id: str,
   640→   640→    ) -> None:
   641→   641→        """Drain the middleware event sink and append events to the Thread.
   642→   642→
   643→   643→        Middleware (CostGuard, TokenOptimizer, VerificationLayer) emit
   644→   644→        events to a shared ``_events`` list because they do not have direct
   645→   645→        access to the Thread. The events are created with a nil thread_id
   646→   646→        placeholder; here we patch the real thread_id and step_id and
   647→   647→        append them to the Thread.
   648→   648→
   649→   649→        Idempotent: clears the sink after draining so the same events are
   650→   650→        not appended twice.
   651→   651→        """
   652→   652→        events = getattr(cost_guard, "_events", None)
   653→   653→        if not events:
   654→   654→            return
   655→   655→
   656→   656→        thread_id = thread_holder[0].id
   657→   657→        for event in events:
   658→   658→            # Events are frozen pydantic models; use model_copy(update=...)
   659→   659→            # to set the real thread_id and step_id without mutating the
   660→   660→            # original (which may be referenced by middleware state).
   661→   661→            patched = event.model_copy(
   662→   662→                update={"thread_id": thread_id, "step_id": event.step_id or step_id}
   663→   663→            )
   664→   664→            thread_holder[0] = thread_holder[0].append(patched)
   665→   665→
   666→   666→        events.clear()
   667→   667→
   668→   668→    async def _execute_specialist(
   669→   669→        self,
   670→   670→        step: PlaybookStep,
   671→   671→        thread_holder: list[Thread],
   672→   672→        outputs: dict[str, Any],
   673→   673→        cost_guard: CostGuard,
   674→   674→        playbook: Playbook,
   675→   675→    ) -> dict[str, Any]:
   676→   676→        """Invoke a specialist."""
   677→   677→        specialist = self.specialist_registry.get(step.specialist or "")
   678→   678→        if not specialist:
   679→   679→            return {
   680→   680→                "success": False,
   681→   681→                "error": f"Specialist '{step.specialist}' not registered. Available: {self.specialist_registry.list_names()}",
   682→   682→            }
   683→   683→
   684→   684→        # Resolve input (may contain Jinja2-style template refs)
   685→   685→        input_data = self._resolve_input(step.input, outputs)
   686→   686→
   687→   687→        # Build tool context — sandbox state is detected once at executor
   688→   688→        # construction time (see __init__) and propagated to every
   689→   689→        # specialist invocation so the ShellTool can pick the right
   690→   690→        # execution path (Docker sandbox vs. gated local execution).
   691→   691→        ctx = ToolContext(
   692→   692→            thread_id=thread_holder[0].id,
   693→   693→            step_id=step.id,
   694→   694→            specialist=step.specialist,
   695→   695→            working_dir=".",
   696→   696→            sandbox_enabled=self._sandbox_enabled,
   697→   697→            sandbox_container=self._sandbox_container,
   698→   698→            budget_remaining_usd=(
   699→   699→                (cost_guard.budget.effective_budget() or 0) - cost_guard.spent_usd
   700→   700→            ),
   701→   701→            metadata={"interactive": self.interactive},
   702→   702→        )
   703→   703→
   704→   704→        # Use the cost_guard-wrapped provider
   705→   705→        result = await specialist.run(
   706→   706→            input_data,
   707→   707→            ctx,
   708→   708→            provider=cost_guard,
   709→   709→            tool_registry=self.tool_registry,
   710→   710→        )
   711→   711→
   712→   712→        success = result.get("success", False)
   713→   713→        if not success and result.get("budget_exceeded"):
   714→   714→            raise BudgetExceeded(
   715→   715→                f"Budget exceeded during specialist '{step.specialist}' invocation",
   716→   716→                spent=cost_guard.spent_usd,
   717→   717→                budget=cost_guard.budget.effective_budget() or 0.0,
   718→   718→                level="specialist",
   719→   719→            )
   720→   720→
   721→   721→        # Propagate error message for non-budget failures
   722→   722→        if not success and result.get("error"):
   723→   723→            return {
   724→   724→                "success": False,
   725→   725→                "error": result["error"],
   726→   726→                "output": None,
   727→   727→                "usage": result.get("usage", {}),
   728→   728→            }
   729→   729→
   730→   730→        return {
   731→   731→            "success": success,
   732→   732→            "output": result.get("output"),
   733→   733→            "usage": result.get("usage", {}),
   734→   734→        }
   735→   735→
   736→   736→    async def _execute_tool(
   737→   737→        self,
   738→   738→        step: PlaybookStep,
   739→   739→        thread_holder: list[Thread],
   740→   740→        outputs: dict[str, Any],
   741→   741→        playbook: Playbook,
   742→   742→    ) -> dict[str, Any]:
   743→   743→        """Invoke a tool."""
   744→   744→        tool = self.tool_registry.get(step.tool or "")
   745→   745→        if not tool:
   746→   746→            return {
   747→   747→                "success": False,
   748→   748→                "error": f"Tool '{step.tool}' not registered. Available: {self.tool_registry.list_names()}",
   749→   749→            }
   750→   750→
   751→   751→        input_data = self._resolve_input(step.input, outputs)
   752→   752→
   753→   753→        ctx = ToolContext(
   754→   754→            thread_id=thread_holder[0].id,
   755→   755→            step_id=step.id,
   756→   756→            metadata={"interactive": self.interactive},
   757→   757→        )
   758→   758→
   759→   759→        result = await tool.execute(input_data, ctx)
   760→   760→        return {
   761→   761→            "success": result.success,
   762→   762→            "output": result.output if result.success else None,
   763→   763→            "error": result.error,
   764→   764→        }

   766→   766→    async def _execute_parallel(
   767→   767→        self,
   768→   768→        step: PlaybookStep,
   769→   769→        thread_holder: list[Thread],
   770→   770→        outputs: dict[str, Any],
   771→   771→        cost_guard: CostGuard,
   772→   772→        playbook: Playbook,
   773→   773→    ) -> dict[str, Any]:
   774→   774→        """Execute parallel sub-steps concurrently with ``asyncio.gather``.
   775→   775→
   776→   776→        Each sub-step gets its OWN thread_holder (a copy of the parent thread
   777→   777→        at this point) so appends are isolated — no race on the shared
   778→   778→        ``thread_holder[0]`` reference. After all sub-steps complete, their
   779→   779→        event deltas are merged back into the parent thread_holder in
   780→   780→        timestamp order. This preserves the audit-log pattern while
   781→   781→        enabling true parallelism (the previous implementation ran sub-steps
   782→   782→        sequentially in a for-loop, which was correct but not concurrent).
   783→   783→
   784→   784→        ``Thread.append`` mutates in place (O(1) per append, replacing the
   785→   785→        old O(N²) ``[*self.events, event]`` rebuild). Because the sub-step
   786→   786→        coroutines mutate their own Thread objects, we MUST give each one
   787→   787→        an isolated copy (with its own ``events`` list reference) — sharing
   788→   788→        the parent reference would let every sub-step clobber the others'
   789→   789→        appends. The copy is shallow on the list (a new list of the same
   790→   790→        Event references); Events themselves are frozen pydantic models so
   791→   791→        sharing the references is safe.
   792→   792→
   793→   793→        The shared ``cost_guard._events`` sink is drained by each sub-step's
   794→   794→        own ``_execute_step`` call; because ``_drain_middleware_events`` is
   795→   795→        synchronous, drains run atomically in the single-threaded asyncio
   796→   796→        loop. An event emitted by sub-step B's specialist may end up drained
   797→   797→        into sub-step A's thread_holder, but each event carries its own
   798→   798→        ``step_id`` (set by ``_emit_assistant_message`` from the
   799→   799→        ``ToolContext``) so the merged audit log is still correctly
   800→   800→        attributed — the delta is just a container for the merge, not an
   801→   801→        authoritative attribution.
   802→   802→
   803→   803→        Emits ``PARALLEL_BRANCH_STARTED`` before ``asyncio.gather`` and
   804→   804→        ``PARALLEL_BRANCH_COMPLETED`` after the merge so the audit log
   805→   805→        marks the parallel block boundaries (previously these event types
   806→   806→        were defined but never instantiated).
   807→   807→        """
   808→   808→        if not step.parallel:
   809→   809→            return {"success": False, "error": "No parallel steps defined"}
   810→   810→
   811→   811→        # Snapshot the parent thread so each sub-step's delta is exactly
   812→   812→        # the events it appends beyond this point. The parent thread already
   813→   813→        # has the outer StepStartedEvent(parallel) appended by _execute_step.
   814→   814→        parent_snapshot = thread_holder[0]
   815→   815→        parent_event_count = len(parent_snapshot.events)
   816→   816→
   817→   817→        # Emit PARALLEL_BRANCH_STARTED so the audit log records the
   818→   818→        # parallel-block boundary. ``step_id`` is the outer parallel step's
   819→   819→        # id; sub-step ids live inside the ``sub_steps`` payload.
   820→   820→        thread_holder[0] = thread_holder[0].append(
   821→   821→            Event(
   822→   822→                type=EventType.PARALLEL_BRANCH_STARTED,
   823→   823→                thread_id=parent_snapshot.id,
   824→   824→                step_id=step.id,
   825→   825→                data={
   826→   826→                    "step_id": step.id,
   827→   827→                    "sub_step_ids": [s.id for s in step.parallel],
   828→   828→                    "sub_step_count": len(step.parallel),
   829→   829→                },
   830→   830→            )
   831→   831→        )
   832→   832→        # Re-snapshot so the STARTED event is part of every sub-step's
   833→   833→        # parent_event_count baseline (it belongs to the parent, not to
   834→   834→        # any individual sub-step's delta).
   835→   835→        parent_snapshot = thread_holder[0]
   836→   836→        parent_event_count = len(parent_snapshot.events)
   837→   837→
   838→   838→        # Each sub-step gets its OWN Thread copy with a fresh events list.
   839→   839→        # ``Thread.append`` mutates in place, so without this copy every
   840→   840→        # sub-step would share the same parent.events list and clobber
   841→   841→        # each other. The new list contains the same Event references
   842→   842→        # (Events are frozen, so sharing is safe).
   843→   843→        sub_holders: list[list[Thread]] = [
   844→   844→            [Thread(id=parent_snapshot.id, events=list(parent_snapshot.events))]
   845→   845→            for _ in step.parallel
   846→   846→        ]
   847→   847→
   848→   848→        # Run all sub-steps concurrently. return_exceptions=True so a single
   849→   849→        # failure doesn't cancel the others — every sub-step runs to
   850→   850→        # completion (or to its own failure) and we merge everything.
   851→   851→        coros = [
   852→   852→            self._execute_step(
   853→   853→                sub_step,
   854→   854→                sub_holders[i],
   855→   855→                outputs,
   856→   856→                cost_guard,
   857→   857→                playbook,
   858→   858→            )
   859→   859→            for i, sub_step in enumerate(step.parallel)
   860→   860→        ]
   861→   861→        raw_results = await asyncio.gather(*coros, return_exceptions=True)
   862→   862→
   863→   863→        # Merge sub-step deltas back into the parent thread_holder in
   864→   864→        # timestamp order. Each delta is the events the sub-step appended
   865→   865→        # beyond the shared snapshot (StepStarted, AssistantMessage(s),
   866→   866→        # StepCompleted/Failed for the sub-step, plus any middleware events
   867→   867→        # it drained from the shared cost_guard sink).
   868→   868→        merged_events: list[Event] = []
   869→   869→        for holder in sub_holders:
   870→   870→            merged_events.extend(holder[0].events[parent_event_count:])
   871→   871→        # Stable sort by timestamp preserves intra-sub-step order for events
   872→   872→        # with identical timestamps (the natural audit order within a single
   873→   873→        # sub-step's lifecycle: Started → AssistantMessage → Completed).
   874→   874→        merged_events.sort(key=lambda e: e.timestamp)
   875→   875→
   876→   876→        thread_holder[0] = thread_holder[0].extend(merged_events)
   877→   877→
   878→   878→        # Emit PARALLEL_BRANCH_COMPLETED with the per-sub-step outcome so
   879→   879→        # the audit log marks the parallel-block end and records which
   880→   880→        # branches succeeded / failed.
   881→   881→        sub_step_outcomes: list[dict[str, Any]] = []
   882→   882→        all_success = True
   883→   883→        first_error: str | None = None
   884→   884→        outputs_map: dict[str, dict[str, Any]] = {}
   885→   885→
   886→   886→        for sub_step, raw in zip(step.parallel, raw_results, strict=True):
   887→   887→            if isinstance(raw, BaseException):
   888→   888→                # A sub-step coroutine raised (shouldn't happen — _execute_step
   889→   889→                # catches exceptions — but defend against executor bugs).
   890→   890→                outputs_map[sub_step.id] = {
   891→   891→                    "output": None,
   892→   892→                    "success": False,
   893→   893→                    "error": str(raw),
   894→   894→                }
   895→   895→                sub_step_outcomes.append(
   896→   896→                    {"sub_step_id": sub_step.id, "success": False, "error": str(raw)}
   897→   897→                )
   898→   898→                all_success = False
   899→   899→                if first_error is None:
   900→   900→                    first_error = str(raw)
   901→   901→            else:
   902→   902→                outputs_map[sub_step.id] = {
   903→   903→                    "output": raw.get("output"),
   904→   904→                    "success": raw.get("success", False),
   905→   905→                    "error": raw.get("error"),
   906→   906→                }
   907→   907→                sub_step_outcomes.append(
   908→   908→                    {
   909→   909→                        "sub_step_id": sub_step.id,
   910→   910→                        "success": raw.get("success", False),
   911→   911→                        "error": raw.get("error"),
   912→   912→                    }
   913→   913→                )
   914→   914→                if not raw.get("success", False):
   915→   915→                    all_success = False
   916→   916→                    if first_error is None:
   917→   917→                        first_error = raw.get("error")
   918→   918→
   919→   919→        thread_holder[0] = thread_holder[0].append(
   920→   920→            Event(
   921→   921→                type=EventType.PARALLEL_BRANCH_COMPLETED,
   922→   922→                thread_id=thread_holder[0].id,
   923→   923→                step_id=step.id,
   924→   924→                data={
   925→   925→                    "step_id": step.id,
   926→   926→                    "all_success": all_success,
   927→   927→                    "sub_step_outcomes": sub_step_outcomes,
   928→   928→                    "merged_event_count": len(merged_events),
   929→   929→                },
   930→   930→            )
   931→   931→        )
   932→   932→
   933→   933→        result: dict[str, Any] = {
   934→   934→            "success": all_success,
   935→   935→            "output": outputs_map,
   936→   936→        }
   937→   937→        if first_error is not None:
   938→   938→            result["error"] = first_error
   939→   939→        return result

   941→   941→    async def _handle_conditional_branch(
   942→   942→        self,
   943→   943→        step: PlaybookStep,
   944→   944→        branch: ConditionalBranch,
   945→   945→        thread_holder: list[Thread],
   946→   946→        outputs: dict[str, Any],
   947→   947→        cost_guard: CostGuard,
   948→   948→        playbook: Playbook,
   949→   949→    ) -> dict[str, Any]:
   950→   950→        """Handle an if_not_met or conditional branch."""
   951→   951→        thread_holder[0] = thread_holder[0].append(
   952→   952→            ConditionalBranchEvent(
   953→   953→                thread_id=thread_holder[0].id,
   954→   954→                step_id=step.id,
   955→   955→                data={
   956→   956→                    "condition": branch.when or "if_not_met",
   957→   957→                    "action": branch.action,
   958→   958→                },
   959→   959→            )
   960→   960→        )
   961→   961→
   962→   962→        if branch.action == "call" and branch.specialist:
   963→   963→            # Invoke the fallback specialist
   964→   964→            fallback_step = PlaybookStep(
   965→   965→                id=f"{step.id}__fallback",
   966→   966→                specialist=branch.specialist,
   967→   967→                input=branch.input or {},
   968→   968→            )
   969→   969→            result = await self._execute_specialist(
   970→   970→                fallback_step, thread_holder, outputs, cost_guard, playbook
   971→   971→            )
   972→   972→            outputs[fallback_step.id] = result.get("output")
   973→   973→            return {"terminate": None, "result": result}
   974→   974→
   975→   975→        if branch.action == "terminate":
   976→   976→            return {"terminate": branch.terminate}
   977→   977→
   978→   978→        if branch.action == "skip" and branch.skip_to:
   979→   979→            # skip_to means "jump TO this step" — skip all steps between
   980→   980→            # current and target
   981→   981→            skip_set = outputs.setdefault("__skip_steps_until", {})
   982→   982→            skip_set[branch.skip_to] = True  # Will be cleared when we reach it
   983→   983→            return {"terminate": None}
   984→   984→
   985→   985→        return {"terminate": None}

   991→   991→    _TEMPLATE_RE = re.compile(r"\{\{\s*([^}]+)\s*\}\}")

   993→   993→    def _resolve_input(
   994→   994→        self,
   995→   995→        input_value: dict[str, Any] | str | None,
   996→   996→        outputs: dict[str, Any],
   997→   997→    ) -> dict[str, Any]:
   998→   998→        """Resolve Jinja2-style template references in input.
   999→   999→
  1000→  1000→        Example: "{{ pasos.leer_diff.salida }}" → outputs["leer_diff"]["output"]
  1001→  1001→        Handles MULTIPLE templates in the same string.
  1002→  1002→        """
  1003→  1003→        if input_value is None:
  1004→  1004→            return {}
  1005→  1005→
  1006→  1006→        if isinstance(input_value, str):
  1007→  1007→            return {"__resolved_str__": self._resolve_template(input_value, outputs)}
  1008→  1008→
  1009→  1009→        if isinstance(input_value, dict):
  1010→  1010→            resolved = {}
  1011→  1011→            for k, v in input_value.items():
  1012→  1012→                if isinstance(v, str):
  1013→  1013→                    resolved[k] = self._resolve_template(v, outputs)
  1014→  1014→                elif isinstance(v, dict):
  1015→  1015→                    resolved[k] = self._resolve_input(v, outputs)
  1016→  1016→                elif isinstance(v, list):
  1017→  1017→                    resolved[k] = [
  1018→  1018→                        self._resolve_template(item, outputs)
  1019→  1019→                        if isinstance(item, str)
  1020→  1020→                        else self._resolve_input(item, outputs)
  1021→  1021→                        if isinstance(item, dict)
  1022→  1022→                        else item
  1023→  1023→                        for item in v
  1024→  1024→                    ]
  1025→  1025→                else:
  1026→  1026→                    resolved[k] = v
  1027→  1027→            return resolved
  1028→  1028→
  1029→  1029→        # Unreachable: ``input_value`` is typed as ``dict[str, Any] | str | None``
  1030→  1030→        # and all three branches above return early. Kept as a defensive
  1031→  1031→        # fallback for callers that bypass the type system (defence-in-depth).
  1032→  1032→        return {"__input__": input_value}  # type: ignore[unreachable]
  1033→  1033→
  1034→  1034→    def _resolve_template(self, template: str, outputs: dict[str, Any]) -> Any:
  1035→  1035→        """Resolve a template string, handling MULTIPLE {{ }} references.
  1036→  1036→
  1037→  1037→        Examples:
  1038→  1038→            "{{ pasos.X.salida }}" → outputs["X"]["output"]
  1039→  1039→            "Plan: {{ variables.nombre }} for PR {{ variables.pr_number }}" → "Plan: foo for PR 1234"
  1040→  1040→            "Diff: {{ pasos.leer_diff.salida }}, Sec: {{ pasos.auditoria.salida }}" → "Diff: ..., Sec: ..."
  1041→  1041→            "{{ }}" → "{{ }}"  (empty template body — returned as literal)
  1042→  1042→        """
  1043→  1043→        # Find ALL template references
  1044→  1044→        matches = list(self._TEMPLATE_RE.finditer(template))
  1045→  1045→
  1046→  1046→        if not matches:
  1047→  1047→            return template
  1048→  1048→
  1049→  1049→        # If the entire string is ONE template, return the resolved value (preserve type)
  1050→  1050→        if len(matches) == 1 and matches[0].group(0) == template:
  1051→  1051→            expr = matches[0].group(1).strip()
  1052→  1052→            if not expr:
  1053→  1053→                # Empty template body (e.g. "{{ }}") — return the original
  1054→  1054→                # literal verbatim rather than re-rendering it with different
  1055→  1055→                # whitespace.
  1056→  1056→                return template
  1057→  1057→            return self._resolve_expr(expr, outputs)
  1058→  1058→
  1059→  1059→        # Otherwise, interpolate ALL matches into the string
  1060→  1060→        result = template
  1061→  1061→        # Process in reverse order to keep indexes valid
  1062→  1062→        for match in reversed(matches):
  1063→  1063→            expr = match.group(1).strip()
  1064→  1064→            if not expr:
  1065→  1065→                # Empty template body — leave the original match untouched
  1066→  1066→                continue
  1067→  1067→            resolved = self._resolve_expr(expr, outputs)
  1068→  1068→            result = result[: match.start()] + str(resolved) + result[match.end() :]
  1069→  1069→
  1070→  1070→        return result

  1072→  1072→    def _resolve_expr(self, expr: str, outputs: dict[str, Any]) -> Any:
  1073→  1073→        """Resolve a single template expression like 'steps.X.output'.
  1074→  1074→
  1075→  1075→        Supported forms:
  1076→  1076→            "steps.X.output"        → outputs["X"]["output"] (or outputs["X"] if raw)
  1077→  1077→            "steps.X.output.field"  → outputs["X"]["output"]["field"] (or outputs["X"]["field"])
  1078→  1078→            "variables.X"           → outputs["X"]
  1079→  1079→            "pasos.X.salida"        → legacy ES form of "steps.X.output"
  1080→  1080→
  1081→  1081→        Two important behaviors:
  1082→  1082→
  1083→  1083→        1. Only the LEADING prefix ("steps.", "variables.", "pasos.") is
  1084→  1084→           stripped. Interior occurrences of these substrings (e.g. when a
  1085→  1085→           step's output literally contains a key named "steps") are
  1086→  1086→           preserved. This makes deep nesting like
  1087→  1087→           ``{{ steps.s1.output.steps.s2.output }}`` work correctly.
  1088→  1088→
  1089→  1089→        2. For STEP references (``steps.*`` / legacy ``pasos.*``), the
  1090→  1090→           "output" (and legacy "salida") segment is treated as a VIRTUAL
  1091→  1091→           accessor: if the current dict has an "output" key, it is
  1092→  1092→           dereferenced; if not, the segment is skipped (the current dict
  1093→  1093→           IS the output). This makes ``{{ steps.X.output }}`` work whether
  1094→  1094→           the step's output was stored raw (``outputs["X"] = output_dict``)
  1095→  1095→           or wrapped (``outputs["X"] = {"output": output_dict, ...}``).
  1096→  1096→
  1097→  1097→           The virtual accessor does NOT apply to ``variables.*`` refs —
  1098→  1098→           variables are user-defined and a missing key is a real error.
  1099→  1099→        """
  1100→  1100→        # Strip leading prefix only — NOT all occurrences. Stripping all
  1101→  1101→        # occurrences of "steps." would corrupt paths whose intermediate
  1102→  1102→        # dicts literally contain a key named "steps" (deep nesting).
  1103→  1103→        stripped = expr.strip()
  1104→  1104→        is_step_ref = False
  1105→  1105→        for prefix in ("steps.", "pasos.", "variables."):
  1106→  1106→            if stripped.startswith(prefix):
  1107→  1107→                stripped = stripped[len(prefix) :]
  1108→  1108→                # Only steps/pasos get the virtual "output" accessor;
  1109→  1109→                # variables use strict key lookup.
  1110→  1110→                is_step_ref = prefix in ("steps.", "pasos.")
  1111→  1111→                break
  1112→  1112→
  1113→  1113→        parts = stripped.split(".")
  1114→  1114→
  1115→  1115→        current: Any = outputs
  1116→  1116→        for raw_part in parts:
  1117→  1117→            # Legacy ES translation per-segment
  1118→  1118→            part = "output" if raw_part == "salida" else raw_part
  1119→  1119→            if isinstance(current, dict):
  1120→  1120→                if part in current:
  1121→  1121→                    current = current[part]
  1122→  1122→                elif is_step_ref and part == "output":
  1123→  1123→                    # Virtual accessor: the current dict is itself the
  1124→  1124→                    # step's output (stored raw). Skip this segment so
  1125→  1125→                    # downstream segments (e.g. ".verdict") resolve against
  1126→  1126→                    # the output dict directly.
  1127→  1127→                    continue
  1128→  1128→                else:
  1129→  1129→                    return f"{{{{ {expr} }}}}"  # Leave template as-is if not found
  1130→  1130→            else:
  1131→  1131→                return f"{{{{ {expr} }}}}"
  1132→  1132→
  1133→  1133→        return current
