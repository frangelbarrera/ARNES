     1→"""
     2→ARNES Specialist — a pre-built, role-based agent with tool-use loop.
     3→
     4→A Specialist is a (system_prompt + tools + output_schema) bundle. The default
     5→`run()` implementation executes a ReAct-style tool-use loop:
     6→
     7→1. Format input as user message.
     8→2. Call LLM with tools registered.
     9→3. If LLM returns tool_calls, execute each tool and append results.
    10→4. Repeat until LLM returns final response (no tool_calls) or max_iterations.
    11→5. Validate response against output_schema (pydantic).
    12→6. Return structured result.
    13→
    14→Specialists are stateless. State lives in the Thread, not in the specialist.
    15→"""
    16→
    17→from __future__ import annotations
    18→
    19→import json
    20→from abc import ABC
    21→from collections.abc import AsyncIterator
    22→from typing import Any, ClassVar
    23→
    24→import structlog
    25→from pydantic import BaseModel, ConfigDict, Field, ValidationError
    26→
    27→from arnes.llm.base import LLMMessage, LLMProvider, LLMResponse, LLMUsage
    28→from arnes.middleware.cost_guard import BudgetExceeded
    29→from arnes.thread.events import AssistantMessageEvent
    30→from arnes.tools.base import Tool, ToolContext, ToolRegistry

    35→def _drain_event_to_sink(
    36→    wrapped_provider: Any,
    37→    event: AssistantMessageEvent,
    38→) -> None:
    39→    """Append an ``AssistantMessageEvent`` to the wrapped provider's ``_events`` sink.
    40→
    41→    Shared helper used by both :class:`Specialist._emit_assistant_message`
    42→    and :class:`arnes.agent.Harness._emit_stream_audit_event` (R13 DRY pass).
    43→
    44→    The middleware (``CostGuard``) sets up a shared ``_events`` list on the
    45→    wrapped provider so events can be drained later by the executor's
    46→    ``_drain_middleware_events`` and appended to the actual ``Thread``. This
    47→    helper centralises the defensive ``getattr`` + ``isinstance(list)`` guard
    48→    that both call sites previously duplicated.
    49→
    50→    If ``wrapped_provider`` has no ``_events`` attribute (e.g. a raw
    51→    third-party provider without our middleware), the emission is a no-op —
    52→    the caller still records step-level tokens/cost on the
    53→    ``StepCompletedEvent``, so no data is lost.
    54→    """
    55→    events_list = getattr(wrapped_provider, "_events", None)
    56→    if not isinstance(events_list, list):
    57→        return
    58→    events_list.append(event)

    61→class SpecialistConfig(BaseModel):
    62→    """Configuration for a specialist."""
    63→
    64→    model_config = ConfigDict(arbitrary_types_allowed=True)
    65→
    66→    name: str  # e.g. "@planner"
    67→    description: str
    68→    system_prompt: str
    69→    tools: list[str] = Field(default_factory=list)  # tool names
    70→    output_schema: dict[str, Any] | None = None  # JSON schema for structured output
    71→    pydantic_model: type[BaseModel] | None = None  # Stronger than output_schema
    72→    default_model: str | None = None  # If set, overrides the global default
    73→    temperature: float = 0.0
    74→    max_tokens: int | None = None
    75→    max_iterations: int = 5  # ReAct loop limit
    76→

    78→class Specialist(ABC):
    79→    """Base class for all ARNES specialists.
    80→
    81→    To add a specialist:
    82→        class MySpecialist(Specialist):
    83→            config = SpecialistConfig(
    84→                name="@my-specialist",
    85→                description="Does X",
    86→                system_prompt="You are an expert in X...",
    87→                tools=["fs_read", "shell"],
    88→                output_schema={"type": "object", "required": ["result"]},
    89→            )
    90→    """
    91→
    92→    config: ClassVar[SpecialistConfig]
    93→
    94→    # Auto-registry
    95→    _registry: ClassVar[dict[str, type[Specialist]]] = {}
    96→
    97→    def __init_subclass__(cls, **kwargs: Any) -> None:
    98→        super().__init_subclass__(**kwargs)
    99→        if hasattr(cls, "config") and cls.config.name:
   100→            Specialist._registry[cls.config.name] = cls

   102→    async def run(
   103→        self,
   104→        input_data: dict[str, Any],
   105→        ctx: ToolContext,
   106→        *,
   107→        provider: LLMProvider,
   108→        tool_registry: ToolRegistry | None = None,
   109→    ) -> dict[str, Any]:
   110→        """Default specialist run: ReAct tool-use loop + schema validation.
   111→
   112→        Override this for custom logic if needed, but most specialists should
   113→        use the default implementation.
   114→        """
   115→        # Build initial messages
   116→        user_content = self._format_input(input_data)
   117→        messages: list[LLMMessage] = [
   118→            LLMMessage(role="system", content=self.config.system_prompt),
   119→            LLMMessage(role="user", content=user_content),
   120→        ]
   121→
   122→        # Get available tools (intersect config.tools with registry)
   123→        available_tools: list[Tool] = []
   124→        tool_schemas: list[dict[str, Any]] = []
   125→        if tool_registry and self.config.tools:
   126→            for tool_name in self.config.tools:
   127→                tool = tool_registry.get(tool_name)
   128→                if tool:
   129→                    available_tools.append(tool)
   130→                    tool_schemas.append(self._tool_to_schema(tool))
   131→
   132→        # Build middleware-wrapped provider
   133→        # The caller (Harness or PlaybookExecutor) is responsible for wrapping
   134→        # the provider with the full middleware stack (CostGuard → Verification →
   135→        # TokenOptimizer → provider). The specialist should NOT re-wrap.
   136→        #
   137→        # We detect already-wrapped providers by checking for the _arnes_wrapped
   138→        # marker attribute that all our middleware classes set.
   139→        wrapped_provider: LLMProvider = provider
   140→
   141→        # If the provider is not already wrapped, wrap it with the full stack.
   142→        # This is a safety net — in normal usage, the caller always wraps first.
   143→        if not getattr(provider, "_arnes_wrapped", False):
   144→            from arnes.middleware import build_middleware_stack
   145→
   146→            wrapped_provider = build_middleware_stack(
   147→                provider,
   148→                enable_cache=True,
   149→                enable_verification=True,
   150→                budget_usd=1.0,
   151→                output_schema=self.config.output_schema,
   152→                pydantic_model=self.config.pydantic_model,
   153→            )
   154→
   155→        # ReAct tool-use loop
   156→        total_usage = LLMUsage()
   157→        all_tool_results: list[dict[str, Any]] = []
   158→        model = self.config.default_model or "ollama/llama3.2"
   159→
   160→        # Derive the response_schema sent to middleware: prefer the explicit
   161→        # output_schema, but fall back to a pydantic_model's JSON schema so that
   162→        # specialists that only declare a `pydantic_model` still get JSON-mode
   163→        # forcing AND schema validation in the VerificationLayer. Without this,
   164→        # `response_schema=None` would silently disable both.
   165→        effective_response_schema = self.config.output_schema
   166→        if effective_response_schema is None and self.config.pydantic_model is not None:
   167→            effective_response_schema = self.config.pydantic_model.model_json_schema()
   168→
   169→        wants_json = bool(self.config.output_schema or self.config.pydantic_model)
   170→
   171→        # Track whether the loop produced a *final* response (no tool_calls).
   172→        # If the LLM keeps calling tools until max_iterations is hit, the last
   173→        # `response` will still have tool_calls and we must NOT validate it as
   174→        # if it were the final answer — that would surface an empty/intermediate
   175→        # tool-call payload as a malformed "final" response.
   176→        final_response: LLMResponse | None = None
   177→        response: LLMResponse | None = None
   178→
   179→        for iteration in range(self.config.max_iterations):
   180→            try:
   181→                response = await wrapped_provider.complete(
   182→                    messages,
   183→                    model=model,
   184→                    tools=tool_schemas if tool_schemas else None,
   185→                    temperature=self.config.temperature,
   186→                    max_tokens=self.config.max_tokens,
   187→                    response_format={"type": "json_object"} if wants_json else None,
   188→                    response_schema=effective_response_schema,
   189→                    interactive=ctx.metadata.get("interactive", False),
   190→                )
   191→            except BudgetExceeded as e:
   192→                logger.error(
   193→                    "specialist_budget_exceeded", specialist=self.config.name, error=str(e)
   194→                )
   195→                return {
   196→                    "specialist": self.config.name,
   197→                    "success": False,
   198→                    "error": f"Budget exceeded: {e}",
   199→                    "budget_exceeded": True,
   200→                }
   201→
   202→            total_usage = total_usage + response.usage
   203→
   204→            # Emit an AssistantMessageEvent for every LLM call so the
   205→            # conversation history and per-call token/cost are observable
   206→            # in the thread bitácora. The specialist has direct access to
   207→            # ctx.thread_id and ctx.step_id, so we can construct the event
   208→            # with the correct ids (no nil-UUID patching needed). The
   209→            # event is appended to the wrapped provider's shared event sink
   210→            # (CostGuard._events), which the PlaybookExecutor drains after
   211→            # the step completes.
   212→            self._emit_assistant_message(wrapped_provider, ctx, response, model)
   213→
   214→            # If no tool calls, we have the final response
   215→            if not response.tool_calls:
   216→                final_response = response
   217→                break
   218→
   219→            # Execute each tool call
   220→            messages.append(
   221→                LLMMessage(
   222→                    role="assistant",
   223→                    content=response.content,
   224→                    tool_calls=response.tool_calls,
   225→                )
   226→            )
   227→
   228→            for tc in response.tool_calls:
   229→                tool_result = await self._execute_tool_call(tc, available_tools, ctx)
   230→                all_tool_results.append(tool_result)
   231→                messages.append(
   232→                    LLMMessage(
   233→                        role="tool",
   234→                        content=json.dumps(tool_result, default=str),
   235→                        tool_call_id=tc.get("id"),
   236→                        name=tc.get("function", {}).get("name"),
   237→                    )
   238→                )
   239→
   240→            # Continue loop for next iteration
   241→
   242→        # If we exited the loop without a final (tool-call-free) response,
   243→        # the LLM kept calling tools past max_iterations. Return a clear error
   244→        # instead of trying to validate an empty / intermediate tool-call payload.
   245→        if final_response is None:
   246→            max_iter = self.config.max_iterations
   247→            logger.error(
   248→                "specialist_max_iterations_exceeded",
   249→                specialist=self.config.name,
   250→                max_iterations=max_iter,
   251→            )
   252→            return {
   253→                "specialist": self.config.name,
   254→                "success": False,
   255→                "error": (
   256→                    f"Specialist exceeded max_iterations ({max_iter}) "
   257→                    "without producing a final response"
   258→                ),
   259→                "raw": response.content if response is not None else None,
   260→                "usage": total_usage.model_dump(),
   261→                "tool_results": all_tool_results,
   262→            }
   263→
   264→        # Validate output against schema
   265→        result = self._parse_and_validate_output(final_response, total_usage, all_tool_results)
   266→        return result
   267→
   268→    async def stream(
   269→        self,
   270→        input_data: dict[str, Any],
   271→        ctx: ToolContext,
   272→        *,
   273→        provider: LLMProvider,
   274→        tool_registry: ToolRegistry | None = None,
   275→    ) -> AsyncIterator[LLMResponse]:
   276→        """Stream a specialist's response token by token.
   277→
   278→        Mirrors :meth:`run` but uses ``provider.stream_complete()`` instead
   279→        of ``provider.complete()``. Yields ``LLMResponse`` chunks as they
   280→        arrive from the provider. The final chunk carries the full
   281→        ``LLMUsage`` (``tokens_in``, ``tokens_out``, ``cost_usd``).
   282→
   283→        Streaming specialists do NOT execute the ReAct tool-use loop — if
   284→        the specialist needs tools, callers should use :meth:`run` instead.
   285→        This is a best-effort streaming path for read-only / generation
   286→        tasks (planning, summarising, drafting code without shell access).
   287→
   288→        Audit trail: after the stream completes, a single
   289→        :class:`AssistantMessageEvent` is appended to the wrapped
   290→        provider's ``_events`` sink (same pattern as :meth:`run`). The
   291→        event carries the full accumulated content and the final usage,
   292→        so the bitácora records streaming runs the same way it records
   293→        non-streaming runs. Streaming produces ONE event per call (not
   294→        per-chunk events) — per-chunk events would balloon the audit log
   295→        without adding forensic value, and the chunks themselves are not
   296→        durable (they're a transport optimisation, not a semantic unit).
   297→
   298→        Usage::
   299→
   300→            async for chunk in specialist.stream(input_data, ctx, provider=p):
   301→                print(chunk.content, end="", flush=True)
   302→        """
   303→        # Build messages (same as run(), minus the tool-use loop)
   304→        user_content = self._format_input(input_data)
   305→        messages: list[LLMMessage] = [
   306→            LLMMessage(role="system", content=self.config.system_prompt),
   307→            LLMMessage(role="user", content=user_content),
   308→        ]
   309→
   310→        # Wrap provider with the full middleware stack if not already wrapped
   311→        # (mirrors run()).
   312→        wrapped_provider: LLMProvider = provider
   313→        if not getattr(provider, "_arnes_wrapped", False):
   314→            from arnes.middleware import build_middleware_stack
   315→
   316→            wrapped_provider = build_middleware_stack(
   317→                provider,
   318→                enable_cache=True,
   319→                enable_verification=True,
   320→                budget_usd=1.0,
   321→                output_schema=self.config.output_schema,
   322→                pydantic_model=self.config.pydantic_model,
   323→            )
   324→
   325→        model = self.config.default_model or "ollama/llama3.2"
   326→        effective_response_schema = self.config.output_schema
   327→        if effective_response_schema is None and self.config.pydantic_model is not None:
   328→            effective_response_schema = self.config.pydantic_model.model_json_schema()
   329→        wants_json = bool(self.config.output_schema or self.config.pydantic_model)
   330→
   331→        # Accumulate content + usage so we can emit a single
   332→        # AssistantMessageEvent after the stream ends. Per-chunk usage is
   333→        # zero on intermediate chunks (see LLMProvider.stream_complete
   334→        # contract); the final chunk carries the full usage.
   335→        accumulated_content: list[str] = []
   336→        total_usage = LLMUsage()
   337→
   338→        async for chunk in wrapped_provider.stream_complete(
   339→            messages,
   340→            model=model,
   341→            temperature=self.config.temperature,
   342→            max_tokens=self.config.max_tokens,
   343→            response_format={"type": "json_object"} if wants_json else None,
   344→            response_schema=effective_response_schema,
   345→        ):
   346→            if chunk.content:
   347→                accumulated_content.append(chunk.content)
   348→            # The last non-zero usage wins (providers send the full count
   349→            # on the final chunk, not a running delta).
   350→            if chunk.usage.tokens_in > 0 or chunk.usage.tokens_out > 0 or chunk.usage.cost_usd > 0:
   351→                total_usage = chunk.usage
   352→            yield chunk
   353→
   354→        # Emit a single AssistantMessageEvent for the bitácora. We
   355→        # construct a synthetic LLMResponse carrying the accumulated
   356→        # content + final usage so we can reuse the same _emit_assistant_message
   357→        # helper that run() uses (DRY: same event shape, same sink).
   358→        synthetic_response = LLMResponse(
   359→            content="".join(accumulated_content),
   360→            tool_calls=[],
   361→            usage=total_usage,
   362→            model=model,
   363→        )
   364→        self._emit_assistant_message(wrapped_provider, ctx, synthetic_response, model)
   365→

   370→    async def _execute_tool_call(
   371→        self,
   372→        tool_call: dict[str, Any],
   373→        available_tools: list[Tool],
   374→        ctx: ToolContext,
   375→    ) -> dict[str, Any]:
   376→        """Execute a single tool call from the LLM."""
   377→        function = tool_call.get("function", {})
   378→        tool_name = function.get("name", "")
   379→        args_str = function.get("arguments", "{}")
   380→
   381→        try:
   382→            args = json.loads(args_str) if isinstance(args_str, str) else args_str
   383→        except json.JSONDecodeError:
   384→            return {
   385→                "tool": tool_name,
   386→                "success": False,
   387→                "error": f"Invalid JSON arguments: {args_str}",
   388→            }
   389→
   390→        # Find the tool
   391→        tool = next((t for t in available_tools if t.name == tool_name), None)
   392→        if not tool:
   393→            return {
   394→                "tool": tool_name,
   395→                "success": False,
   396→                "error": f"Tool '{tool_name}' not available",
   397→            }
   398→
   399→        # HITL check: if tool requires approval, fingerprint args and
   400→        # compare against any pre-approved fingerprint (rug-pull defense).
   401→        # NOTE: use setdefault (not .get) so the SAME dict object is stored
   402→        # back into ctx.metadata — otherwise approvals are never persisted
   403→        # and the rug-pull detector is defeated.
   404→        if tool.requires_approval:
   405→            fingerprint = Tool.fingerprint(args)
   406→            approved_fingerprints = ctx.metadata.setdefault("_approved_fingerprints", {})
   407→
   408→            if fingerprint in approved_fingerprints:
   409→                # Pre-approved: execute
   410→                logger.info(
   411→                    "tool_approved_fingerprint_match",
   412→                    tool=tool_name,
   413→                    fingerprint=fingerprint,
   414→                )
   415→            elif not ctx.metadata.get("interactive", False):
   416→                # Non-interactive and not pre-approved: auto-reject
   417→                return {
   418→                    "tool": tool_name,
   419→                    "success": False,
   420→                    "error": f"Tool '{tool_name}' requires human approval. Set interactive=True.",
   421→                    "fingerprint": fingerprint,
   422→                }
   423→            else:
   424→                # Interactive: prompt for approval
   425→                approval = await self._request_human_approval(tool_name, args, fingerprint, ctx)
   426→                if not approval:
   427→                    return {
   428→                        "tool": tool_name,
   429→                        "success": False,
   430→                        "error": f"Human rejected tool call for '{tool_name}'.",
   431→                        "fingerprint": fingerprint,
   432→                    }
   433→                # Record the approved fingerprint so we can detect rug-pull
   434→                # if the LLM tries to call the same tool with different args
   435→                approved_fingerprints[fingerprint] = tool_name
   436→
   437→        try:
   438→            result = await tool.execute(args, ctx)
   439→            return {
   440→                "tool": tool_name,
   441→                "success": result.success,
   442→                "output": result.output,
   443→                "error": result.error,
   444→            }
   445→        except Exception as e:
   446→            logger.exception("tool_execution_failed", tool=tool_name, error=str(e))
   447→            return {
   448→                "tool": tool_name,
   449→                "success": False,
   450→                "error": str(e),
   451→            }
   452→
   453→    async def _request_human_approval(
   454→        self,
   455→        tool_name: str,
   456→        args: dict[str, Any],
   457→        fingerprint: str,
   458→        ctx: ToolContext,
   459→    ) -> bool:
   460→        """Request human approval for a tool call. Returns True if approved."""
   461→        try:
   462→            from rich.console import Console
   463→            from rich.prompt import Confirm
   464→
   465→            console = Console()
   466→            console.print(f"\n[yellow]⚠ Approval required for tool:[/yellow] {tool_name}")
   467→            console.print(f"  [dim]Args fingerprint:[/dim] {fingerprint}")
   468→            console.print(f"  [dim]Args:[/dim] {json.dumps(args, indent=2, default=str)}")
   469→            return Confirm.ask("  Approve this tool call?", default=False)
   470→        except ImportError:
   471→            logger.warning("rich not available — auto-rejecting tool call")
   472→            return False

   475→    # Event emission
   476→    # ============================================================
   477→
   478→    def _emit_assistant_message(
   479→        self,
   480→        wrapped_provider: Any,
   481→        ctx: ToolContext,
   482→        response: LLMResponse,
   483→        model: str,
   484→    ) -> None:
   485→        """Emit an ``AssistantMessageEvent`` for a single LLM call.
   486→
   487→        The event carries the response content, the model used, and the
   488→        per-call token usage and cost so that the conversation history is
   489→        fully observable in the thread bitácora. The event is appended to
   490→        the wrapped provider's shared ``_events`` sink (set up by
   491→        ``CostGuard``); the ``PlaybookExecutor`` drains that sink after each
   492→        step and appends the events to the ``Thread``.
   493→
   494→        Delegates to the shared module-level :func:`_drain_event_to_sink`
   495→        helper so the "get list / type-guard / append" defensive pattern is
   496→        not duplicated between :class:`Specialist` and :class:`Harness`.
   497→
   498→        If ``wrapped_provider`` has no ``_events`` attribute (e.g. a raw
   499→        third-party provider), the emission is a no-op — the executor will
   500→        still record step-level tokens/cost on the ``StepCompletedEvent``.
   501→        """
   502→        event = AssistantMessageEvent(
   503→            thread_id=ctx.thread_id,
   504→            step_id=ctx.step_id,
   505→            specialist=self.config.name,
   506→            data={
   507→                "content": response.content,
   508→                "model": response.usage.model or model,
   509→                "tokens_in": response.usage.tokens_in,
   510→                "tokens_out": response.usage.tokens_out,
   511→                "cost_usd": response.usage.cost_usd,
   512→                "cached": response.usage.cached,
   513→            },
   514→        )
   515→        _drain_event_to_sink(wrapped_provider, event)
   516→
   517→    # ============================================================
   518→    # Schema validation
   519→    # ============================================================
   520→
   521→    def _parse_and_validate_output(
   522→        self,
   523→        response: LLMResponse,
   524→        total_usage: LLMUsage,
   525→        tool_results: list[dict[str, Any]],
   526→    ) -> dict[str, Any]:
   527→        """Parse LLM response and validate against pydantic_model or output_schema."""
   528→        if not response.content:
   529→            return {
   530→                "specialist": self.config.name,
   531→                "success": False,
   532→                "error": "Empty response from LLM",
   533→                "raw": None,
   534→                "usage": total_usage.model_dump(),
   535→                "tool_results": tool_results,
   536→            }
   537→
   538→        # Strip ```json ... ``` / ``` ... ``` fences that some models (notably
   539→        # Llama 3.2) wrap around a JSON payload despite being asked for raw JSON.
   540→        cleaned = self._clean_json_response(response.content)
   541→
   542→        # Try JSON parse
   543→        try:
   544→            parsed = json.loads(cleaned)
   545→        except json.JSONDecodeError:
   546→            # If we expected JSON, this is a failure
   547→            if self.config.output_schema or self.config.pydantic_model:
   548→                return {
   549→                    "specialist": self.config.name,
   550→                    "success": False,
   551→                    "error": f"LLM did not return valid JSON. Got: {response.content[:200]}",
   552→                    "raw": response.content,
   553→                    "usage": total_usage.model_dump(),
   554→                    "tool_results": tool_results,
   555→                }
   556→            # If no schema expected, return raw content
   557→            parsed = {"raw": response.content}
   558→
   559→        # Strong validation with pydantic model if defined (preferred over
   560→        # the weak JSON-schema `required`-fields check below — pydantic gives
   561→        # us type coercion + full validation).
   562→        if self.config.pydantic_model:
   563→            try:
   564→                validated = self.config.pydantic_model.model_validate(parsed)
   565→                return {
   566→                    "specialist": self.config.name,
   567→                    "success": True,
   568→                    "output": validated.model_dump(),
   569→                    "usage": total_usage.model_dump(),
   570→                    "tool_results": tool_results,
   571→                }
   572→            except ValidationError as e:
   573→                return {
   574→                    "specialist": self.config.name,
   575→                    "success": False,
   576→                    "error": f"Output schema validation failed: {e}",
   577→                    "raw": parsed,
   578→                    "usage": total_usage.model_dump(),
   579→                    "tool_results": tool_results,
   580→                }
   581→
   582→        # Weak validation with JSON schema (required fields only)
   583→        if self.config.output_schema:
   584→            required = self.config.output_schema.get("required", [])
   585→            missing = [f for f in required if f not in parsed]
   586→            if missing:
   587→                return {
   588→                    "specialist": self.config.name,
   589→                    "success": False,
   590→                    "error": f"Missing required fields: {missing}",
   591→                    "raw": parsed,
   592→                    "usage": total_usage.model_dump(),
   593→                    "tool_results": tool_results,
   594→                }
   595→
   596→        return {
   597→            "specialist": self.config.name,
   598→            "success": True,
   599→            "output": parsed,
   600→            "usage": total_usage.model_dump(),
   601→            "tool_results": tool_results,
   602→        }

   616→    @staticmethod
   617→    def _clean_json_response(content: str) -> str:
   618→        """Strip markdown code-fences wrappers from an LLM JSON response.
   619→
   620→        Many local models (Llama 3.2 in particular) ignore a ``response_format``
   621→        hint and wrap their JSON in ```` ```json ... ``` ```` or plain ```` ``` ... ``` ````.
   622→        ``json.loads`` can't parse that, so peel the fences off before parsing.
   623→
   624→        Only strips the *outer* fence — JSON string values that legitimately
   625→        contain ``` are left untouched because we only trim a leading fence
   626→        and a trailing fence, never interior backticks.
   627→        """
   628→        # ``content`` is typed as ``str``; the isinstance guard below is a
   629→        # defensive runtime check for callers that bypass the type system
   630→        # (e.g. third-party providers returning bytes). It is unreachable per
   631→        # mypy but kept as defense-in-depth.
   632→        if not isinstance(content, str):
   633→            return content  # type: ignore[unreachable]
   634→        text = content.strip()
   635→        # Match an opening fence ```` ```json ```` / ```` ```jsonYAML ```` / ```` ``` ````
   636→        # followed by content and a closing ```` ``` ````. Non-greedy so we grab
   637→        # the smallest outer fence.
   638→        if text.startswith("```"):
   639→            # Drop the opening fence line (and any language tag after it).
   640→            first_newline = text.find("\n")
   641→            # Single-line fence like ```{} ``` (no newline) → strip 3 backticks
   642→            # either side; otherwise slice off the whole opening fence line.
   643→            text = text[first_newline + 1 :] if first_newline != -1 else text[3:]
   644→            # Strip a trailing ``` if present.
   645→            if text.endswith("```"):
   646→                text = text[:-3]
   647→        return text.strip()

   649→    def _tool_to_schema(self, tool: Tool) -> dict[str, Any]:
   650→        """Convert an ARNES Tool to OpenAI tool schema for LLM."""
   651→        args_schema = getattr(tool, "Args", None)
   652→        return {
   653→            "type": "function",
   654→            "function": {
   655→                "name": tool.name,
   656→                "description": tool.description,
   657→                "parameters": args_schema.model_json_schema()
   658→                if args_schema
   659→                else {"type": "object", "properties": {}},
   660→            },
   661→        }

   664→class SpecialistRegistry:
   665→    """Registry of available specialists."""
   666→
   667→    def __init__(self) -> None:
   668→        self._specialists: dict[str, Specialist] = {}
   669→
   670→    def register(self, specialist: Specialist) -> None:
   671→        if not specialist.config.name:
   672→            raise ValueError("Specialist must have a name in its config")
   673→        self._specialists[specialist.config.name] = specialist
   674→
   675→    def register_class(self, specialist_class: type[Specialist]) -> None:
   676→        instance = specialist_class()
   677→        self.register(instance)
   678→
   679→    def get(self, name: str) -> Specialist | None:
   680→        if not name.startswith("@"):
   681→            name = "@" + name
   682→        return self._specialists.get(name)
   683→
   684→    def list_names(self) -> list[str]:
   685→        """Return a sorted list of registered specialist names."""
   686→        return sorted(self._specialists.keys())
   687→
   688→    def has(self, name: str) -> bool:
   689→        return self.get(name) is not None
   690→
   691→    def configs(self) -> list[SpecialistConfig]:
   692→        return [s.config for s in self._specialists.values()]
   693→
   694→
   695→def get_default_specialist_registry() -> SpecialistRegistry:
   696→    """Return a registry with all built-in specialists registered."""
   697→    registry = SpecialistRegistry()
   698→    from arnes.specialists.coder import Coder
   699→    from arnes.specialists.debugger import Debugger
   700→    from arnes.specialists.planner import Planner
   701→    from arnes.specialists.reviewer import Reviewer
   702→    from arnes.specialists.tester import Tester
   703→
   704→    for cls in [Planner, Coder, Reviewer, Tester, Debugger]:
   705→        registry.register_class(cls)
   706→    return registry
