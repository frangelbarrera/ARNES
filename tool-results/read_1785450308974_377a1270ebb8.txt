    34→# ============================================================
    35→# Shell tool — executes commands in sandbox by default
    36→# ============================================================
    37→
    38→
    39→class ShellTool(Tool):
    40→    """Execute a shell command. Sandboxed by default (Docker Tier 1).
    41→
    42→    SECURITY: Local execution (no sandbox) requires BOTH:
    43→    - ctx.sandbox_enabled = False
    44→    - ARNES_DEV_MODE environment variable set to "1"
    45→
    46→    This double-gate prevents accidental RCE in production.
    47→
    48→    .. warning::
    49→        The dangerous-command regex checked by ``_is_dangerous_command`` is
    50→        **defense-in-depth only** and is trivially bypassable by an
    51→        adversarial prompt (obfuscation, env-var expansion, heredocs,
    52→        aliasing, etc.). It exists to catch careless commands, not to
    53→        substitute for a real sandbox (Docker / nsjail / gVisor). For
    54→        untrusted input, ALWAYS run inside ``ctx.sandbox_container``.
    55→    """
    56→
    57→    name: ClassVar[str] = "shell"
    58→    description: ClassVar[str] = (
    59→        "Execute a shell command. Returns stdout, stderr, and exit code. "
    60→        "Use for: build, test, run scripts, inspect files. "
    61→        "Sandboxed by default — local execution requires ARNES_DEV_MODE=1."
    62→    )
    63→    requires_approval: ClassVar[bool] = True  # Destructive by default
    64→    sandbox_tier: ClassVar[int | None] = 1
    65→
    66→    class Args(BaseModel):
    67→        command: str = Field(..., description="Shell command to execute")
    68→        cwd: str = Field(default=".", description="Working directory inside sandbox")
    69→        timeout_s: int = Field(default=30, ge=1, le=300)
    70→        env: dict[str, str] = Field(default_factory=dict)
    71→
    72→    async def execute(self, args: dict[str, Any], ctx: ToolContext) -> ToolResult:
    73→        start = time.monotonic()
    74→        try:
    75→            validated = self.Args.model_validate(args)
    76→        except Exception as e:
    77→            return ToolResult.fail("shell", f"Invalid args: {e}")
    78→
    79→        cmd = validated.command
    80→        if _is_dangerous_command(cmd):
    81→            return ToolResult.fail(
    82→                "shell",
    83→                "Blocked: command matches dangerous pattern. Use a more specific tool.",
    84→                duration_s=time.monotonic() - start,
    85→            )
    86→
    87→        if ctx.sandbox_enabled and ctx.sandbox_container:
    88→            result = await self._execute_in_sandbox(validated, ctx)
    89→        else:
    90→            # SECURITY: Require explicit dev mode override
    91→            if os.getenv("ARNES_DEV_MODE", "0") != "1":
    92→                return ToolResult.fail(
    93→                    "shell",
    94→                    "Local shell execution disabled by default. Set ARNES_DEV_MODE=1 or "
    95→                    "configure a sandbox container in ToolContext.",
    96→                    duration_s=time.monotonic() - start,
    97→                )
    98→            result = await self._execute_local(validated, ctx)
    99→
   100→        result.duration_s = time.monotonic() - start
   101→        return result
   102→
   103→    async def _execute_local(self, args: ShellTool.Args, ctx: ToolContext) -> ToolResult:
   104→        """Local execution (dev mode only). NEVER inherits os.environ — only
   105→        explicit env vars from args.env are passed (and secrets are filtered)."""
   106→        env: dict[str, str] = {}
   107→
   108→        for key, value in args.env.items():
   109→            if _looks_like_secret(key):
   110→                continue
   111→            if key in ("PATH", "LD_PRELOAD", "LD_LIBRARY_PATH", "PYTHONPATH"):
   112→                continue
   113→            env[key] = value
   114→
   115→        # Cross-platform PATH: inherit from os.environ if available
   116→        # (needed for basic commands like 'python', 'echo', etc.)
   117→        if "PATH" not in env:
   118→            env["PATH"] = os.environ.get("PATH", "/usr/local/bin:/usr/bin:/bin")
   119→        # On Windows, also preserve SystemRoot and COMSPEC (needed for subprocess)
   120→        if sys.platform == "win32":
   121→            if "SYSTEMROOT" not in env:
   122→                env["SYSTEMROOT"] = os.environ.get("SYSTEMROOT", r"C:\Windows")
   123→            if "COMSPEC" not in env:
   124→                env["COMSPEC"] = os.environ.get("COMSPEC", r"C:\Windows\System32\cmd.exe")
   125→
   126→        try:
   127→            proc = await asyncio.create_subprocess_shell(
   128→                args.command,
   129→                stdout=asyncio.subprocess.PIPE,
   130→                stderr=asyncio.subprocess.PIPE,
   131→                cwd=args.cwd,
   132→                env=env,
   133→            )
   134→            stdout, stderr = await asyncio.wait_for(proc.communicate(), timeout=args.timeout_s)
   135→            return ToolResult(
   136→                tool="shell",
   137→                success=proc.returncode == 0,
   138→                output={
   139→                    "stdout": stdout.decode("utf-8", errors="replace"),
   140→                    "stderr": stderr.decode("utf-8", errors="replace"),
   141→                    "exit_code": proc.returncode,
   142→                    "mode": "local-dev",
   143→                },
   144→                error=None if proc.returncode == 0 else f"Exit code: {proc.returncode}",
   145→            )
   146→        except TimeoutError:
   147→            return ToolResult.fail("shell", f"Timeout after {args.timeout_s}s")
   148→        except Exception as e:
   149→            return ToolResult.fail("shell", str(e))
   150→
   151→    async def _execute_in_sandbox(self, args: ShellTool.Args, ctx: ToolContext) -> ToolResult:
   152→        """Docker-hardened execution (Tier 1 sandbox)."""
   153→        docker_cmd = [
   154→            "docker",
   155→            "run",
   156→            "--rm",
   157→            "--security-opt=no-new-privileges",
   158→            "--cap-drop=ALL",
   159→            "--network=none",
   160→            "--read-only",
   161→            "--tmpfs",
   162→            "/workspace:size=100M",
   163→            "-w",
   164→            "/workspace",
   165→        ]
   166→        for key, value in args.env.items():
   167→            if _looks_like_secret(key):
   168→                continue
   169→            if key in ("PATH", "LD_PRELOAD", "LD_LIBRARY_PATH", "PYTHONPATH"):
   170→                continue
   171→            docker_cmd.extend(["-e", f"{key}={value}"])
   172→
   173→        docker_cmd.extend(
   174→            [ctx.sandbox_container or "arnes-sandbox:latest", "sh", "-c", args.command]
   175→        )
   176→
   177→        try:
   178→            proc = await asyncio.create_subprocess_exec(
   179→                *docker_cmd,
   180→                stdout=asyncio.subprocess.PIPE,
   181→                stderr=asyncio.subprocess.PIPE,
   182→            )
   183→            stdout, stderr = await asyncio.wait_for(proc.communicate(), timeout=args.timeout_s)
   184→            return ToolResult(
   185→                tool="shell",
   186→                success=proc.returncode == 0,
   187→                output={
   188→                    "stdout": stdout.decode("utf-8", errors="replace"),
   189→                    "stderr": stderr.decode("utf-8", errors="replace"),
   190→                    "exit_code": proc.returncode,
   191→                    "sandbox": "docker-tier1",
   192→                },
   193→                error=None if proc.returncode == 0 else f"Exit code: {proc.returncode}",
   194→            )
   195→        except FileNotFoundError:
   196→            return ToolResult.fail(
   197→                "shell", "Docker not available. Set ARNES_DEV_MODE=1 for local execution."
   198→            )
   199→        except TimeoutError:
   200→            return ToolResult.fail("shell", f"Sandbox timeout after {args.timeout_s}s")

   203→# ============================================================
   204→# HTTP tool — calls external APIs with full SSRF protection
   205→# ============================================================
   206→
   207→
   208→class HttpTool(Tool):
   209→    """Make HTTP requests with full SSRF protection.
   210→
   211→    Security measures:
   212→    - Blocks localhost, private IPs, link-local, multicast by default
   213→    - Performs DNS resolution and validates ALL resolved IPs
   214→    - Blocks cloud metadata endpoints (AWS, GCP, Azure)
   215→    - **DNS-rebinding mitigation**: the resolved IP is pinned for the actual
   216→      httpx request by rewriting the URL to use the IP directly, setting the
   217→      ``Host`` header to the original hostname, and (for HTTPS) setting
   218→      ``sni_hostname`` in the request extensions. Redirects are NOT followed
   219→      by default — a ``Location`` header could point to a brand-new hostname
   220→      and re-trigger DNS resolution.
   221→    """
   222→
   223→    name: ClassVar[str] = "http"
   224→    description: ClassVar[str] = (
   225→        "Make an HTTP request. Returns status, headers, body. "
   226→        "SSRF-protected: blocks localhost, private IPs, cloud metadata endpoints."
   227→    )
   228→
   229→    class Args(BaseModel):
   230→        url: str
   231→        method: str = Field(default="GET", pattern="^(GET|POST|PUT|DELETE|PATCH|HEAD)$")
   232→        headers: dict[str, str] = Field(default_factory=dict)
   233→        body: str | None = None
   234→        timeout_s: float = Field(default=30.0, ge=1.0, le=120.0)
   235→
   236→    async def execute(self, args: dict[str, Any], ctx: ToolContext) -> ToolResult:
   237→        start = time.monotonic()
   238→        try:
   239→            validated = self.Args.model_validate(args)
   240→        except Exception as e:
   241→            return ToolResult.fail("http", f"Invalid args: {e}")
   242→
   243→        # Full SSRF check with DNS resolution. Returns the resolved IP so we
   244→        # can pin it for the request (DNS-rebinding TOCTOU mitigation).
   245→        ssrf_error, resolved_ip = await _check_ssrf_async(validated.url)
   246→        if ssrf_error:
   247→            return ToolResult.fail("http", ssrf_error, duration_s=time.monotonic() - start)
   248→
   249→        headers = dict(validated.headers)
   250→
   251→        # Rewrite the URL to use the resolved IP directly. This prevents
   252→        # httpx from re-resolving DNS (which would re-open the DNS-rebinding
   253→        # TOCTOU window that _check_ssrf_async just closed). The original
   254→        # hostname is preserved as the Host header (HTTP) and as SNI (HTTPS).
   255→        try:
   256→            pinned_url, original_host, scheme = _build_ip_pinned_url(validated.url, resolved_ip)
   257→        except ValueError as e:
   258→            return ToolResult.fail("http", str(e), duration_s=time.monotonic() - start)
   259→
   260→        # The Host header must match the original hostname for virtual-host
   261→        # routing to work. httpx lets us override it explicitly.
   262→        headers.setdefault("Host", original_host)
   263→
   264→        try:
   265→            async with httpx.AsyncClient(
   266→                timeout=validated.timeout_s,
   267→                follow_redirects=False,  # redirects would re-trigger DNS
   268→            ) as client:
   269→                request = client.build_request(
   270→                    method=validated.method,
   271→                    url=pinned_url,
   272→                    headers=headers,
   273→                    content=validated.body,
   274→                )
   275→                # For HTTPS, set SNI to the original hostname so the TLS
   276→                # handshake presents the correct cert and validates against
   277→                # the original hostname (not the IP). The httpcore backend
   278→                # expects ``sni_hostname`` to be a ``str``.
   279→                if scheme == "https":
   280→                    request.extensions["sni_hostname"] = original_host
   281→                response = await client.send(request)
   282→            return ToolResult.ok(
   283→                "http",
   284→                {
   285→                    "status_code": response.status_code,
   286→                    "headers": dict(response.headers),
   287→                    "body": response.text[:10000],  # Truncate to prevent context bloat
   288→                },
   289→                duration_s=time.monotonic() - start,
   290→                url=validated.url,
   291→            )
   292→        except httpx.HTTPError as e:
   293→            return ToolResult.fail("http", str(e), duration_s=time.monotonic() - start)
   294→

   297→# Filesystem tools — path-validated with symlink protection

   299→
   300→
   301→class FilesystemReadTool(Tool):
   302→    """Read a file with path traversal AND symlink protection."""
   303→
   304→    name: ClassVar[str] = "fs_read"
   305→    description: ClassVar[str] = "Read a file. Path must be inside working_dir."
   306→
   307→    class Args(BaseModel):
   308→        path: str
   309→        max_bytes: int = Field(default=65536, ge=1, le=1048576)
   310→
   311→    async def execute(self, args: dict[str, Any], ctx: ToolContext) -> ToolResult:
   312→        try:
   313→            validated = self.Args.model_validate(args)
   314→        except Exception as e:
   315→            return ToolResult.fail("fs_read", f"Invalid args: {e}")
   316→
   317→        safe_path = _validate_path(validated.path, ctx.working_dir)
   318→        if not safe_path:
   319→            return ToolResult.fail("fs_read", f"Path outside working_dir: {validated.path}")
   320→
   321→        # Symlink protection: don't follow symlinks outside working_dir
   322→        if safe_path.is_symlink():
   323→            target = safe_path.resolve()
   324→            base = Path(ctx.working_dir).resolve()
   325→            if base not in target.parents and target != base:
   326→                return ToolResult.fail("fs_read", f"Symlink escapes working_dir: {validated.path}")
   327→
   328→        try:
   329→            content = safe_path.read_bytes()[: validated.max_bytes]
   330→            return ToolResult.ok(
   331→                "fs_read",
   332→                {
   333→                    "path": str(safe_path),
   334→                    "content": content.decode("utf-8", errors="replace"),
   335→                    "size_bytes": len(content),
   336→                },
   337→            )
   338→        except FileNotFoundError:
   339→            return ToolResult.fail("fs_read", f"File not found: {validated.path}")
   340→        except PermissionError:
   341→            return ToolResult.fail("fs_read", f"Permission denied: {validated.path}")
   342→
   343→
   344→class FilesystemWriteTool(Tool):
   345→    """Write a file with path traversal protection. Requires approval."""
   346→
   347→    name: ClassVar[str] = "fs_write"
   348→    description: ClassVar[str] = "Write a file. Path must be inside working_dir. Requires approval."
   349→    requires_approval: ClassVar[bool] = True
   350→
   351→    class Args(BaseModel):
   352→        path: str
   353→        content: str
   354→        mode: str = Field(default="w", pattern="^(w|a)$")
   355→
   356→    async def execute(self, args: dict[str, Any], ctx: ToolContext) -> ToolResult:
   357→        try:
   358→            validated = self.Args.model_validate(args)
   359→        except Exception as e:
   360→            return ToolResult.fail("fs_write", f"Invalid args: {e}")
   361→
   362→        safe_path = _validate_path(validated.path, ctx.working_dir)
   363→        if not safe_path:
   364→            return ToolResult.fail("fs_write", f"Path outside working_dir: {validated.path}")
   365→
   366→        # Symlink protection for write.
   367→        #
   368→        # SECURITY (FIX-R3-SEC): use ``is_symlink()`` ALONE, NOT
   369→        # ``exists() and is_symlink()``. ``Path.exists()`` follows the
   370→        # link and returns False for a DANGLING symlink (target missing),
   371→        # which previously caused this guard to be skipped — letting a
   372→        # write through a dangling symlink that points outside
   373→        # ``working_dir`` (e.g. ``link -> /etc/cron.d/evil`` with the
   374→        # target not yet created). ``is_symlink()`` checks the link
   375→        # entry itself, so it catches both dangling and non-dangling
   376→        # symlinks.
   377→        if safe_path.is_symlink():
   378→            target = safe_path.resolve()
   379→            base = Path(ctx.working_dir).resolve()
   380→            if base not in target.parents and target != base:
   381→                return ToolResult.fail("fs_write", f"Symlink escapes working_dir: {validated.path}")
   382→
   383→        try:
   384→            safe_path.parent.mkdir(parents=True, exist_ok=True)
   385→            with safe_path.open(validated.mode) as f:
   386→                f.write(validated.content)
   387→            return ToolResult.ok(
   388→                "fs_write",
   389→                {"path": str(safe_path), "bytes_written": len(validated.content)},
   390→            )
   391→        except PermissionError:
   392→            return ToolResult.fail("fs_write", f"Permission denied: {validated.path}")
   393→

   450→
   451→_DANGEROUS_PATTERNS = [
   452→    r"\brm\s+-rf\s+/",
   453→    r"\bmkfs\b",
   454→    r"\bdd\s+if=",
   455→    r":\(\)\s*\{",  # fork bomb
   456→    r"\b>\s*/dev/sd[a-z]",
   457→    r"\bchmod\s+-R\s+777\s+/",
   458→    r"\bcurl\s+.*\|\s*sh",
   459→    r"\bwget\s+.*\|\s*sh",
   460→    r"\bnc\s+-l",  # reverse shell
   461→    r"\b/dev/tcp/",  # bash reverse shell
   462→    r"\bnohup\s+.*&\s*$",  # daemon escape
   463→    # --- Added in v0.1.x security hardening (defense-in-depth) ---
   464→    r"\bpython\s+-c\b",  # arbitrary code execution via -c
   465→    r"\bpython3\s+-c\b",  # arbitrary code execution via -c
   466→    r"\beval\s*\(",  # JS/Python eval() invocation
   467→    r"\bexec\s*\(",  # Python exec() invocation
   468→    r"\bfind\s+.*-delete\b",  # find ... -delete
   469→    r"\bbase64\s+-d\b",  # base64 decode (payload decoding)
   470→    r"\bbase64\s+--decode\b",  # base64 decode long form
   471→    r"\b\s{2,}.*&&",  # suspiciously-indented chained commands
   472→]

   474→
   475→def _is_dangerous_command(cmd: str) -> bool:
   476→    """Basic dangerous command detection.
   477→
   478→    .. warning::
   479→        This regex is **defense-in-depth only**. It catches common careless
   480→        payloads but is trivially bypassable by an adversarial prompt
   481→        (obfuscation, env-var expansion, heredocs, aliasing, etc.). It is
   482→        NOT a substitute for running shell commands inside a real sandbox
   483→        (Docker / nsjail / gVisor). For untrusted input, ALWAYS configure
   484→        ``ctx.sandbox_container``.
   485→    """
   486→    return any(re.search(p, cmd, re.IGNORECASE) for p in _DANGEROUS_PATTERNS)
   487→
   488→
   489→def _looks_like_secret(key: str) -> bool:
   490→    """Heuristic: does this env var name suggest it's a secret?"""
   491→    secret_patterns = [
   492→        "API_KEY",
   493→        "SECRET",
   494→        "TOKEN",
   495→        "PASSWORD",
   496→        "PASSWD",
   497→        "CREDENTIAL",
   498→        "PRIVATE_KEY",
   499→    ]
   500→    return any(p in key.upper() for p in secret_patterns)
   501→
   502→
   503→def _validate_path(path: str, working_dir: str) -> Path | None:
   504→    """Validate that path is inside working_dir (path traversal protection).
   505→
   506→    Uses Path.resolve() to canonicalize the path, then checks containment.
   507→    """
   508→    try:
   509→        base = Path(working_dir).resolve(strict=False)
   510→        target = (base / path).resolve(strict=False)
   511→        # Ensure target is inside base
   512→        try:
   513→            target.relative_to(base)
   514→        except ValueError:
   515→            return None
   516→        return target
   517→    except (ValueError, OSError):
   518→        return None

   521→# SSRF protection — full DNS resolution
   522→_BLOCKED_HOSTS = {
   523→    "localhost",
   524→    "ip6-localhost",
   525→    "ip6-loopback",
   526→    "metadata.google.internal",  # GCP metadata
   527→    "metadata.aws.internal",  # AWS metadata (alias)
   528→}
   529→
   530→# Cloud metadata IPs
   531→_BLOCKED_IPS = {
   532→    "169.254.169.254",  # AWS/Azure/GCP metadata
   533→    "100.100.100.200",  # Alibaba Cloud metadata
   534→    "fd00:ec2::254",  # AWS IPv6 metadata

   537→
   538→async def _check_ssrf_async(url: str) -> tuple[str | None, str | None]:
   539→    """Full SSRF check with DNS resolution.
   540→
   541→    Returns ``(error_message, resolved_ip)``:
   542→    - On success: ``(None, ip_str)`` — ``ip_str`` is the resolved IP that
   543→      the caller should pin into the actual HTTP request to defeat
   544→      DNS-rebinding TOCTOU attacks.
   545→    - On failure: ``(error_message, None)``.
   546→
   547→    The caller is responsible for using ``resolved_ip`` when issuing the
   548→    request (see ``_build_ip_pinned_url``). Returning the IP here means
   549→    the SSRF check and the request share the same DNS resolution result.
   550→    """
   551→    try:
   552→        parsed = urllib.parse.urlparse(url)
   553→    except ValueError:
   554→        return f"Invalid URL: {url}", None
   555→
   556→    if parsed.scheme not in ("http", "https"):
   557→        return f"Blocked scheme: {parsed.scheme}", None
   558→
   559→    if not parsed.hostname:
   560→        return "No hostname in URL", None
   561→
   562→    hostname = parsed.hostname.lower()
   563→
   564→    # Block obvious internal hostnames
   565→    if hostname in _BLOCKED_HOSTS:
   566→        return f"Blocked internal host: {hostname}", None
   567→
   568→    # Try to parse as IP first — if the URL already contains an IP, we
   569→    # validate it but don't need to resolve DNS. The IP itself is the
   570→    # pinned IP.
   571→    try:
   572→        ip = ipaddress.ip_address(hostname)
   573→        if _is_blocked_ip(ip):
   574→            return f"Blocked private/loopback IP: {hostname}", None
   575→        return None, str(ip)
   576→    except ValueError:
   577→        pass
   578→
   579→    # It's a hostname — resolve DNS and validate ALL resolved IPs.
   580→    # We also pick the FIRST safe IP to pin for the request.
   581→    try:
   582→        infos = await asyncio.to_thread(socket.getaddrinfo, hostname, None)
   583→    except socket.gaierror:
   584→        return f"DNS resolution failed for: {hostname}", None
   585→
   586→    safe_ip: str | None = None
   587→    for _, _, _, _, sockaddr in infos:
   588→        ip_str = sockaddr[0]
   589→        try:
   590→            ip = ipaddress.ip_address(ip_str)
   591→        except ValueError:
   592→            continue
   593→        if _is_blocked_ip(ip):
   594→            return f"Blocked: {hostname} resolves to private IP {ip_str}", None
   595→        if safe_ip is None:
   596→            safe_ip = str(ip_str)
   597→
   598→    if safe_ip is None:
   599→        return f"No usable IPs resolved for: {hostname}", None
   600→    return None, safe_ip
   601→

   603→def _build_ip_pinned_url(url: str, resolved_ip: str | None) -> tuple[str, str, str]:
   604→    """Rewrite ``url`` to use ``resolved_ip`` as the host.
   605→
   606→    Returns ``(pinned_url, original_hostname, scheme)``.
   607→
   608→    - The path / query / fragment / port / scheme are preserved.
   609→    - The hostname is replaced by the resolved IP (IPv6 is bracketed per
   610→      RFC 3986 so httpx parses it correctly).
   611→    - The original hostname is returned separately so the caller can set
   612→      the ``Host`` header (HTTP) and ``sni_hostname`` extension (HTTPS)
   613→      for TLS cert validation + virtual-host routing.
   614→
   615→    Raises ``ValueError`` if the URL cannot be parsed or no IP is given.
   616→    """
   617→    if resolved_ip is None:
   618→        raise ValueError("No resolved IP available to pin")
   619→    parsed = urllib.parse.urlparse(url)
   620→    if not parsed.hostname:
   621→        raise ValueError("URL has no hostname")
   622→    scheme = parsed.scheme.lower()
   623→    hostname = parsed.hostname.lower()
   624→
   625→    # Bracket IPv6 literals per RFC 3986.
   626→    ip_for_url = resolved_ip
   627→    if ":" in resolved_ip and not resolved_ip.startswith("["):
   628→        ip_for_url = f"[{resolved_ip}]"
   629→
   630→    # Reconstruct netloc: [user[:pass]@]host[:port]
   631→    netloc = ip_for_url
   632→    if parsed.port is not None:
   633→        netloc = f"{ip_for_url}:{parsed.port}"
   634→    if parsed.username:
   635→        userinfo = parsed.username
   636→        if parsed.password:
   637→            userinfo = f"{userinfo}:{parsed.password}"
   638→        netloc = f"{userinfo}@{netloc}"
   639→
   640→    pinned = urllib.parse.urlunparse(
   641→        (
   642→            parsed.scheme,
   643→            netloc,
   644→            parsed.path or "/",
   645→            parsed.params,
   646→            parsed.query,
   647→            parsed.fragment,
   648→        )
   649→    )
   650→    return pinned, hostname, scheme
   651→

   653→def _is_blocked_ip(ip: ipaddress.IPv4Address | ipaddress.IPv6Address) -> bool:
   654→    """Check if an IP should be blocked for SSRF protection."""
   655→    if str(ip) in _BLOCKED_IPS:
   656→        return True
   657→    return (
   658→        ip.is_private
   659→        or ip.is_loopback
   660→        or ip.is_link_local
   661→        or ip.is_multicast
   662→        or ip.is_reserved
   663→        or ip.is_unspecified
   664→    )
