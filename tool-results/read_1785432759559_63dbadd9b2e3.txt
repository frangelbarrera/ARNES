    34→# ============================================================
    35→# Shell tool — executes commands in sandbox by default
    36→# ============================================================
    37→
    38→
    39→class ShellTool(Tool):
    40→    """Execute a shell command. Sandboxed by default (Docker Tier 1).
    41→
    42→    SECURITY: Local execution (no sandbox) requires BOTH:
    43→    - ctx.sandbox_enabled = False
    44→    - ARNES_DEV_MODE environment variable set to "1"
    45→
    46→    This double-gate prevents accidental RCE in production.
    47→
    48→    .. warning::
    49→        The dangerous-command regex checked by ``_is_dangerous_command`` is
    50→        **defense-in-depth only** and is trivially bypassable by an
    51→        adversarial prompt (obfuscation, env-var expansion, heredocs,
    52→        aliasing, etc.). It exists to catch careless commands, not to
    53→        substitute for a real sandbox (Docker / nsjail / gVisor). For
    54→        untrusted input, ALWAYS run inside ``ctx.sandbox_container``.
    55→    """
    56→
    57→    name: ClassVar[str] = "shell"
    58→    description: ClassVar[str] = (
    59→        "Execute a shell command. Returns stdout, stderr, and exit code. "
    60→        "Use for: build, test, run scripts, inspect files. "
    61→        "Sandboxed by default — local execution requires ARNES_DEV_MODE=1."
    62→    )
    63→    requires_approval: ClassVar[bool] = True  # Destructive by default
    64→    sandbox_tier: ClassVar[int | None] = 1
    65→
    66→    class Args(BaseModel):
    67→        command: str = Field(..., description="Shell command to execute")
    68→        cwd: str = Field(default=".", description="Working directory inside sandbox")
    69→        timeout_s: int = Field(default=30, ge=1, le=300)
    70→        env: dict[str, str] = Field(default_factory=dict)
    71→
    72→    async def execute(self, args: dict[str, Any], ctx: ToolContext) -> ToolResult:
    73→        start = time.monotonic()
    74→        try:
    75→            validated = self.Args.model_validate(args)
    76→        except Exception as e:
    77→            return ToolResult.fail("shell", f"Invalid args: {e}")
    78→
    79→        cmd = validated.command
    80→        if _is_dangerous_command(cmd):
    81→            return ToolResult.fail(
    82→                "shell",
    83→                "Blocked: command matches dangerous pattern. Use a more specific tool.",
    84→                duration_s=time.monotonic() - start,
    85→            )
    86→
    87→        if ctx.sandbox_enabled and ctx.sandbox_container:
    88→            result = await self._execute_in_sandbox(validated, ctx)
    89→        else:
    90→            # SECURITY: Require explicit dev mode override
    91→            if os.getenv("ARNES_DEV_MODE", "0") != "1":
    92→                return ToolResult.fail(
    93→                    "shell",
    94→                    "Local shell execution disabled by default. Set ARNES_DEV_MODE=1 or "
    95→                    "configure a sandbox container in ToolContext.",
    96→                    duration_s=time.monotonic() - start,
    97→                )
    98→            result = await self._execute_local(validated, ctx)
    99→
   100→        result.duration_s = time.monotonic() - start
   101→        return result
   102→
   103→    async def _execute_local(self, args: ShellTool.Args, ctx: ToolContext) -> ToolResult:
   104→        """Local execution (dev mode only). NEVER inherits os.environ — only
   105→        explicit env vars from args.env are passed (and secrets are filtered)."""
   106→        env: dict[str, str] = {}
   107→
   108→        for key, value in args.env.items():
   109→            if _looks_like_secret(key):
   110→                continue
   111→            if key in ("PATH", "LD_PRELOAD", "LD_LIBRARY_PATH", "PYTHONPATH"):
   112→                continue
   113→            env[key] = value
   114→
   115→        # Cross-platform PATH: inherit from os.environ if available
   116→        # (needed for basic commands like 'python', 'echo', etc.)
   117→        if "PATH" not in env:
   118→            env["PATH"] = os.environ.get("PATH", "/usr/local/bin:/usr/bin:/bin")
   119→        # On Windows, also preserve SystemRoot and COMSPEC (needed for subprocess)
   120→        if sys.platform == "win32":
   121→            if "SYSTEMROOT" not in env:
   122→                env["SYSTEMROOT"] = os.environ.get("SYSTEMROOT", r"C:\Windows")
   123→            if "COMSPEC" not in env:
   124→                env["COMSPEC"] = os.environ.get("COMSPEC", r"C:\Windows\System32\cmd.exe")
   125→
   126→        try:
   127→            proc = await asyncio.create_subprocess_shell(
   128→                args.command,
   129→                stdout=asyncio.subprocess.PIPE,
   130→                stderr=asyncio.subprocess.PIPE,
   131→                cwd=args.cwd,
   132→                env=env,
   133→            )
   134→            stdout, stderr = await asyncio.wait_for(proc.communicate(), timeout=args.timeout_s)
   135→            return ToolResult(
   136→                tool="shell",
   137→                success=proc.returncode == 0,
   138→                output={
   139→                    "stdout": stdout.decode("utf-8", errors="replace"),
   140→                    "stderr": stderr.decode("utf-8", errors="replace"),
   141→                    "exit_code": proc.returncode,
   142→                    "mode": "local-dev",
   143→                },
   144→                error=None if proc.returncode == 0 else f"Exit code: {proc.returncode}",
   145→            )
   146→        except TimeoutError:
   147→            return ToolResult.fail("shell", f"Timeout after {args.timeout_s}s")
   148→        except Exception as e:
   149→            return ToolResult.fail("shell", str(e))
   150→
   151→    async def _execute_in_sandbox(self, args: ShellTool.Args, ctx: ToolContext) -> ToolResult:
   152→        """Docker-hardened execution (Tier 1 sandbox)."""
   153→        docker_cmd = [
   154→            "docker",
   155→            "run",
   156→            "--rm",
   157→            "--security-opt=no-new-privileges",
   158→            "--cap-drop=ALL",
   159→            "--network=none",
   160→            "--read-only",
   161→            "--tmpfs",
   162→            "/workspace:size=100M",
   163→            "-w",
   164→            "/workspace",
   165→        ]
   166→        for key, value in args.env.items():
   167→            if _looks_like_secret(key):
   168→                continue
   169→            if key in ("PATH", "LD_PRELOAD", "LD_LIBRARY_PATH", "PYTHONPATH"):
   170→                continue
   171→            docker_cmd.extend(["-e", f"{key}={value}"])
   172→
   173→        docker_cmd.extend(
   174→            [ctx.sandbox_container or "arnes-sandbox:latest", "sh", "-c", args.command]
   175→        )
   176→
   177→        try:
   178→            proc = await asyncio.create_subprocess_exec(
   179→                *docker_cmd,
   180→                stdout=asyncio.subprocess.PIPE,
   181→                stderr=asyncio.subprocess.PIPE,
   182→            )
   183→            stdout, stderr = await asyncio.wait_for(proc.communicate(), timeout=args.timeout_s)
   184→            return ToolResult(
   185→                tool="shell",
   186→                success=proc.returncode == 0,
   187→                output={
   188→                    "stdout": stdout.decode("utf-8", errors="replace"),
   189→                    "stderr": stderr.decode("utf-8", errors="replace"),
   190→                    "exit_code": proc.returncode,
   191→                    "sandbox": "docker-tier1",
   192→                },
   193→                error=None if proc.returncode == 0 else f"Exit code: {proc.returncode}",
   194→            )
   195→        except FileNotFoundError:
   196→            return ToolResult.fail(
   197→                "shell", "Docker not available. Set ARNES_DEV_MODE=1 for local execution."
   198→            )
   199→        except TimeoutError:
   200→            return ToolResult.fail("shell", f"Sandbox timeout after {args.timeout_s}s")

   203→# ============================================================
   204→# HTTP tool — calls external APIs with full SSRF protection
   205→# ============================================================
   206→
   207→
   208→class HttpTool(Tool):
   209→    """Make HTTP requests with full SSRF protection.
   210→
   211→    Security measures:
   212→    - Blocks localhost, private IPs, link-local, multicast by default
   213→    - Performs DNS resolution and validates ALL resolved IPs
   214→    - Blocks cloud metadata endpoints (AWS, GCP, Azure)
   215→    - **DNS-rebinding mitigation**: the resolved IP is pinned for the actual
   216→      httpx request by rewriting the URL to use the IP directly, setting the
   217→      ``Host`` header to the original hostname, and (for HTTPS) setting
   218→      ``sni_hostname`` in the request extensions. Redirects are NOT followed
   219→      by default — a ``Location`` header could point to a brand-new hostname
   220→      and re-trigger DNS resolution.
   221→    """
   222→
   223→    name: ClassVar[str] = "http"
   224→    description: ClassVar[str] = (
   225→        "Make an HTTP request. Returns status, headers, body. "
   226→        "SSRF-protected: blocks localhost, private IPs, cloud metadata endpoints."
   227→    )
   228→
   229→    class Args(BaseModel):
   230→        url: str
   231→        method: str = Field(default="GET", pattern="^(GET|POST|PUT|DELETE|PATCH|HEAD)$")
   232→        headers: dict[str, str] = Field(default_factory=dict)
   233→        body: str | None = None
   234→        timeout_s: float = Field(default=30.0, ge=1.0, le=120.0)
   235→
   236→    async def execute(self, args: dict[str, Any], ctx: ToolContext) -> ToolResult:
   237→        start = time.monotonic()
   238→        try:
   239→            validated = self.Args.model_validate(args)
   240→        except Exception as e:
   241→            return ToolResult.fail("http", f"Invalid args: {e}")
   242→
   243→        # Full SSRF check with DNS resolution. Returns the resolved IP so we
   244→        # can pin it for the request (DNS-rebinding TOCTOU mitigation).
   245→        ssrf_error, resolved_ip = await _check_ssrf_async(validated.url)
   246→        if ssrf_error:
   247→            return ToolResult.fail("http", ssrf_error, duration_s=time.monotonic() - start)
   248→
   249→        # If secret broker is set, inject secrets JIT (never in LLM context)
   250→        headers = dict(validated.headers)
   251→        if ctx.secret_broker:
   252→            headers = ctx.secret_broker.inject_secrets(headers, ctx)
   253→
   254→        # Rewrite the URL to use the resolved IP directly. This prevents
   255→        # httpx from re-resolving DNS (which would re-open the DNS-rebinding
   256→        # TOCTOU window that _check_ssrf_async just closed). The original
   257→        # hostname is preserved as the Host header (HTTP) and as SNI (HTTPS).
   258→        try:
   259→            pinned_url, original_host, scheme = _build_ip_pinned_url(validated.url, resolved_ip)
   260→        except ValueError as e:
   261→            return ToolResult.fail("http", str(e), duration_s=time.monotonic() - start)
   262→
   263→        # The Host header must match the original hostname for virtual-host
   264→        # routing to work. httpx lets us override it explicitly.
   265→        headers.setdefault("Host", original_host)
   266→
   267→        try:
   268→            async with httpx.AsyncClient(
   269→                timeout=validated.timeout_s,
   270→                follow_redirects=False,  # redirects would re-trigger DNS
   271→            ) as client:
   272→                request = client.build_request(
   273→                    method=validated.method,
   274→                    url=pinned_url,
   275→                    headers=headers,
   276→                    content=validated.body,
   277→                )
   278→                # For HTTPS, set SNI to the original hostname so the TLS
   279→                # handshake presents the correct cert and validates against
   280→                # the original hostname (not the IP). The httpcore backend
   281→                # expects ``sni_hostname`` to be a ``str``.
   282→                if scheme == "https":
   283→                    request.extensions["sni_hostname"] = original_host
   284→                response = await client.send(request)
   285→            return ToolResult.ok(
   286→                "http",
   287→                {
   288→                    "status_code": response.status_code,
   289→                    "headers": dict(response.headers),
   290→                    "body": response.text[:10000],  # Truncate to prevent context bloat
   291→                },
   292→                duration_s=time.monotonic() - start,
   293→                url=validated.url,
   294→            )
   295→        except httpx.HTTPError as e:
   296→            return ToolResult.fail("http", str(e), duration_s=time.monotonic() - start)
   297→

   300→# Filesystem tools — path-validated with symlink protection
   301→# ============================================================
   302→
   303→
   304→class FilesystemReadTool(Tool):
   305→    """Read a file with path traversal AND symlink protection."""
   306→
   307→    name: ClassVar[str] = "fs_read"
   308→    description: ClassVar[str] = "Read a file. Path must be inside working_dir."
   309→
   310→    class Args(BaseModel):
   311→        path: str
   312→        max_bytes: int = Field(default=65536, ge=1, le=1048576)
   313→
   314→    async def execute(self, args: dict[str, Any], ctx: ToolContext) -> ToolResult:
   315→        try:
   316→            validated = self.Args.model_validate(args)
   317→        except Exception as e:
   318→            return ToolResult.fail("fs_read", f"Invalid args: {e}")
   319→
   320→        safe_path = _validate_path(validated.path, ctx.working_dir)
   321→        if not safe_path:
   322→            return ToolResult.fail("fs_read", f"Path outside working_dir: {validated.path}")
   323→
   324→        # Symlink protection: don't follow symlinks outside working_dir
   325→        if safe_path.is_symlink():
   326→            target = safe_path.resolve()
   327→            base = Path(ctx.working_dir).resolve()
   328→            if base not in target.parents and target != base:
   329→                return ToolResult.fail("fs_read", f"Symlink escapes working_dir: {validated.path}")
   330→
   331→        try:
   332→            content = safe_path.read_bytes()[: validated.max_bytes]
   333→            return ToolResult.ok(
   334→                "fs_read",
   335→                {
   336→                    "path": str(safe_path),
   337→                    "content": content.decode("utf-8", errors="replace"),
   338→                    "size_bytes": len(content),
   339→                },
   340→            )
   341→        except FileNotFoundError:
   342→            return ToolResult.fail("fs_read", f"File not found: {validated.path}")
   343→        except PermissionError:
   344→            return ToolResult.fail("fs_read", f"Permission denied: {validated.path}")
   345→
   346→
   347→class FilesystemWriteTool(Tool):
   348→    """Write a file with path traversal protection. Requires approval."""
   349→
   350→    name: ClassVar[str] = "fs_write"
   351→    description: ClassVar[str] = "Write a file. Path must be inside working_dir. Requires approval."
   352→    requires_approval: ClassVar[bool] = True
   353→
   354→    class Args(BaseModel):
   355→        path: str
   356→        content: str
   357→        mode: str = Field(default="w", pattern="^(w|a)$")
   358→
   359→    async def execute(self, args: dict[str, Any], ctx: ToolContext) -> ToolResult:
   360→        try:
   361→            validated = self.Args.model_validate(args)
   362→        except Exception as e:
   363→            return ToolResult.fail("fs_write", f"Invalid args: {e}")
   364→
   365→        safe_path = _validate_path(validated.path, ctx.working_dir)
   366→        if not safe_path:
   367→            return ToolResult.fail("fs_write", f"Path outside working_dir: {validated.path}")
   368→
   369→        # Symlink protection for write.
   370→        #
   371→        # SECURITY (FIX-R3-SEC): use ``is_symlink()`` ALONE, NOT
   372→        # ``exists() and is_symlink()``. ``Path.exists()`` follows the
   373→        # link and returns False for a DANGLING symlink (target missing),
   374→        # which previously caused this guard to be skipped — letting a
   375→        # write through a dangling symlink that points outside
   376→        # ``working_dir`` (e.g. ``link -> /etc/cron.d/evil`` with the
   377→        # target not yet created). ``is_symlink()`` checks the link
   378→        # entry itself, so it catches both dangling and non-dangling
   379→        # symlinks.
   380→        if safe_path.is_symlink():
   381→            target = safe_path.resolve()
   382→            base = Path(ctx.working_dir).resolve()
   383→            if base not in target.parents and target != base:
   384→                return ToolResult.fail("fs_write", f"Symlink escapes working_dir: {validated.path}")
   385→
   386→        try:
   387→            safe_path.parent.mkdir(parents=True, exist_ok=True)
   388→            with safe_path.open(validated.mode) as f:
   389→                f.write(validated.content)
   390→            return ToolResult.ok(
   391→                "fs_write",
   392→                {"path": str(safe_path), "bytes_written": len(validated.content)},
   393→            )
   394→        except PermissionError:
   395→            return ToolResult.fail("fs_write", f"Permission denied: {validated.path}")

   454→_DANGEROUS_PATTERNS = [
   455→    r"\brm\s+-rf\s+/",
   456→    r"\bmkfs\b",
   457→    r"\bdd\s+if=",
   458→    r":\(\)\s*\{",  # fork bomb
   459→    r"\b>\s*/dev/sd[a-z]",
   460→    r"\bchmod\s+-R\s+777\s+/",
   461→    r"\bcurl\s+.*\|\s*sh",
   462→    r"\bwget\s+.*\|\s*sh",
   463→    r"\bnc\s+-l",  # reverse shell
   464→    r"\b/dev/tcp/",  # bash reverse shell
   465→    r"\bnohup\s+.*&\s*$",  # daemon escape
   466→    # --- Added in v0.1.x security hardening (defense-in-depth) ---
   467→    r"\bpython\s+-c\b",  # arbitrary code execution via -c
   468→    r"\bpython3\s+-c\b",  # arbitrary code execution via -c
   469→    r"\beval\s*\(",  # JS/Python eval() invocation
   470→    r"\bexec\s*\(",  # Python exec() invocation
   471→    r"\bfind\s+.*-delete\b",  # find ... -delete
   472→    r"\bbase64\s+-d\b",  # base64 decode (payload decoding)
   473→    r"\bbase64\s+--decode\b",  # base64 decode long form
   474→    r"\b\s{2,}.*&&",  # suspiciously-indented chained commands
   475→]

   477→
   478→def _is_dangerous_command(cmd: str) -> bool:
   479→    """Basic dangerous command detection.
   480→
   481→    .. warning::
   482→        This regex is **defense-in-depth only**. It catches common careless
   483→        payloads but is trivially bypassable by an adversarial prompt
   484→        (obfuscation, env-var expansion, heredocs, aliasing, etc.). It is
   485→        NOT a substitute for running shell commands inside a real sandbox
   486→        (Docker / nsjail / gVisor). For untrusted input, ALWAYS configure
   487→        ``ctx.sandbox_container``.
   488→    """
   489→    return any(re.search(p, cmd, re.IGNORECASE) for p in _DANGEROUS_PATTERNS)
   490→
   491→
   492→def _looks_like_secret(key: str) -> bool:
   493→    """Heuristic: does this env var name suggest it's a secret?"""
   494→    secret_patterns = [
   495→        "API_KEY",
   496→        "SECRET",
   497→        "TOKEN",
   498→        "PASSWORD",
   499→        "PASSWD",
   500→        "CREDENTIAL",
   501→        "PRIVATE_KEY",
   502→    ]
   503→    return any(p in key.upper() for p in secret_patterns)
   504→
   505→
   506→def _validate_path(path: str, working_dir: str) -> Path | None:
   507→    """Validate that path is inside working_dir (path traversal protection).
   508→
   509→    Uses Path.resolve() to canonicalize the path, then checks containment.
   510→    """
   511→    try:
   512→        base = Path(working_dir).resolve(strict=False)
   513→        target = (base / path).resolve(strict=False)
   514→        # Ensure target is inside base
   515→        try:
   516→            target.relative_to(base)
   517→        except ValueError:
   518→            return None
   519→        return target
   520→    except (ValueError, OSError):
   521→        return None

   540→
   541→async def _check_ssrf_async(url: str) -> tuple[str | None, str | None]:
   542→    """Full SSRF check with DNS resolution.
   543→
   544→    Returns ``(error_message, resolved_ip)``:
   545→    - On success: ``(None, ip_str)`` — ``ip_str`` is the resolved IP that
   546→      the caller should pin into the actual HTTP request to defeat
   547→      DNS-rebinding TOCTOU attacks.
   548→    - On failure: ``(error_message, None)``.
   549→
   550→    The caller is responsible for using ``resolved_ip`` when issuing the
   551→    request (see ``_build_ip_pinned_url``). Returning the IP here means
   552→    the SSRF check and the request share the same DNS resolution result.
   553→    """
   554→    try:
   555→        parsed = urllib.parse.urlparse(url)
   556→    except ValueError:
   557→        return f"Invalid URL: {url}", None
   558→
   559→    if parsed.scheme not in ("http", "https"):
   560→        return f"Blocked scheme: {parsed.scheme}", None
   561→
   562→    if not parsed.hostname:
   563→        return "No hostname in URL", None
   564→
   565→    hostname = parsed.hostname.lower()
   566→
   567→    # Block obvious internal hostnames
   568→    if hostname in _BLOCKED_HOSTS:
   569→        return f"Blocked internal host: {hostname}", None
   570→
   571→    # Try to parse as IP first — if the URL already contains an IP, we
   572→    # validate it but don't need to resolve DNS. The IP itself is the
   573→    # pinned IP.
   574→    try:
   575→        ip = ipaddress.ip_address(hostname)
   576→        if _is_blocked_ip(ip):
   577→            return f"Blocked private/loopback IP: {hostname}", None
   578→        return None, str(ip)
   579→    except ValueError:
   580→        pass
   581→
   582→    # It's a hostname — resolve DNS and validate ALL resolved IPs.
   583→    # We also pick the FIRST safe IP to pin for the request.
   584→    try:
   585→        infos = await asyncio.to_thread(socket.getaddrinfo, hostname, None)
   586→    except socket.gaierror:
   587→        return f"DNS resolution failed for: {hostname}", None
   588→
   589→    safe_ip: str | None = None
   590→    for _, _, _, _, sockaddr in infos:
   591→        ip_str = sockaddr[0]
   592→        try:
   593→            ip = ipaddress.ip_address(ip_str)
   594→        except ValueError:
   595→            continue
   596→        if _is_blocked_ip(ip):
   597→            return f"Blocked: {hostname} resolves to private IP {ip_str}", None
   598→        if safe_ip is None:
   599→            safe_ip = str(ip_str)
   600→
   601→    if safe_ip is None:
   602→        return f"No usable IPs resolved for: {hostname}", None
   603→    return None, safe_ip
   604→

   606→def _build_ip_pinned_url(url: str, resolved_ip: str | None) -> tuple[str, str, str]:
   607→    """Rewrite ``url`` to use ``resolved_ip`` as the host.
   608→
   609→    Returns ``(pinned_url, original_hostname, scheme)``.
   610→
   611→    - The path / query / fragment / port / scheme are preserved.
   612→    - The hostname is replaced by the resolved IP (IPv6 is bracketed per
   613→      RFC 3986 so httpx parses it correctly).
   614→    - The original hostname is returned separately so the caller can set
   615→      the ``Host`` header (HTTP) and ``sni_hostname`` extension (HTTPS)
   616→      for TLS cert validation + virtual-host routing.
   617→
   618→    Raises ``ValueError`` if the URL cannot be parsed or no IP is given.
   619→    """
   620→    if resolved_ip is None:
   621→        raise ValueError("No resolved IP available to pin")
   622→    parsed = urllib.parse.urlparse(url)
   623→    if not parsed.hostname:
   624→        raise ValueError("URL has no hostname")
   625→    scheme = parsed.scheme.lower()
   626→    hostname = parsed.hostname.lower()
   627→
   628→    # Bracket IPv6 literals per RFC 3986.
   629→    ip_for_url = resolved_ip
   630→    if ":" in resolved_ip and not resolved_ip.startswith("["):
   631→        ip_for_url = f"[{resolved_ip}]"
   632→
   633→    # Reconstruct netloc: [user[:pass]@]host[:port]
   634→    netloc = ip_for_url
   635→    if parsed.port is not None:
   636→        netloc = f"{ip_for_url}:{parsed.port}"
   637→    if parsed.username:
   638→        userinfo = parsed.username
   639→        if parsed.password:
   640→            userinfo = f"{userinfo}:{parsed.password}"
   641→        netloc = f"{userinfo}@{netloc}"
   642→
   643→    pinned = urllib.parse.urlunparse(
   644→        (
   645→            parsed.scheme,
   646→            netloc,
   647→            parsed.path or "/",
   648→            parsed.params,
   649→            parsed.query,
   650→            parsed.fragment,
   651→        )
   652→    )
   653→    return pinned, hostname, scheme
   654→

   656→def _is_blocked_ip(ip: ipaddress.IPv4Address | ipaddress.IPv6Address) -> bool:
   657→    """Check if an IP should be blocked for SSRF protection."""
   658→    if str(ip) in _BLOCKED_IPS:
   659→        return True
   660→    return (
   661→        ip.is_private
   662→        or ip.is_loopback
   663→        or ip.is_link_local
   664→        or ip.is_multicast
   665→        or ip.is_reserved
   666→        or ip.is_unspecified
   667→    )

   671→def _check_ssrf(url: str) -> str | None:
   672→    """Sync SSRF check (basic, no DNS resolution). Use _check_ssrf_async for full protection."""
   673→    try:
   674→        parsed = urllib.parse.urlparse(url)
   675→    except ValueError:
   676→        return f"Invalid URL: {url}"
   677→
   678→    if parsed.scheme not in ("http", "https"):
   679→        return f"Blocked scheme: {parsed.scheme}"
   680→
   681→    if not parsed.hostname:
   682→        return "No hostname in URL"
   683→
   684→    hostname = parsed.hostname.lower()
   685→    if hostname in _BLOCKED_HOSTS:
   686→        return f"Blocked internal host: {hostname}"
   687→
   688→    if hostname in _BLOCKED_IPS:
   689→        return f"Blocked metadata endpoint: {hostname}"
   690→
   691→    try:
   692→        ip = ipaddress.ip_address(hostname)
   693→        if _is_blocked_ip(ip):
   694→            return f"Blocked private/loopback IP: {hostname}"
   695→    except ValueError:
   696→        pass
   697→
   698→    return None
