    34→def _call_tool_response(result: dict[str, Any]) -> dict[str, Any]:
    35→    """Extract the JSON payload from a tools/call response.
    36→
    37→    ``handle_request`` wraps every tool result as ``{"content": [{"text": ...}]}``
    38→    with ``text`` being a JSON-serialized string. This helper deserializes it
    39→    so tests can assert on the structured payload.
    40→    """
    41→    return json.loads(result["result"]["content"][0]["text"])
