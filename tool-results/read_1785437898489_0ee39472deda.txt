   147→# Helpers — streaming provider with configurable cost

   150→
   151→class _StreamingCostProvider(LLMProvider):
   152→    """Provider that streams token chunks and a final usage chunk.
   153→
   154→    Used to test ``CostGuard.stream_complete`` cost tracking — the final
   155→    chunk carries the full ``LLMUsage`` with ``cost_usd > 0``.
   156→    """
   157→
   158→    def __init__(
   159→        self,
   160→        *,
   161→        tokens: list[str],
   162→        tokens_in: int,
   163→        tokens_out: int,
   164→        cost_usd: float,
   165→    ) -> None:
   166→        self.tokens = tokens
   167→        self.tokens_in = tokens_in
   168→        self.tokens_out = tokens_out
   169→        self.cost_usd = cost_usd
   170→
   171→    async def complete(
   172→        self,
   173→        messages: list[LLMMessage],
   174→        *,
   175→        model: str,
   176→        tools: list[dict[str, Any]] | None = None,
   177→        response_format: dict[str, Any] | None = None,
   178→        response_schema: dict[str, Any] | None = None,
   179→        **kwargs: Any,
   180→    ) -> LLMResponse:
   181→        return LLMResponse(
   182→            content="".join(self.tokens),
   183→            tool_calls=[],
   184→            usage=LLMUsage(
   185→                tokens_in=self.tokens_in,
   186→                tokens_out=self.tokens_out,
   187→                cost_usd=self.cost_usd,
   188→                model=model,
   189→                cached=False,
   190→            ),
   191→            model=model,
   192→        )
   193→
   194→    async def stream_complete(
   195→        self,
   196→        messages: list[LLMMessage],
   197→        *,
   198→        model: str,
   199→        tools: list[dict[str, Any]] | None = None,
   200→        temperature: float = 0.0,
   201→        max_tokens: int | None = None,
   202→        response_format: dict[str, Any] | None = None,
   203→        response_schema: dict[str, Any] | None = None,
   204→        **kwargs: Any,
   205→    ) -> AsyncIterator[LLMResponse]:
   206→        # Yield token chunks (zero usage — intermediate chunks carry no stats)
   207→        for token in self.tokens:
   208→            yield LLMResponse(
   209→                content=token,
   210→                tool_calls=[],
   211→                usage=LLMUsage(
   212→                    tokens_in=0,
   213→                    tokens_out=0,
   214→                    cost_usd=0.0,
   215→                    model=model,
   216→                    cached=False,
   217→                ),
   218→                model=model,
   219→            )
   220→        # Yield final usage chunk (empty content, full usage)
   221→        yield LLMResponse(
   222→            content="",
   223→            tool_calls=[],
   224→            usage=LLMUsage(
   225→                tokens_in=self.tokens_in,
   226→                tokens_out=self.tokens_out,
   227→                cost_usd=self.cost_usd,
   228→                model=model,
   229→                cached=False,
   230→            ),
   231→            model=model,
   232→        )
   233→
   234→    def list_models(self) -> list[str]:
   235→        return ["mock"]
