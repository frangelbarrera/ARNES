     1→     1→"""
     2→     2→ARNES Playbook Executor — runs a compiled Playbook as a DAG.
     3→     3→
     4→     4→The executor:
     5→     5→1. Walks the playbook steps in order.
     6→     6→2. For each step, invokes the specialist or tool.
     7→     7→3. Applies conditional branches (if/elif/else).
     8→     8→4. Runs parallel branches concurrently.
     9→     9→5. Handles retries with backoff.
    10→    10→6. Pauses at HITL gates.
    11→    11→7. Tracks budget via CostGuard.
    12→    12→8. Appends events to the Thread.
    13→    13→9. Returns a PlaybookRunResult with full trace.
    14→    14→
    15→    15→The executor is async and supports both fire-and-forget and streaming modes.
    16→    16→"""
    17→    17→
    18→    18→from __future__ import annotations
    19→    19→
    20→    20→import re

    22→    22→from typing import Any
    23→    23→

    30→    30→from arnes.playbooks.schema import ConditionalBranch, Playbook, PlaybookStep
    31→    31→from arnes.specialists.base import SpecialistRegistry, get_default_specialist_registry
    32→    32→from arnes.thread import Thread
    33→    33→from arnes.thread.events import (
    34→    34→    ConditionalBranchEvent,

    58→    58→    total_tokens_out: int = 0
    59→    59→    total_cost_usd: float = 0.0

    68→    68→class PlaybookExecutor:
    69→    69→    """Executes a compiled Playbook.
    70→    70→
    71→    71→    Usage:
    72→    72→        playbook = PlaybookCompiler.from_file("manuals/audit-pr.yaml")
    73→    73→        executor = PlaybookExecutor()
    74→    74→        result = await executor.run(playbook)
    75→    75→        print(result.to_markdown())
    76→    76→    """
    77→    77→
    78→    78→    def __init__(
    79→    79→        self,
    80→    80→        *,
    81→    81→        provider: LLMProvider | None = None,
    82→    82→        specialist_registry: SpecialistRegistry | None = None,
    83→    83→        tool_registry: ToolRegistry | None = None,
    84→    84→        cost_budget: CostBudget | None = None,
    85→    85→        interactive: bool = False,
    86→    86→    ) -> None:
    87→    87→        self.provider = provider or get_provider()
    88→    88→        self.specialist_registry = specialist_registry or get_default_specialist_registry()
    89→    89→        self.tool_registry = tool_registry or get_default_registry()
    90→    90→        self.cost_budget = cost_budget or CostBudget()
    91→    91→        self.interactive = interactive
    92→    92→
    93→    93→    async def run(
    94→    94→        self,
    95→    95→        playbook: Playbook,
    96→    96→        *,
    97→    97→        initial_input: dict[str, Any] | None = None,
    98→    98→    ) -> PlaybookRunResult:
    99→    99→        """Execute a playbook. Returns a PlaybookRunResult."""
   100→   100→        # Use a mutable list to hold the thread (so step helpers can append)
   101→   101→        thread_holder: list[Thread] = [Thread.create()]
   102→   102→        start_time = time.monotonic()
   103→   103→        outputs: dict[str, Any] = dict(playbook.variables)
   104→   104→        if initial_input:
   105→   105→            outputs.update(initial_input)
   106→   106→
   107→   107→        # Track cost guard at the run level (shared across steps)
   108→   108→        cost_guard = CostGuard(self.provider, budget=self.cost_budget)
   109→   109→
   110→   110→        steps_executed = 0
   111→   111→        steps_failed = 0
   112→   112→        total_tokens_in = 0
   113→   113→        total_tokens_out = 0
   114→   114→        total_cost_usd = 0.0
   115→   115→        aborted = False
   116→   116→        abort_error: str | None = None
   117→   117→
   118→   118→        try:
   119→   119→            for step in playbook.steps:
   120→   120→                # Check if step should be skipped due to prior saltar_a
   121→   121→                skip_until = outputs.get("__skip_steps_until", {})
   122→   122→                if skip_until:
   123→   123→                    # If we've reached the target step, clear the skip marker
   124→   124→                    if step.id in skip_until:
   125→   125→                        del skip_until[step.id]
   126→   126→                        if not skip_until:
   127→   127→                            del outputs["__skip_steps_until"]
   128→   128→                        logger.info("saltar_a_reached", step_id=step.id)
   129→   129→                        # Don't skip this step — execute it
   130→   130→                    else:
   131→   131→                        # Skip this step
   132→   132→                        logger.info("step_skipped", step_id=step.id, reason="saltar_a")
   133→   133→                        continue
   134→   134→
   135→   135→                # Execute the step
   136→   136→                step_result = await self._execute_step(
   137→   137→                    step,
   138→   138→                    thread_holder,
   139→   139→                    outputs,
   140→   140→                    cost_guard,
   141→   141→                    playbook,
   142→   142→                )
   143→   143→
   144→   144→                if step_result["success"]:
   145→   145→                    steps_executed += 1
   146→   146→                    outputs[step.id] = step_result.get("output")
   147→   147→                    # Track usage
   148→   148→                    usage = step_result.get("usage", {})
   149→   149→                    total_tokens_in += usage.get("tokens_in", 0)
   150→   150→                    total_tokens_out += usage.get("tokens_out", 0)
   151→   151→                    total_cost_usd += usage.get("cost_usd", 0.0)
   152→   152→                else:
   153→   153→                    steps_failed += 1
   154→   154→                    # Check if step has si_no_se_cumple fallback
   155→   155→                    if step.if_not_met:
   156→   156→                        branch_result = await self._handle_conditional_branch(
   157→   157→                            step,
   158→   158→                            step.if_not_met,
   159→   159→                            thread_holder,
   160→   160→                            outputs,
   161→   161→                            cost_guard,
   162→   162→                            playbook,
   163→   163→                        )
   164→   164→                        if branch_result.get("terminate"):
   165→   165→                            logger.info(
   166→   166→                                "run_terminated_by_conditional",
   167→   167→                                step_id=step.id,
   168→   168→                                termination=branch_result["terminate"],
   169→   169→                            )
   170→   170→                            break
   171→   171→                    else:
   172→   172→                        # No fallback — abort
   173→   173→                        thread_holder[0] = thread_holder[0].append(
   174→   174→                            RunFailedEvent(
   175→   175→                                thread_id=thread_holder[0].id,
   176→   176→                                step_id=step.id,
   177→   177→                                data={
   178→   178→                                    "error": step_result.get("error", "Unknown error"),
   179→   179→                                    "recoverable": False,
   180→   180→                                },
   181→   181→                            )
   182→   182→                        )
   183→   183→                        aborted = True
   184→   184→                        abort_error = step_result.get("error")
   185→   185→                        break
   186→   186→
   187→   187→            # Run completed (or aborted)
   188→   188→            if not aborted:
   189→   189→                thread_holder[0] = thread_holder[0].append(
   190→   190→                    RunCompletedEvent(
   191→   191→                        thread_id=thread_holder[0].id,
   192→   192→                        data={
   193→   193→                            "steps_executed": steps_executed,
   194→   194→                            "duration_s": time.monotonic() - start_time,
   195→   195→                            "total_tokens": total_tokens_in + total_tokens_out,
   196→   196→                            "total_cost_usd": total_cost_usd,
   197→   197→                        },
   198→   198→                    )
   199→   199→                )
   200→   200→
   201→   201→            return PlaybookRunResult(
   202→   202→                thread=thread_holder[0],
   203→   203→                success=not aborted,
   204→   204→                steps_executed=steps_executed,
   205→   205→                steps_failed=steps_failed,
   206→   206→                duration_s=time.monotonic() - start_time,
   207→   207→                total_tokens_in=total_tokens_in,
   208→   208→                total_tokens_out=total_tokens_out,
   209→   209→                total_cost_usd=total_cost_usd,
   210→   210→                outputs=outputs,
   211→   211→                error=abort_error,
   212→   212→            )
   213→   213→
   214→   214→        except BudgetExceeded as e:
   215→   215→            logger.error("budget_exceeded", error=str(e), spent=e.spent, budget=e.budget)
   216→   216→            thread_holder[0] = thread_holder[0].append(
   217→   217→                RunFailedEvent(
   218→   218→                    thread_id=thread_holder[0].id,
   219→   219→                    data={
   220→   220→                        "error": f"Budget exceeded: {e}",
   221→   221→                        "spent_usd": e.spent,
   222→   222→                        "budget_usd": e.budget,
   223→   223→                        "level": e.level,
   224→   224→                    },
   225→   225→                )
   226→   226→            )
   227→   227→            return PlaybookRunResult(
   228→   228→                thread=thread_holder[0],
   229→   229→                success=False,
   230→   230→                steps_executed=steps_executed,
   231→   231→                steps_failed=steps_failed,
   232→   232→                duration_s=time.monotonic() - start_time,
   233→   233→                total_tokens_in=total_tokens_in,
   234→   234→                total_tokens_out=total_tokens_out,
   235→   235→                total_cost_usd=total_cost_usd,
   236→   236→                outputs=outputs,
   237→   237→                error=str(e),
   238→   238→            )
   239→   239→
   240→   240→    # ============================================================
   241→   241→    # Step execution
   242→   242→    # ============================================================
   243→   243→
   244→   244→    async def _execute_step(
   245→   245→        self,
   246→   246→        step: PlaybookStep,
   247→   247→        thread_holder: list[Thread],
   248→   248→        outputs: dict[str, Any],
   249→   249→        cost_guard: CostGuard,
   250→   250→        playbook: Playbook,
   251→   251→    ) -> dict[str, Any]:
   252→   252→        """Execute a single step (specialist, tool, or parallel branch)."""
   253→   253→        thread_holder[0] = thread_holder[0].append(
   254→   254→            StepStartedEvent(
   255→   255→                thread_id=thread_holder[0].id,
   256→   256→                step_id=step.id,
   257→   257→                specialist=step.specialist or step.tool,
   258→   258→                data={"step_id": step.id, "specialist": step.specialist or step.tool},
   259→   259→            )
   260→   260→        )
   261→   261→
   262→   262→        step_start = time.monotonic()
   263→   263→
   264→   264→        try:
   265→   265→            # Parallel branch
   266→   266→            if step.parallel:
   267→   267→                result = await self._execute_parallel(
   268→   268→                    step, thread_holder, outputs, cost_guard, playbook
   269→   269→                )
   270→   270→            # Specialist invocation
   271→   271→            elif step.specialist:
   272→   272→                result = await self._execute_specialist(
   273→   273→                    step, thread_holder, outputs, cost_guard, playbook
   274→   274→                )
   275→   275→            # Tool invocation
   276→   276→            elif step.tool:
   277→   277→                result = await self._execute_tool(step, thread_holder, outputs, playbook)
   278→   278→            else:
   279→   279→                raise ValueError(f"Step '{step.id}' has no action defined")
   280→   280→
   281→   281→            # Drain middleware event sink BEFORE recording completion so
   282→   282→            # that AssistantMessageEvent / CostThresholdEvent / CACHE_HIT /
   283→   283→            # REFUSAL_TRIGGERED events emitted during the step appear in
   284→   284→            # the thread before the StepCompletedEvent. The middleware
   285→   285→            # creates events with a nil thread_id placeholder (it does not
   286→   286→            # have access to the Thread); we patch the real thread_id and
   287→   287→            # step_id here.
   288→   288→            self._drain_middleware_events(thread_holder, cost_guard, step.id)
   289→   289→
   290→   290→            # Pull token / cost usage out of the step result so the
   291→   291→            # StepCompletedEvent carries the per-step aggregate. The
   292→   292→            # specialist returns ``usage`` as a model_dump() of LLMUsage
   293→   293→            # (tokens_in, tokens_out, cost_usd, model, cached).
   294→   294→            usage = result.get("usage") or {}
   295→   295→
   296→   296→            # Record completion (with token + cost accounting)
   297→   297→            thread_holder[0] = thread_holder[0].append(
   298→   298→                StepCompletedEvent(
   299→   299→                    thread_id=thread_holder[0].id,
   300→   300→                    step_id=step.id,
   301→   301→                    specialist=step.specialist or step.tool,
   302→   302→                    data={
   303→   303→                        "step_id": step.id,
   304→   304→                        "output": result.get("output"),
   305→   305→                        "duration_s": time.monotonic() - step_start,
   306→   306→                        "tokens_in": usage.get("tokens_in", 0),
   307→   307→                        "tokens_out": usage.get("tokens_out", 0),
   308→   308→                        "cost_usd": usage.get("cost_usd", 0.0),
   309→   309→                    },
   310→   310→                )
   311→   311→            )
   312→   312→
   313→   313→            return result
   314→   314→
   315→   315→        except Exception as e:
   316→   316→            logger.exception("step_failed", step_id=step.id, error=str(e))
   317→   317→            # Drain any events emitted before the failure too — they are
   318→   318→            # still useful for debugging (e.g. a CostThresholdEvent that
   319→   319→            # fired right before the crash, or an AssistantMessageEvent
   320→   320→            # from the failed LLM call).
   321→   321→            self._drain_middleware_events(thread_holder, cost_guard, step.id)
   322→   322→            thread_holder[0] = thread_holder[0].append(
   323→   323→                StepFailedEvent(
   324→   324→                    thread_id=thread_holder[0].id,
   325→   325→                    step_id=step.id,
   326→   326→                    specialist=step.specialist or step.tool,
   327→   327→                    data={"step_id": step.id, "error": str(e), "retry": False},
   328→   328→                )
   329→   329→            )
   330→   330→            return {"success": False, "error": str(e)}
   331→   331→
   332→   332→    def _drain_middleware_events(
   333→   333→        self,
   334→   334→        thread_holder: list[Thread],
   335→   335→        cost_guard: CostGuard,
   336→   336→        step_id: str,
   337→   337→    ) -> None:
   338→   338→        """Drain the middleware event sink and append events to the Thread.
   339→   339→
   340→   340→        Middleware (CostGuard, TokenOptimizer, VerificationLayer) emit
   341→   341→        events to a shared ``_events`` list because they do not have direct
   342→   342→        access to the Thread. The events are created with a nil thread_id
   343→   343→        placeholder; here we patch the real thread_id and step_id and
   344→   344→        append them to the Thread.
   345→   345→
   346→   346→        Idempotent: clears the sink after draining so the same events are
   347→   347→        not appended twice.
   348→   348→        """
   349→   349→        events = getattr(cost_guard, "_events", None)
   350→   350→        if not events:
   351→   351→            return
   352→   352→
   353→   353→        thread_id = thread_holder[0].id
   354→   354→        for event in events:
   355→   355→            # Events are frozen pydantic models; use model_copy(update=...)
   356→   356→            # to set the real thread_id and step_id without mutating the
   357→   357→            # original (which may be referenced by middleware state).
   358→   358→            patched = event.model_copy(
   359→   359→                update={"thread_id": thread_id, "step_id": event.step_id or step_id}
   360→   360→            )
   361→   361→            thread_holder[0] = thread_holder[0].append(patched)
   362→   362→
   363→   363→        events.clear()

   365→   365→    async def _execute_specialist(
   366→   366→        self,
   367→   367→        step: PlaybookStep,
   368→   368→        thread_holder: list[Thread],
   369→   369→        outputs: dict[str, Any],
   370→   370→        cost_guard: CostGuard,
   371→   371→        playbook: Playbook,
   372→   372→    ) -> dict[str, Any]:
   373→   373→        """Invoke a specialist."""
   374→   374→        specialist = self.specialist_registry.get(step.specialist or "")
   375→   375→        if not specialist:
   376→   376→            return {
   377→   377→                "success": False,
   378→   378→                "error": f"Specialist '{step.specialist}' not registered. Available: {self.specialist_registry.list_names()}",
   379→   379→            }
   380→   380→
   381→   381→        # Resolve input (may contain Jinja2-style template refs)
   382→   382→        input_data = self._resolve_input(step.input, outputs)
   383→   383→
   384→   384→        # Build tool context
   385→   385→        ctx = ToolContext(
   386→   386→            thread_id=thread_holder[0].id,
   387→   387→            step_id=step.id,
   388→   388→            specialist=step.specialist,
   389→   389→            working_dir=".",
   390→   390→            sandbox_enabled=False,  # Disabled for MVP; enable in v0.2
   391→   391→            budget_remaining_usd=(
   392→   392→                (cost_guard.budget.effective_budget() or 0) - cost_guard.spent_usd
   393→   393→            ),
   394→   394→            metadata={"interactive": self.interactive},
   395→   395→        )
   396→   396→
   397→   397→        # Use the cost_guard-wrapped provider
   398→   398→        result = await specialist.run(
   399→   399→            input_data,
   400→   400→            ctx,
   401→   401→            provider=cost_guard,
   402→   402→            tool_registry=self.tool_registry,
   403→   403→        )
   404→   404→
   405→   405→        success = result.get("success", False)
   406→   406→        if not success and result.get("budget_exceeded"):
   407→   407→            raise BudgetExceeded(
   408→   408→                f"Budget exceeded during specialist '{step.specialist}' invocation",
   409→   409→                spent=cost_guard.spent_usd,
   410→   410→                budget=cost_guard.budget.effective_budget() or 0.0,
   411→   411→                level="specialist",
   412→   412→            )
   413→   413→
   414→   414→        # Propagate error message for non-budget failures
   415→   415→        if not success and result.get("error"):
   416→   416→            return {
   417→   417→                "success": False,
   418→   418→                "error": result["error"],
   419→   419→                "output": None,
   420→   420→                "usage": result.get("usage", {}),
   421→   421→            }
   422→   422→
   423→   423→        return {
   424→   424→            "success": success,
   425→   425→            "output": result.get("output"),
   426→   426→            "usage": result.get("usage", {}),
   427→   427→        }
   428→   428→
   429→   429→    async def _execute_tool(
   430→   430→        self,
   431→   431→        step: PlaybookStep,
   432→   432→        thread_holder: list[Thread],
   433→   433→        outputs: dict[str, Any],
   434→   434→        playbook: Playbook,
   435→   435→    ) -> dict[str, Any]:
   436→   436→        """Invoke a tool."""
   437→   437→        tool = self.tool_registry.get(step.tool or "")
   438→   438→        if not tool:
   439→   439→            return {
   440→   440→                "success": False,
   441→   441→                "error": f"Tool '{step.tool}' not registered. Available: {self.tool_registry.list_names()}",
   442→   442→            }
   443→   443→
   444→   444→        input_data = self._resolve_input(step.input, outputs)
   445→   445→
   446→   446→        ctx = ToolContext(
   447→   447→            thread_id=thread_holder[0].id,
   448→   448→            step_id=step.id,
   449→   449→            metadata={"interactive": self.interactive},
   450→   450→        )
   451→   451→
   452→   452→        result = await tool.execute(input_data, ctx)
   453→   453→        return {
   454→   454→            "success": result.success,
   455→   455→            "output": result.output if result.success else None,
   456→   456→            "error": result.error,
   457→   457→        }
   458→   458→
   459→   459→    async def _execute_parallel(
   460→   460→        self,
   461→   461→        step: PlaybookStep,
   462→   462→        thread_holder: list[Thread],
   463→   463→        outputs: dict[str, Any],
   464→   464→        cost_guard: CostGuard,
   465→   465→        playbook: Playbook,
   466→   466→    ) -> dict[str, Any]:
   467→   467→        """Execute parallel sub-steps concurrently.
   468→   468→
   469→   469→        Note: parallel sub-steps share the same thread_holder, but since each
   470→   470→        appends to the immutable Thread, we need to merge results back.
   471→   471→        For MVP, parallel steps are executed sequentially (true parallelism
   472→   472→        requires a different state model — coming in v0.2).
   473→   473→        """
   474→   474→        if not step.parallel:
   475→   475→            return {"success": False, "error": "No parallel steps defined"}
   476→   476→
   477→   477→        outputs_map = {}
   478→   478→        all_success = True
   479→   479→
   480→   480→        # For MVP: sequential execution of "parallel" steps (correctness > parallelism)
   481→   481→        # In v0.2 we'll use asyncio.gather with proper thread merging
   482→   482→        for sub_step in step.parallel:
   483→   483→            result = await self._execute_step(
   484→   484→                sub_step, thread_holder, outputs, cost_guard, playbook
   485→   485→            )
   486→   486→            # Wrap output in {"output": ...} structure so templates like
   487→   487→            # {{ steps.parallel.sub_step_id.output }} resolve correctly
   488→   488→            outputs_map[sub_step.id] = {
   489→   489→                "output": result.get("output"),
   490→   490→                "success": result.get("success", False),
   491→   491→            }
   492→   492→            if not result.get("success", False):
   493→   493→                all_success = False
   494→   494→
   495→   495→        return {
   496→   496→            "success": all_success,
   497→   497→            "output": outputs_map,
   498→   498→        }
   499→   499→
   500→   500→    async def _handle_conditional_branch(
   501→   501→        self,
   502→   502→        step: PlaybookStep,
   503→   503→        branch: ConditionalBranch,
   504→   504→        thread_holder: list[Thread],
   505→   505→        outputs: dict[str, Any],
   506→   506→        cost_guard: CostGuard,
   507→   507→        playbook: Playbook,
   508→   508→    ) -> dict[str, Any]:
   509→   509→        """Handle an if_not_met or conditional branch."""
   510→   510→        thread_holder[0] = thread_holder[0].append(
   511→   511→            ConditionalBranchEvent(
   512→   512→                thread_id=thread_holder[0].id,
   513→   513→                step_id=step.id,
   514→   514→                data={
   515→   515→                    "condition": branch.when or "if_not_met",
   516→   516→                    "action": branch.action,
   517→   517→                },
   518→   518→            )
   519→   519→        )
   520→   520→
   521→   521→        if branch.action == "call" and branch.specialist:
   522→   522→            # Invoke the fallback specialist
   523→   523→            fallback_step = PlaybookStep(
   524→   524→                id=f"{step.id}__fallback",
   525→   525→                specialist=branch.specialist,
   526→   526→                input=branch.input or {},
   527→   527→            )
   528→   528→            result = await self._execute_specialist(
   529→   529→                fallback_step, thread_holder, outputs, cost_guard, playbook
   530→   530→            )
   531→   531→            outputs[fallback_step.id] = result.get("output")
   532→   532→            return {"terminate": None, "result": result}
   533→   533→
   534→   534→        if branch.action == "terminate":
   535→   535→            return {"terminate": branch.terminate}
   536→   536→
   537→   537→        if branch.action == "skip" and branch.skip_to:
   538→   538→            # skip_to means "jump TO this step" — skip all steps between
   539→   539→            # current and target
   540→   540→            skip_set = outputs.setdefault("__skip_steps_until", {})
   541→   541→            skip_set[branch.skip_to] = True  # Will be cleared when we reach it
   542→   542→            return {"terminate": None}
   543→   543→
   544→   544→        return {"terminate": None}

   549→   549→
   550→   550→    _TEMPLATE_RE = re.compile(r"\{\{\s*([^}]+)\s*\}\}")
   551→   551→
   552→   552→    def _resolve_input(
   553→   553→        self,
   554→   554→        input_value: dict[str, Any] | str | None,
   555→   555→        outputs: dict[str, Any],
   556→   556→    ) -> dict[str, Any]:
   557→   557→        """Resolve Jinja2-style template references in input.
   558→   558→
   559→   559→        Example: "{{ pasos.leer_diff.salida }}" → outputs["leer_diff"]["output"]
   560→   560→        Handles MULTIPLE templates in the same string.
   561→   561→        """
   562→   562→        if input_value is None:
   563→   563→            return {}
   564→   564→
   565→   565→        if isinstance(input_value, str):
   566→   566→            return {"__resolved_str__": self._resolve_template(input_value, outputs)}
   567→   567→
   568→   568→        if isinstance(input_value, dict):
   569→   569→            resolved = {}
   570→   570→            for k, v in input_value.items():
   571→   571→                if isinstance(v, str):
   572→   572→                    resolved[k] = self._resolve_template(v, outputs)
   573→   573→                elif isinstance(v, dict):
   574→   574→                    resolved[k] = self._resolve_input(v, outputs)
   575→   575→                elif isinstance(v, list):
   576→   576→                    resolved[k] = [
   577→   577→                        self._resolve_template(item, outputs)
   578→   578→                        if isinstance(item, str)
   579→   579→                        else self._resolve_input(item, outputs)
   580→   580→                        if isinstance(item, dict)
   581→   581→                        else item
   582→   582→                        for item in v
   583→   583→                    ]
   584→   584→                else:
   585→   585→                    resolved[k] = v
   586→   586→            return resolved
   587→   587→
   588→   588→        # Unreachable: ``input_value`` is typed as ``dict[str, Any] | str | None``
   589→   589→        # and all three branches above return early. Kept as a defensive
   590→   590→        # fallback for callers that bypass the type system (defence-in-depth).
   591→   591→        return {"__input__": input_value}  # type: ignore[unreachable]
   592→   592→
   593→   593→    def _resolve_template(self, template: str, outputs: dict[str, Any]) -> Any:
   594→   594→        """Resolve a template string, handling MULTIPLE {{ }} references.
   595→   595→
   596→   596→        Examples:
   597→   597→            "{{ pasos.X.salida }}" → outputs["X"]["output"]
   598→   598→            "Plan: {{ variables.nombre }} for PR {{ variables.pr_number }}" → "Plan: foo for PR 1234"
   599→   599→            "Diff: {{ pasos.leer_diff.salida }}, Sec: {{ pasos.auditoria.salida }}" → "Diff: ..., Sec: ..."
   600→   600→            "{{ }}" → "{{ }}"  (empty template body — returned as literal)
   601→   601→        """
   602→   602→        # Find ALL template references
   603→   603→        matches = list(self._TEMPLATE_RE.finditer(template))
   604→   604→
   605→   605→        if not matches:
   606→   606→            return template
   607→   607→
   608→   608→        # If the entire string is ONE template, return the resolved value (preserve type)
   609→   609→        if len(matches) == 1 and matches[0].group(0) == template:
   610→   610→            expr = matches[0].group(1).strip()
   611→   611→            if not expr:
   612→   612→                # Empty template body (e.g. "{{ }}") — return the original
   613→   613→                # literal verbatim rather than re-rendering it with different
   614→   614→                # whitespace.
   615→   615→                return template
   616→   616→            return self._resolve_expr(expr, outputs)
   617→   617→
   618→   618→        # Otherwise, interpolate ALL matches into the string
   619→   619→        result = template
   620→   620→        # Process in reverse order to keep indexes valid
   621→   621→        for match in reversed(matches):
   622→   622→            expr = match.group(1).strip()
   623→   623→            if not expr:
   624→   624→                # Empty template body — leave the original match untouched
   625→   625→                continue
   626→   626→            resolved = self._resolve_expr(expr, outputs)
   627→   627→            result = result[: match.start()] + str(resolved) + result[match.end() :]
   628→   628→
   629→   629→        return result
   630→   630→
   631→   631→    def _resolve_expr(self, expr: str, outputs: dict[str, Any]) -> Any:
   632→   632→        """Resolve a single template expression like 'steps.X.output'.
   633→   633→
   634→   634→        Supported forms:
   635→   635→            "steps.X.output"        → outputs["X"]["output"] (or outputs["X"] if raw)
   636→   636→            "steps.X.output.field"  → outputs["X"]["output"]["field"] (or outputs["X"]["field"])
   637→   637→            "variables.X"           → outputs["X"]
   638→   638→            "pasos.X.salida"        → legacy ES form of "steps.X.output"
   639→   639→
   640→   640→        Two important behaviors:
   641→   641→
   642→   642→        1. Only the LEADING prefix ("steps.", "variables.", "pasos.") is
   643→   643→           stripped. Interior occurrences of these substrings (e.g. when a
   644→   644→           step's output literally contains a key named "steps") are
   645→   645→           preserved. This makes deep nesting like
   646→   646→           ``{{ steps.s1.output.steps.s2.output }}`` work correctly.
   647→   647→
   648→   648→        2. For STEP references (``steps.*`` / legacy ``pasos.*``), the
   649→   649→           "output" (and legacy "salida") segment is treated as a VIRTUAL
   650→   650→           accessor: if the current dict has an "output" key, it is
   651→   651→           dereferenced; if not, the segment is skipped (the current dict
   652→   652→           IS the output). This makes ``{{ steps.X.output }}`` work whether
   653→   653→           the step's output was stored raw (``outputs["X"] = output_dict``)
   654→   654→           or wrapped (``outputs["X"] = {"output": output_dict, ...}``).
   655→   655→
   656→   656→           The virtual accessor does NOT apply to ``variables.*`` refs —
   657→   657→           variables are user-defined and a missing key is a real error.
   658→   658→        """
   659→   659→        # Strip leading prefix only — NOT all occurrences. Stripping all
   660→   660→        # occurrences of "steps." would corrupt paths whose intermediate
   661→   661→        # dicts literally contain a key named "steps" (deep nesting).
   662→   662→        stripped = expr.strip()
   663→   663→        is_step_ref = False
   664→   664→        for prefix in ("steps.", "pasos.", "variables."):
   665→   665→            if stripped.startswith(prefix):
   666→   666→                stripped = stripped[len(prefix) :]
   667→   667→                # Only steps/pasos get the virtual "output" accessor;
   668→   668→                # variables use strict key lookup.
   669→   669→                is_step_ref = prefix in ("steps.", "pasos.")
   670→   670→                break
   671→   671→
   672→   672→        parts = stripped.split(".")
   673→   673→
   674→   674→        current: Any = outputs
   675→   675→        for raw_part in parts:
   676→   676→            # Legacy ES translation per-segment
   677→   677→            part = "output" if raw_part == "salida" else raw_part
   678→   678→            if isinstance(current, dict):
   679→   679→                if part in current:
   680→   680→                    current = current[part]
   681→   681→                elif is_step_ref and part == "output":
   682→   682→                    # Virtual accessor: the current dict is itself the
   683→   683→                    # step's output (stored raw). Skip this segment so
   684→   684→                    # downstream segments (e.g. ".verdict") resolve against
   685→   685→                    # the output dict directly.
   686→   686→                    continue
   687→   687→                else:
   688→   688→                    return f"{{{{ {expr} }}}}"  # Leave template as-is if not found
   689→   689→            else:
   690→   690→                return f"{{{{ {expr} }}}}"
   691→   691→
   692→   692→        return current
