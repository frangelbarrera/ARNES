   204→async def _stream_specialist(specialist: str, task: str, model: str, mock: bool) -> None:
   205→    """Stream a specialist's response token-by-token and save a bitácora.
   206→
   207→    Uses :meth:`Harness.stream_with_audit` so the audit trail is recorded in
   208→    a real ``Thread`` (mutated in place as the stream is consumed) and the
   209→    bitácora is rendered via :meth:`Thread.to_markdown`. This keeps the
   210→    streaming CLI output consistent with the markdown produced by
   211→    ``arnes run`` / ``PlaybookRunResult.to_markdown`` — previously the
   212→    streaming path emitted a bespoke markdown format that diverged from
   213→    the rest of the bitácora system.
   214→    """
   215→    from arnes import Harness, HarnessConfig
   216→
   217→    if mock or model.startswith("mock/"):
   218→        provider: LLMProvider = _SchemaValidMockLLMProvider()
   219→    else:
   220→        provider = get_provider(model)
   221→
   222→    harness = Harness(
   223→        config=HarnessConfig(model=model, budget_usd=0.50),
   224→        provider=provider,
   225→    )
   226→
   227→    # Normalize specialist name
   228→    if not specialist.startswith("@"):
   229→        specialist = "@" + specialist
   230→
   231→    # Check specialist exists before streaming
   232→    specialist_obj = harness.specialist_registry.get(specialist)
   233→    if not specialist_obj:
   234→        available = harness.specialist_registry.list_names()
   235→        console.print(f"[red]✗[/red] Specialist '{specialist}' not found. Available: {available}")
   236→        sys.exit(1)
   237→
   238→    console.print(f"[cyan]Streaming[/cyan] {specialist}...")
   239→    console.print("[dim]---[/dim]")
   240→
   241→    # stream_with_audit returns (chunks, thread) where the thread is
   242→    # mutated in place as the chunks are consumed. After the stream ends,
   243→    # the thread carries a single AssistantMessageEvent with the full
   244→    # accumulated content + final usage — exactly what the rest of the
   245→    # bitácora system records for non-streaming runs.
   246→    chunks_iter, thread = harness.stream_with_audit(specialist, {"task": task})
   247→
   248→    total_in = 0
   249→    total_out = 0
   250→    total_cost = 0.0
   251→    chunks_received = 0
   252→
   253→    async for chunk in chunks_iter:
   254→        chunks_received += 1
   255→        if chunk.content:
   256→            console.print(chunk.content, end="", style="white")
   257→        total_in += chunk.usage.tokens_in
   258→        total_out += chunk.usage.tokens_out
   259→        total_cost += chunk.usage.cost_usd
   260→
   261→    if chunks_received == 0:
   262→        console.print("[yellow]No response received.[/yellow]")
   263→        sys.exit(1)
   264→
   265→    console.print()
   266→    console.print("[dim]---[/dim]")
   267→    console.print(f"[dim]Tokens: {total_in} in, {total_out} out. Cost: ${total_cost:.4f}[/dim]")
   268→
   269→    # Save bitácora — same format as ``arnes run`` (Thread.to_markdown)
   270→    # so streaming runs can be diffed against non-streaming runs.
   271→    ts = datetime.now().strftime("%Y%m%d-%H%M%S")
   272→    bitacora_path = f"bitacora-stream-{specialist.lstrip('@')}-{ts}.md"
   273→
   274→    Path(bitacora_path).write_text(thread.to_markdown(), encoding="utf-8")
   275→    console.print(f"\n[cyan]Bitácora saved to:[/cyan] {bitacora_path}")
   276→

   331→
   332→async def _run_benchmark(seeds: int, concurrent: int, manuals_dir: str | None, output: str) -> None:
   333→    """Run the basic benchmark suite and print/save results.
   334→
   335→    Pulls in :class:`arnes.benchmarks.BenchmarkRunner` lazily so the
   336→    CLI startup cost stays low — ``arnes --version`` doesn't pay for
   337→    importing the benchmark harness.
   338→    """
   339→    from pathlib import Path
   340→
   341→    from arnes.benchmarks import BenchmarkRunner
   342→    from arnes.benchmarks.suites.basic import BasicBenchmarkSuite
   343→
   344→    suite_manuals = Path(manuals_dir) if manuals_dir else None
   345→    suite = BasicBenchmarkSuite(manuals_dir=suite_manuals)
   346→
   347→    playbooks = suite.playbooks()
   348→    if not playbooks:
   349→        manuals_display = manuals_dir or "<repo>/manuals"
   350→        console.print(f"[red]✗[/red] No playbooks found in {manuals_display}")
   351→        sys.exit(1)
   352→
   353→    console.print(
   354→        Panel.fit(
   355→            f"[bold cyan]ARNES[/bold cyan] — Benchmark suite: {suite.name}\n"
   356→            f"  [dim]Playbooks:[/dim] {len(playbooks)}\n"
   357→            f"  [dim]Seeds:[/dim] {seeds}\n"
   358→            f"  [dim]Concurrent:[/dim] {concurrent}\n"
   359→            f"  [dim]Total runs:[/dim] {len(playbooks) * seeds}",
   360→            border_style="cyan",
   361→        )
   362→    )
   363→
   364→    runner = BenchmarkRunner()
   365→    seed_values = list(range(seeds))
   366→    with console.status("[cyan]Running benchmarks...[/cyan]"):
   367→        results = await runner.run_suite(suite, seeds=seed_values, concurrent=concurrent)
   368→
   369→    # Print results as a rich table.
   370→    table = Table(title=f"Benchmark Results — {suite.name} suite")
   371→    table.add_column("Playbook", style="cyan")
   372→    table.add_column("Runs", justify="right")
   373→    table.add_column("Success", justify="right")
   374→    table.add_column("Avg dur (s)", justify="right")
   375→    table.add_column("P95 dur (s)", justify="right")
   376→    table.add_column("Avg tokens", justify="right")
   377→    table.add_column("Avg cost", justify="right")
   378→
   379→    for m in results.per_playbook:
   380→        avg_tokens = m.avg_tokens_in + m.avg_tokens_out
   381→        table.add_row(
   382→            m.playbook_name,
   383→            str(m.runs),
   384→            f"{m.success_rate:.0%}",
   385→            f"{m.avg_duration_s:.4f}",
   386→            f"{m.p95_duration_s:.4f}",
   387→            str(avg_tokens),
   388→            f"${m.avg_cost_usd:.6f}",
   389→        )
   390→
   391→    console.print(table)
   392→    console.print(
   393→        f"\n[bold]Overall:[/bold] success={results.overall_success_rate:.0%}, "
   394→        f"avg_dur={results.overall_avg_duration_s:.4f}s, "
   395→        f"avg_tokens={results.overall_avg_tokens_in + results.overall_avg_tokens_out}, "
   396→        f"avg_cost=${results.overall_avg_cost_usd:.6f}"
   397→    )
   398→
   399→    # Save JSON results.
   400→    Path(output).write_text(results.to_json(), encoding="utf-8")
   401→    console.print(f"\n[cyan]Results saved to:[/cyan] {output}")

   464→async def _run_playbook(
   465→    playbook_path: str,
   466→    model: str,
   467→    budget: float,
   468→    mock: bool,
   469→    interactive: bool,
   470→    output: str | None,
   471→    stream: bool = False,
   472→) -> None:
   473→    """Execute a playbook and print results.
   474→
   475→    When ``stream=True``, uses :meth:`PlaybookExecutor.stream` to yield
   476→    step-level events as each step completes (instead of buffering the
   477→    whole run behind a spinner). The final ``PlaybookRunResult`` is
   478→    captured from the last yield for stats + bitácora persistence.
   479→    """
   480→    # Compile
   481→    try:
   482→        playbook = PlaybookCompiler.from_file(playbook_path)
   483→    except PlaybookCompileError as e:
   484→        console.print(f"[red]✗ Compile error:[/red]\n{e}")
   485→        sys.exit(1)
   486→
   487→    # The schema validator guarantees metadata is non-None after compile.
   488→    assert playbook.metadata is not None
   489→
   490→    console.print(
   491→        Panel.fit(
   492→            f"[bold cyan]ARNES[/bold cyan] — Executing playbook\n"
   493→            f"  [dim]Name:[/dim] {playbook.metadata.name}\n"
   494→            f"  [dim]Objective:[/dim] {playbook.metadata.objective}\n"
   495→            f"  [dim]Model:[/dim] {model}\n"
   496→            f"  [dim]Budget:[/dim] ${budget:.2f}",
   497→            border_style="cyan",
   498→        )
   499→    )
   500→
   501→    # Setup provider
   502→    if mock or model.startswith("mock/"):
   503→        provider: LLMProvider = _SchemaValidMockLLMProvider()
   504→    else:
   505→        provider = get_provider(model)
   506→
   507→    # Execute
   508→    executor = PlaybookExecutor(
   509→        provider=provider,
   510→        cost_budget=CostBudget(task_budget_usd=budget),
   511→        interactive=interactive,
   512→    )
   513→
   514→    if stream:
   515→        result = await _run_playbook_streaming(executor, playbook)
   516→    else:
   517→        with console.status("[cyan]Executing...[/cyan]"):
   518→            result = await executor.run(playbook)
   519→
   520→    # Print results
   521→    if result.success:
   522→        console.print("\n[green]✅ Manual executed[/green]")
   523→    else:
   524→        console.print("\n[red]❌ Execution failed[/red]")
   525→        if result.error:
   526→            console.print(f"  [red]Error:[/red] {result.error}")
   527→
   528→    # Stats
   529→    console.print(f"\n[dim]Steps executed:[/dim] {result.steps_executed}")
   530→    console.print(f"[dim]Steps failed:[/dim] {result.steps_failed}")
   531→    console.print(f"[dim]Duration:[/dim] {result.duration_s:.2f}s")
   532→    console.print(f"[dim]Tokens in/out:[/dim] {result.total_tokens_in}/{result.total_tokens_out}")
   533→    console.print(f"[dim]Total cost:[/dim] ${result.total_cost_usd:.4f}")
   534→
   535→    # Save bitácora
   536→    if output:
   537→        Path(output).write_text(result.to_markdown(), encoding="utf-8")
   538→        console.print(f"\n[cyan]Bitácora saved to:[/cyan] {output}")
   539→    else:
   540→        ts = datetime.now().strftime("%Y%m%d-%H%M%S")
   541→        default_path = f"bitacora-{playbook.metadata.name}-{ts}.md"
   542→        Path(default_path).write_text(result.to_markdown(), encoding="utf-8")
   543→        console.print(f"\n[cyan]Bitácora saved to:[/cyan] {default_path}")

   546→async def _run_playbook_streaming(
   547→    executor: PlaybookExecutor,
   548→    playbook: Playbook,
   549→) -> PlaybookRunResult:
   550→    """Run a playbook in streaming mode, printing step events as they arrive.
   551→
   552→    Uses :meth:`PlaybookExecutor.stream` to yield ``Event`` objects as each
   553→    step completes, then captures the final ``PlaybookRunResult`` from the
   554→    last yield. Best-effort: parallel branches stream in completion order,
   555→    not definition order.
   556→    """
   557→    from arnes.thread.events import (
   558→        RunCompletedEvent,
   559→        RunFailedEvent,
   560→        StepCompletedEvent,
   561→        StepFailedEvent,
   562→    )
   563→
   564→    final_result: PlaybookRunResult | None = None
   565→    console.print("[cyan]Streaming...[/cyan]\n[dim]---[/dim]")
   566→
   567→    async for item in executor.stream(playbook):
   568→        if isinstance(item, PlaybookRunResult):
   569→            final_result = item
   570→            break
   571→        # Print step-level transitions as they happen
   572→        if isinstance(item, StepCompletedEvent):
   573→            console.print(f"  [green]✓[/green] step_completed: {item.step_id}")
   574→        elif isinstance(item, StepFailedEvent):
   575→            console.print(
   576→                f"  [red]✗[/red] step_failed: {item.step_id} — {item.data.get('error', '')}"
   577→            )
   578→        elif isinstance(item, RunCompletedEvent):
   579→            console.print("  [green]✓[/green] run_completed")
   580→        elif isinstance(item, RunFailedEvent):
   581→            console.print(f"  [red]✗[/red] run_failed — {item.data.get('error', '')}")
   582→        else:
   583→            # Other event types (StepStarted, AssistantMessage, CostThreshold, etc.)
   584→            # are not printed to keep the streaming output focused on step transitions.
   585→            console.print(f"  [dim]·[/dim] {item.type.value}")
   586→
   587→    console.print("[dim]---[/dim]")
   588→    assert final_result is not None, "stream() must yield a final PlaybookRunResult"
   589→    return final_result

   592→class _SchemaValidMockLLMProvider(LLMProvider):
   593→    """Mock LLM provider that returns schema-valid JSON for each specialist.
   594→
   595→    Used by `arnes run --mock` for testing without network calls.
   596→    Detects which specialist is being invoked based on system prompt content.
   597→    """
   598→
   599→    def __init__(self) -> None:
   600→        self.call_count = 0
   601→
   602→    async def complete(
   603→        self,
   604→        messages: list[LLMMessage],
   605→        *,
   606→        model: str = "mock",
   607→        tools: list[dict[str, Any]] | None = None,
   608→        temperature: float = 0.0,
   609→        max_tokens: int | None = None,
   610→        response_format: dict[str, Any] | None = None,
   611→        response_schema: dict[str, Any] | None = None,
   612→        **kwargs: Any,
   613→    ) -> LLMResponse:
   614→        self.call_count += 1
   615→
   616→        # Detect specialist from system prompt
   617→        sys_msg = next((m for m in messages if m.role == "system"), None)
   618→        sys_content = sys_msg.content if sys_msg else ""
   619→
   620→        if "@planner" in sys_content:
   621→            content = '{"steps": [{"id": "s1", "specialist": "@coder", "input": {}}]}'
   622→        elif "@coder" in sys_content:
   623→            content = '{"files": [{"path": "out.py", "language": "python", "content": "# Mock code\\npass"}], "summary": "Mock implementation", "assumptions": [], "warnings": []}'
   624→        elif "@reviewer" in sys_content:
   625→            content = '{"verdict": "approve", "issues": [], "summary": "Mock review: looks good"}'
   626→        elif "@tester" in sys_content:
   627→            content = '{"test_files": [{"path": "test_mock.py", "content": "def test_mock():\\n    pass"}], "test_results": {"passed": 1, "failed": 0, "skipped": 0, "failures": []}, "summary": "Mock tests pass", "coverage_pct": 100.0}'
   628→        elif "@debugger" in sys_content:
   629→            content = '{"root_cause": "Mock root cause identified", "confidence": 0.95, "fix": {"file": "src/app.py", "line": 42, "original": "broken()", "fixed": "fixed()", "explanation": "Mock fix applied"}, "verification": "Run tests to verify", "alternative_causes": []}'
   630→        else:
   631→            content = '{"result": "mock output"}'
   632→
   633→        tokens_in = sum(len(m.content) // 4 for m in messages)
   634→        tokens_out = len(content) // 4
   635→
   636→        return LLMResponse(
   637→            content=content,
   638→            tool_calls=[],
   639→            usage=LLMUsage(
   640→                tokens_in=tokens_in,
   641→                tokens_out=tokens_out,
   642→                cost_usd=0.0,
   643→                model=model,
   644→                cached=False,
   645→            ),
   646→            model=model,
   647→        )
   648→
   649→    def list_models(self) -> list[str]:
   650→        return ["mock"]
   651→
   652→    async def stream_complete(
   653→        self,
   654→        messages: list[LLMMessage],
   655→        *,
   656→        model: str = "mock",
   657→        tools: list[dict[str, Any]] | None = None,
   658→        temperature: float = 0.0,
   659→        max_tokens: int | None = None,
   660→        response_format: dict[str, Any] | None = None,
   661→        response_schema: dict[str, Any] | None = None,
   662→        **kwargs: Any,
   663→    ) -> AsyncIterator[LLMResponse]:
   664→        """Default streaming: yield the full response in a single chunk.
   665→
   666→        Matches the ``MockLLMProvider.stream_complete`` contract. Real
   667→        token-by-token streaming is available in ``OllamaProvider`` and
   668→        ``LiteLLMProvider``; this mock yields the full response on first
   669→        iteration. AG-UI transport support lands in v0.2.

   671→        response = await self.complete(
   672→            messages,
   673→            model=model,
   674→            tools=tools,
   675→            temperature=temperature,
   676→            max_tokens=max_tokens,
   677→            response_format=response_format,
   678→            response_schema=response_schema,
   679→            **kwargs,
   680→        )
   681→        yield response
