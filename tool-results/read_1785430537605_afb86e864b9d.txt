     1→"""
     2→ARNES Playbook Executor — runs a compiled Playbook as a DAG.
     3→
     4→The executor:
     5→1. Walks the playbook steps in order.
     6→2. For each step, invokes the specialist or tool.
     7→3. Applies conditional branches (if/elif/else).
     8→4. Runs parallel branches concurrently.
     9→5. Handles retries with backoff.
    10→6. Pauses at HITL gates.
    11→7. Tracks budget via CostGuard.
    12→8. Appends events to the Thread.
    13→9. Returns a PlaybookRunResult with full trace.
    14→
    15→The executor is async and supports both fire-and-forget and streaming modes.
    16→"""

    56→def _is_docker_available() -> bool:
    57→    """Return True if the ``docker`` CLI is on PATH.
    58→
    59→    Used by the executor to decide whether to wire the Docker sandbox into
    60→    the default ``ToolContext``. This is a presence check only — it does NOT
    61→    verify the daemon is running or that ``arnes-sandbox:latest`` exists.
    62→    The ``ShellTool`` surfaces a clear error if either is missing at
    63→    execution time (``FileNotFoundError`` on ``docker run``).
    64→
    65→    We deliberately avoid probing the daemon (``docker info`` / ``docker
    66→    version``) here because:
    67→
    68→    1. It spawns a subprocess on every ``PlaybookExecutor`` construction,
    69→       which is wasteful for tests and high-throughput runs.
    70→    2. The daemon may be temporarily down even if the CLI is installed —
    71→       failing fast at construction time would prevent the user from
    72→       running non-shell playbooks that don't need Docker at all.
    73→    3. The ``ShellTool._execute_in_sandbox`` already handles the
    74→       ``FileNotFoundError`` case (docker binary missing) and returns a
    75→       actionable error message.
    76→    """
    77→    return shutil.which("docker") is not None

   101→class PlaybookExecutor:
   102→    """Executes a compiled Playbook.
   103→
   104→    Usage:
   105→        playbook = PlaybookCompiler.from_file("manuals/audit-pr.yaml")
   106→        executor = PlaybookExecutor()
   107→        result = await executor.run(playbook)
   108→        print(result.to_markdown())
   109→    """
   110→
   111→    def __init__(
   112→        self,
   113→        *,
   114→        provider: LLMProvider | None = None,
   115→        specialist_registry: SpecialistRegistry | None = None,
   116→        tool_registry: ToolRegistry | None = None,
   117→        cost_budget: CostBudget | None = None,
   118→        interactive: bool = False,
   119→        sandbox_enabled: bool | None = None,
   120→        sandbox_container: str | None = None,
   121→    ) -> None:
   122→        self.provider = provider or get_provider()
   123→        self.specialist_registry = specialist_registry or get_default_specialist_registry()
   124→        self.tool_registry = tool_registry or get_default_registry()
   125→        self.cost_budget = cost_budget or CostBudget()
   126→        self.interactive = interactive
   127→
   128→        # Sandbox wiring (Issue 1 / FIX-R3-SEC).
   129→        #
   130→        # Default behaviour:
   131→        #   - If ``sandbox_enabled`` is explicitly passed, honour it (caller
   132→        #     knows best — e.g. tests, Harness, MCP server).
   133→        #   - Otherwise, auto-detect: enable the Docker sandbox when the
   134→        #     ``docker`` CLI is on PATH, fall back to non-sandboxed mode
   135→        #     when it isn't.
   136→        #
   137→        # In non-sandboxed mode the ShellTool requires ``ARNES_DEV_MODE=1``
   138→        # as a double-gate before it will execute commands locally. We log
   139→        # a warning so operators know shell calls will be gated rather than
   140→        # sandboxed.
   141→        if sandbox_enabled is not None:
   142→            self._sandbox_enabled = sandbox_enabled
   143→            self._sandbox_container = sandbox_container or (
   144→                DEFAULT_SANDBOX_CONTAINER if sandbox_enabled else None
   145→            )
   146→        elif _is_docker_available():
   147→            self._sandbox_enabled = True
   148→            self._sandbox_container = sandbox_container or DEFAULT_SANDBOX_CONTAINER
   149→            logger.info(
   150→                "sandbox_docker_detected",
   151→                container=self._sandbox_container,
   152→                mode="docker-tier1",
   153→            )
   154→        else:
   155→            self._sandbox_enabled = False
   156→            self._sandbox_container = None
   157→            logger.warning(
   158→                "sandbox_docker_unavailable",
   159→                fallback="ARNES_DEV_MODE=1 required for local shell execution",
   160→                hint="Install Docker or build arnes-sandbox:latest to enable the sandbox",
   161→            )

   163→    async def run(
   164→        self,
   165→        playbook: Playbook,
   166→        *,
   167→        initial_input: dict[str, Any] | None = None,
   168→    ) -> PlaybookRunResult:
   169→        """Execute a playbook. Returns a PlaybookRunResult."""
   170→        # Use a mutable list to hold the thread (so step helpers can append)
   171→        thread_holder: list[Thread] = [Thread.create()]
   172→        start_time = time.monotonic()
   173→        outputs: dict[str, Any] = dict(playbook.variables)
   174→        if initial_input:
   175→            outputs.update(initial_input)
   176→
   177→        # Track cost guard at the run level (shared across steps)
   178→        cost_guard = CostGuard(self.provider, budget=self.cost_budget)
   179→
   180→        steps_executed = 0
   181→        steps_failed = 0
   182→        total_tokens_in = 0
   183→        total_tokens_out = 0
   184→        total_cost_usd = 0.0
   185→        aborted = False
   186→        abort_error: str | None = None
   187→
   188→        try:
   189→            for step in playbook.steps:
   190→                # Check if step should be skipped due to prior saltar_a
   191→                skip_until = outputs.get("__skip_steps_until", {})
   192→                if skip_until:
   193→                    # If we've reached the target step, clear the skip marker
   194→                    if step.id in skip_until:
   195→                        del skip_until[step.id]
   196→                        if not skip_until:
   197→                            del outputs["__skip_steps_until"]
   198→                        logger.info("saltar_a_reached", step_id=step.id)
   199→                        # Don't skip this step — execute it
   200→                    else:
   201→                        # Skip this step
   202→                        logger.info("step_skipped", step_id=step.id, reason="saltar_a")
   203→                        continue
   204→
   205→                # Execute the step
   206→                step_result = await self._execute_step(
   207→                    step,
   208→                    thread_holder,
   209→                    outputs,
   210→                    cost_guard,
   211→                    playbook,
   212→                )
   213→
   214→                if step_result["success"]:
   215→                    steps_executed += 1
   216→                    outputs[step.id] = step_result.get("output")
   217→                    # Track usage
   218→                    usage = step_result.get("usage", {})
   219→                    total_tokens_in += usage.get("tokens_in", 0)
   220→                    total_tokens_out += usage.get("tokens_out", 0)
   221→                    total_cost_usd += usage.get("cost_usd", 0.0)
   222→                else:
   223→                    steps_failed += 1
   224→                    # Check if step has si_no_se_cumple fallback
   225→                    if step.if_not_met:
   226→                        branch_result = await self._handle_conditional_branch(
   227→                            step,
   228→                            step.if_not_met,
   229→                            thread_holder,
   230→                            outputs,
   231→                            cost_guard,
   232→                            playbook,
   233→                        )
   234→                        if branch_result.get("terminate"):
   235→                            logger.info(
   236→                                "run_terminated_by_conditional",
   237→                                step_id=step.id,
   238→                                termination=branch_result["terminate"],
   239→                            )
   240→                            break
   241→                    else:
   242→                        # No fallback — abort
   243→                        thread_holder[0] = thread_holder[0].append(
   244→                            RunFailedEvent(
   245→                                thread_id=thread_holder[0].id,
   246→                                step_id=step.id,
   247→                                data={
   248→                                    "error": step_result.get("error", "Unknown error"),
   249→                                    "recoverable": False,
   250→                                },
   251→                            )
   252→                        )
   253→                        aborted = True
   254→                        abort_error = step_result.get("error")
   255→                        break
   256→
   257→            # Run completed (or aborted)
   258→            if not aborted:
   259→                thread_holder[0] = thread_holder[0].append(
   260→                    RunCompletedEvent(
   261→                        thread_id=thread_holder[0].id,
   262→                        data={
   263→                            "steps_executed": steps_executed,
   264→                            "duration_s": time.monotonic() - start_time,
   265→                            "total_tokens": total_tokens_in + total_tokens_out,
   266→                            "total_cost_usd": total_cost_usd,
   267→                        },
   268→                    )
   269→                )
   270→
   271→            return PlaybookRunResult(
   272→                thread=thread_holder[0],
   273→                success=not aborted,
   274→                steps_executed=steps_executed,
   275→                steps_failed=steps_failed,
   276→                duration_s=time.monotonic() - start_time,
   277→                total_tokens_in=total_tokens_in,
   278→                total_tokens_out=total_tokens_out,
   279→                total_cost_usd=total_cost_usd,
   280→                outputs=outputs,
   281→                error=abort_error,
   282→            )
   283→
   284→        except BudgetExceeded as e:
   285→            logger.error("budget_exceeded", error=str(e), spent=e.spent, budget=e.budget)
   286→            thread_holder[0] = thread_holder[0].append(
   287→                RunFailedEvent(
   288→                    thread_id=thread_holder[0].id,
   289→                    data={
   290→                        "error": f"Budget exceeded: {e}",
   291→                        "spent_usd": e.spent,
   292→                        "budget_usd": e.budget,
   293→                        "level": e.level,
   294→                    },
   295→                )
   296→            )
   297→            return PlaybookRunResult(
   298→                thread=thread_holder[0],
   299→                success=False,
   300→                steps_executed=steps_executed,
   301→                steps_failed=steps_failed,
   302→                duration_s=time.monotonic() - start_time,
   303→                total_tokens_in=total_tokens_in,
   304→                total_tokens_out=total_tokens_out,
   305→                total_cost_usd=total_cost_usd,
   306→                outputs=outputs,
   307→                error=str(e),
   308→            )

   314→    async def _execute_step(
   315→        self,
   316→        step: PlaybookStep,
   317→        thread_holder: list[Thread],
   318→        outputs: dict[str, Any],
   319→        cost_guard: CostGuard,
   320→        playbook: Playbook,
   321→    ) -> dict[str, Any]:
   322→        """Execute a single step (specialist, tool, or parallel branch)."""
   323→        thread_holder[0] = thread_holder[0].append(
   324→            StepStartedEvent(
   325→                thread_id=thread_holder[0].id,
   326→                step_id=step.id,
   327→                specialist=step.specialist or step.tool,
   328→                data={"step_id": step.id, "specialist": step.specialist or step.tool},
   329→            )
   330→        )
   331→
   332→        step_start = time.monotonic()
   333→
   334→        try:
   335→            # Parallel branch
   336→            if step.parallel:
   337→                result = await self._execute_parallel(
   338→                    step, thread_holder, outputs, cost_guard, playbook
   339→                )
   340→            # Specialist invocation
   341→            elif step.specialist:
   342→                result = await self._execute_specialist(
   343→                    step, thread_holder, outputs, cost_guard, playbook
   344→                )
   345→            # Tool invocation
   346→            elif step.tool:
   347→                result = await self._execute_tool(step, thread_holder, outputs, playbook)
   348→            else:
   349→                raise ValueError(f"Step '{step.id}' has no action defined")
   350→
   351→            # Drain middleware event sink BEFORE recording completion so
   352→            # that AssistantMessageEvent / CostThresholdEvent / CACHE_HIT /
   353→            # REFUSAL_TRIGGERED events emitted during the step appear in
   354→            # the thread before the StepCompletedEvent. The middleware
   355→            # creates events with a nil thread_id placeholder (it does not
   356→            # have access to the Thread); we patch the real thread_id and
   357→            # step_id here.
   358→            self._drain_middleware_events(thread_holder, cost_guard, step.id)
   359→
   360→            # Pull token / cost usage out of the step result so the
   361→            # StepCompletedEvent carries the per-step aggregate. The
   362→            # specialist returns ``usage`` as a model_dump() of LLMUsage
   363→            # (tokens_in, tokens_out, cost_usd, model, cached).
   364→            usage = result.get("usage") or {}
   365→
   366→            # Record completion (with token + cost accounting)
   367→            thread_holder[0] = thread_holder[0].append(
   368→                StepCompletedEvent(
   369→                    thread_id=thread_holder[0].id,
   370→                    step_id=step.id,
   371→                    specialist=step.specialist or step.tool,
   372→                    data={
   373→                        "step_id": step.id,
   374→                        "output": result.get("output"),
   375→                        "duration_s": time.monotonic() - step_start,
   376→                        "tokens_in": usage.get("tokens_in", 0),
   377→                        "tokens_out": usage.get("tokens_out", 0),
   378→                        "cost_usd": usage.get("cost_usd", 0.0),
   379→                    },
   380→                )
   381→            )
   382→
   383→            return result
   384→
   385→        except Exception as e:
   386→            logger.exception("step_failed", step_id=step.id, error=str(e))
   387→            # Drain any events emitted before the failure too — they are
   388→            # still useful for debugging (e.g. a CostThresholdEvent that
   389→            # fired right before the crash, or an AssistantMessageEvent
   390→            # from the failed LLM call).
   391→            self._drain_middleware_events(thread_holder, cost_guard, step.id)
   392→            thread_holder[0] = thread_holder[0].append(
   393→                StepFailedEvent(
   394→                    thread_id=thread_holder[0].id,
   395→                    step_id=step.id,
   396→                    specialist=step.specialist or step.tool,
   397→                    data={"step_id": step.id, "error": str(e), "retry": False},
   398→                )
   399→            )
   400→            return {"success": False, "error": str(e)}
   401→
   402→    def _drain_middleware_events(
   403→        self,
   404→        thread_holder: list[Thread],
   405→        cost_guard: CostGuard,
   406→        step_id: str,
   407→    ) -> None:
   408→        """Drain the middleware event sink and append events to the Thread.
   409→
   410→        Middleware (CostGuard, TokenOptimizer, VerificationLayer) emit
   411→        events to a shared ``_events`` list because they do not have direct
   412→        access to the Thread. The events are created with a nil thread_id
   413→        placeholder; here we patch the real thread_id and step_id and
   414→        append them to the Thread.
   415→
   416→        Idempotent: clears the sink after draining so the same events are
   417→        not appended twice.
   418→        """
   419→        events = getattr(cost_guard, "_events", None)
   420→        if not events:
   421→            return
   422→
   423→        thread_id = thread_holder[0].id
   424→        for event in events:
   425→            # Events are frozen pydantic models; use model_copy(update=...)
   426→            # to set the real thread_id and step_id without mutating the
   427→            # original (which may be referenced by middleware state).
   428→            patched = event.model_copy(
   429→                update={"thread_id": thread_id, "step_id": event.step_id or step_id}
   430→            )
   431→            thread_holder[0] = thread_holder[0].append(patched)
   432→
   433→        events.clear()

   435→    async def _execute_specialist(
   436→        self,
   437→        step: PlaybookStep,
   438→        thread_holder: list[Thread],
   439→        outputs: dict[str, Any],
   440→        cost_guard: CostGuard,
   441→        playbook: Playbook,
   442→    ) -> dict[str, Any]:
   443→        """Invoke a specialist."""
   444→        specialist = self.specialist_registry.get(step.specialist or "")
   445→        if not specialist:
   446→            return {
   447→                "success": False,
   448→                "error": f"Specialist '{step.specialist}' not registered. Available: {self.specialist_registry.list_names()}",
   449→            }
   450→
   451→        # Resolve input (may contain Jinja2-style template refs)
   452→        input_data = self._resolve_input(step.input, outputs)
   453→
   454→        # Build tool context — sandbox state is detected once at executor
   455→        # construction time (see __init__) and propagated to every
   456→        # specialist invocation so the ShellTool can pick the right
   457→        # execution path (Docker sandbox vs. gated local execution).
   458→        ctx = ToolContext(
   459→            thread_id=thread_holder[0].id,
   460→            step_id=step.id,
   461→            specialist=step.specialist,
   462→            working_dir=".",
   463→            sandbox_enabled=self._sandbox_enabled,
   464→            sandbox_container=self._sandbox_container,
   465→            budget_remaining_usd=(
   466→                (cost_guard.budget.effective_budget() or 0) - cost_guard.spent_usd
   467→            ),
   468→            metadata={"interactive": self.interactive},
   469→        )
   470→
   471→        # Use the cost_guard-wrapped provider
   472→        result = await specialist.run(
   473→            input_data,
   474→            ctx,
   475→            provider=cost_guard,
   476→            tool_registry=self.tool_registry,
   477→        )
   478→
   479→        success = result.get("success", False)
   480→        if not success and result.get("budget_exceeded"):
   481→            raise BudgetExceeded(
   482→                f"Budget exceeded during specialist '{step.specialist}' invocation",
   483→                spent=cost_guard.spent_usd,
   484→                budget=cost_guard.budget.effective_budget() or 0.0,
   485→                level="specialist",
   486→            )
   487→
   488→        # Propagate error message for non-budget failures
   489→        if not success and result.get("error"):
   490→            return {
   491→                "success": False,
   492→                "error": result["error"],
   493→                "output": None,
   494→                "usage": result.get("usage", {}),
   495→            }
   496→
   497→        return {
   498→            "success": success,
   499→            "output": result.get("output"),
   500→            "usage": result.get("usage", {}),
   501→        }
   502→
   503→    async def _execute_tool(
   504→        self,
   505→        step: PlaybookStep,
   506→        thread_holder: list[Thread],
   507→        outputs: dict[str, Any],
   508→        playbook: Playbook,
   509→    ) -> dict[str, Any]:
   510→        """Invoke a tool."""
   511→        tool = self.tool_registry.get(step.tool or "")
   512→        if not tool:
   513→            return {
   514→                "success": False,
   515→                "error": f"Tool '{step.tool}' not registered. Available: {self.tool_registry.list_names()}",
   516→            }
   517→
   518→        input_data = self._resolve_input(step.input, outputs)
   519→
   520→        ctx = ToolContext(
   521→            thread_id=thread_holder[0].id,
   522→            step_id=step.id,
   523→            metadata={"interactive": self.interactive},
   524→        )
   525→
   526→        result = await tool.execute(input_data, ctx)
   527→        return {
   528→            "success": result.success,
   529→            "output": result.output if result.success else None,
   530→            "error": result.error,
   531→        }

   533→    async def _execute_parallel(
   534→        self,
   535→        step: PlaybookStep,
   536→        thread_holder: list[Thread],
   537→        outputs: dict[str, Any],
   538→        cost_guard: CostGuard,
   539→        playbook: Playbook,
   540→    ) -> dict[str, Any]:
   541→        """Execute parallel sub-steps concurrently with ``asyncio.gather``.
   542→
   543→        Each sub-step gets its OWN thread_holder (a copy of the parent thread
   544→        at this point) so appends are isolated — no race on the shared
   545→        ``thread_holder[0]`` reference. After all sub-steps complete, their
   546→        event deltas are merged back into the parent thread_holder in
   547→        timestamp order. This preserves the immutable Thread pattern while
   548→        enabling true parallelism (the previous implementation ran sub-steps
   549→        sequentially in a for-loop, which was correct but not concurrent).
   550→
   551→        The shared ``cost_guard._events`` sink is drained by each sub-step's
   552→        own ``_execute_step`` call; because ``_drain_middleware_events`` is
   553→        synchronous, drains run atomically in the single-threaded asyncio
   554→        loop. An event emitted by sub-step B's specialist may end up drained
   555→        into sub-step A's thread_holder, but each event carries its own
   556→        ``step_id`` (set by ``_emit_assistant_message`` from the
   557→        ``ToolContext``) so the merged audit log is still correctly
   558→        attributed — the delta is just a container for the merge, not an
   559→        authoritative attribution.
   560→        """
   561→        if not step.parallel:
   562→            return {"success": False, "error": "No parallel steps defined"}
   563→
   564→        # Snapshot the parent thread so each sub-step's delta is exactly
   565→        # the events it appends beyond this point. The parent thread already
   566→        # has the outer StepStartedEvent(parallel) appended by _execute_step.
   567→        parent_snapshot = thread_holder[0]
   568→        parent_event_count = len(parent_snapshot.events)
   569→
   570→        # Each sub-step gets its OWN thread_holder starting from the snapshot.
   571→        # Appends to sub_holders[i][0] create new Thread objects (immutability),
   572→        # so sub-steps never share mutable state.
   573→        sub_holders: list[list[Thread]] = [[parent_snapshot] for _ in step.parallel]
   574→
   575→        # Run all sub-steps concurrently. return_exceptions=True so a single
   576→        # failure doesn't cancel the others — every sub-step runs to
   577→        # completion (or to its own failure) and we merge everything.
   578→        coros = [
   579→            self._execute_step(
   580→                sub_step,
   581→                sub_holders[i],
   582→                outputs,
   583→                cost_guard,
   584→                playbook,
   585→            )
   586→            for i, sub_step in enumerate(step.parallel)
   587→        ]
   588→        raw_results = await asyncio.gather(*coros, return_exceptions=True)
   589→
   590→        # Merge sub-step deltas back into the parent thread_holder in
   591→        # timestamp order. Each delta is the events the sub-step appended
   592→        # beyond the shared snapshot (StepStarted, AssistantMessage(s),
   593→        # StepCompleted/Failed for the sub-step, plus any middleware events
   594→        # it drained from the shared cost_guard sink).
   595→        merged_events: list[Event] = []
   596→        for holder in sub_holders:
   597→            merged_events.extend(holder[0].events[parent_event_count:])
   598→        # Stable sort by timestamp preserves intra-sub-step order for events
   599→        # with identical timestamps (the natural audit order within a single
   600→        # sub-step's lifecycle: Started → AssistantMessage → Completed).
   601→        merged_events.sort(key=lambda e: e.timestamp)
   602→
   603→        thread_holder[0] = thread_holder[0].extend(merged_events)
   604→
   605→        # Collect outputs and success status from each sub-step.
   606→        # Wrap output in {"output": ...} structure so templates like
   607→        # {{ steps.parallel.sub_step_id.output }} resolve correctly.
   608→        outputs_map: dict[str, dict[str, Any]] = {}
   609→        all_success = True
   610→        first_error: str | None = None
   611→
   612→        for sub_step, raw in zip(step.parallel, raw_results, strict=True):
   613→            if isinstance(raw, BaseException):
   614→                # A sub-step coroutine raised (shouldn't happen — _execute_step
   615→                # catches exceptions — but defend against executor bugs).
   616→                outputs_map[sub_step.id] = {
   617→                    "output": None,
   618→                    "success": False,
   619→                    "error": str(raw),
   620→                }
   621→                all_success = False
   622→                if first_error is None:
   623→                    first_error = str(raw)
   624→            else:
   625→                outputs_map[sub_step.id] = {
   626→                    "output": raw.get("output"),
   627→                    "success": raw.get("success", False),
   628→                    "error": raw.get("error"),
   629→                }
   630→                if not raw.get("success", False):
   631→                    all_success = False
   632→                    if first_error is None:
   633→                        first_error = raw.get("error")
   634→
   635→        result: dict[str, Any] = {
   636→            "success": all_success,
   637→            "output": outputs_map,
   638→        }
   639→        if first_error is not None:
   640→            result["error"] = first_error
   641→        return result
   642→
   643→    async def _handle_conditional_branch(
   644→        self,
   645→        step: PlaybookStep,
   646→        branch: ConditionalBranch,
   647→        thread_holder: list[Thread],
   648→        outputs: dict[str, Any],
   649→        cost_guard: CostGuard,
   650→        playbook: Playbook,
   651→    ) -> dict[str, Any]:
   652→        """Handle an if_not_met or conditional branch."""
   653→        thread_holder[0] = thread_holder[0].append(
   654→            ConditionalBranchEvent(
   655→                thread_id=thread_holder[0].id,
   656→                step_id=step.id,
   657→                data={
   658→                    "condition": branch.when or "if_not_met",
   659→                    "action": branch.action,
   660→                },
   661→            )
   662→        )
   663→
   664→        if branch.action == "call" and branch.specialist:
   665→            # Invoke the fallback specialist
   666→            fallback_step = PlaybookStep(
   667→                id=f"{step.id}__fallback",
   668→                specialist=branch.specialist,
   669→                input=branch.input or {},
   670→            )
   671→            result = await self._execute_specialist(
   672→                fallback_step, thread_holder, outputs, cost_guard, playbook
   673→            )
   674→            outputs[fallback_step.id] = result.get("output")
   675→            return {"terminate": None, "result": result}
   676→
   677→        if branch.action == "terminate":
   678→            return {"terminate": branch.terminate}
   679→
   680→        if branch.action == "skip" and branch.skip_to:
   681→            # skip_to means "jump TO this step" — skip all steps between
   682→            # current and target
   683→            skip_set = outputs.setdefault("__skip_steps_until", {})
   684→            skip_set[branch.skip_to] = True  # Will be cleared when we reach it
   685→            return {"terminate": None}
   686→
   687→        return {"terminate": None}

   693→    _TEMPLATE_RE = re.compile(r"\{\{\s*([^}]+)\s*\}\}")
   694→
   695→    def _resolve_input(
   696→        self,
   697→        input_value: dict[str, Any] | str | None,
   698→        outputs: dict[str, Any],
   699→    ) -> dict[str, Any]:
   700→        """Resolve Jinja2-style template references in input.
   701→
   702→        Example: "{{ pasos.leer_diff.salida }}" → outputs["leer_diff"]["output"]
   703→        Handles MULTIPLE templates in the same string.
   704→        """
   705→        if input_value is None:
   706→            return {}
   707→
   708→        if isinstance(input_value, str):
   709→            return {"__resolved_str__": self._resolve_template(input_value, outputs)}
   710→
   711→        if isinstance(input_value, dict):
   712→            resolved = {}
   713→            for k, v in input_value.items():
   714→                if isinstance(v, str):
   715→                    resolved[k] = self._resolve_template(v, outputs)
   716→                elif isinstance(v, dict):
   717→                    resolved[k] = self._resolve_input(v, outputs)
   718→                elif isinstance(v, list):
   719→                    resolved[k] = [
   720→                        self._resolve_template(item, outputs)
   721→                        if isinstance(item, str)
   722→                        else self._resolve_input(item, outputs)
   723→                        if isinstance(item, dict)
   724→                        else item
   725→                        for item in v
   726→                    ]
   727→                else:
   728→                    resolved[k] = v
   729→            return resolved
   730→
   731→        # Unreachable: ``input_value`` is typed as ``dict[str, Any] | str | None``
   732→        # and all three branches above return early. Kept as a defensive
   733→        # fallback for callers that bypass the type system (defence-in-depth).
   734→        return {"__input__": input_value}  # type: ignore[unreachable]
   735→
   736→    def _resolve_template(self, template: str, outputs: dict[str, Any]) -> Any:
   737→        """Resolve a template string, handling MULTIPLE {{ }} references.
   738→
   739→        Examples:
   740→            "{{ pasos.X.salida }}" → outputs["X"]["output"]
   741→            "Plan: {{ variables.nombre }} for PR {{ variables.pr_number }}" → "Plan: foo for PR 1234"
   742→            "Diff: {{ pasos.leer_diff.salida }}, Sec: {{ pasos.auditoria.salida }}" → "Diff: ..., Sec: ..."
   743→            "{{ }}" → "{{ }}"  (empty template body — returned as literal)
   744→        """
   745→        # Find ALL template references
   746→        matches = list(self._TEMPLATE_RE.finditer(template))
   747→
   748→        if not matches:
   749→            return template
   750→
   751→        # If the entire string is ONE template, return the resolved value (preserve type)
   752→        if len(matches) == 1 and matches[0].group(0) == template:
   753→            expr = matches[0].group(1).strip()
   754→            if not expr:
   755→                # Empty template body (e.g. "{{ }}") — return the original
   756→                # literal verbatim rather than re-rendering it with different
   757→                # whitespace.
   758→                return template
   759→            return self._resolve_expr(expr, outputs)
   760→
   761→        # Otherwise, interpolate ALL matches into the string
   762→        result = template
   763→        # Process in reverse order to keep indexes valid
   764→        for match in reversed(matches):
   765→            expr = match.group(1).strip()
   766→            if not expr:
   767→                # Empty template body — leave the original match untouched
   768→                continue
   769→            resolved = self._resolve_expr(expr, outputs)
   770→            result = result[: match.start()] + str(resolved) + result[match.end() :]
   771→
   772→        return result
   773→
   774→    def _resolve_expr(self, expr: str, outputs: dict[str, Any]) -> Any:
   775→        """Resolve a single template expression like 'steps.X.output'.
   776→
   777→        Supported forms:
   778→            "steps.X.output"        → outputs["X"]["output"] (or outputs["X"] if raw)
   779→            "steps.X.output.field"  → outputs["X"]["output"]["field"] (or outputs["X"]["field"])
   780→            "variables.X"           → outputs["X"]
   781→            "pasos.X.salida"        → legacy ES form of "steps.X.output"
   782→
   783→        Two important behaviors:
   784→
   785→        1. Only the LEADING prefix ("steps.", "variables.", "pasos.") is
   786→           stripped. Interior occurrences of these substrings (e.g. when a
   787→           step's output literally contains a key named "steps") are
   788→           preserved. This makes deep nesting like
   789→           ``{{ steps.s1.output.steps.s2.output }}`` work correctly.
   790→
   791→        2. For STEP references (``steps.*`` / legacy ``pasos.*``), the
   792→           "output" (and legacy "salida") segment is treated as a VIRTUAL
   793→           accessor: if the current dict has an "output" key, it is
   794→           dereferenced; if not, the segment is skipped (the current dict
   795→           IS the output). This makes ``{{ steps.X.output }}`` work whether
   796→           the step's output was stored raw (``outputs["X"] = output_dict``)
   797→           or wrapped (``outputs["X"] = {"output": output_dict, ...}``).
   798→
   799→           The virtual accessor does NOT apply to ``variables.*`` refs —
   800→           variables are user-defined and a missing key is a real error.
   801→        """
   802→        # Strip leading prefix only — NOT all occurrences. Stripping all
   803→        # occurrences of "steps." would corrupt paths whose intermediate
   804→        # dicts literally contain a key named "steps" (deep nesting).
   805→        stripped = expr.strip()
   806→        is_step_ref = False
   807→        for prefix in ("steps.", "pasos.", "variables."):
   808→            if stripped.startswith(prefix):
   809→                stripped = stripped[len(prefix) :]
   810→                # Only steps/pasos get the virtual "output" accessor;
   811→                # variables use strict key lookup.
   812→                is_step_ref = prefix in ("steps.", "pasos.")
   813→                break
   814→
   815→        parts = stripped.split(".")
   816→
   817→        current: Any = outputs
   818→        for raw_part in parts:
   819→            # Legacy ES translation per-segment
   820→            part = "output" if raw_part == "salida" else raw_part
   821→            if isinstance(current, dict):
   822→                if part in current:
   823→                    current = current[part]
   824→                elif is_step_ref and part == "output":
   825→                    # Virtual accessor: the current dict is itself the
   826→                    # step's output (stored raw). Skip this segment so
   827→                    # downstream segments (e.g. ".verdict") resolve against
   828→                    # the output dict directly.
   829→                    continue
   830→                else:
   831→                    return f"{{{{ {expr} }}}}"  # Leave template as-is if not found
   832→            else:
   833→                return f"{{{{ {expr} }}}}"
   834→
   835→        return current
